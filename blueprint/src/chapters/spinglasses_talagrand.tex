\chapter{The Sherrington--Kirkpatrick Model at High Temperature: The Talagrand Theory}\label{chap:sg-talagrand}

This chapter formalizes Sections~1.1 through 1.5 and, from Section~1.6, the finite Gibbs
cavity identities, the cavity path derivative (Lemma~1.6.3), its domination (Lemma~1.6.4), the decoupled endpoint (Lemma~1.6.2) and Propositions~1.6.5--1.6.6 of Michel Talagrand's \emph{Spin Glasses: A Challenge for Mathematicians} (Volume~I, Springer, 2010).
The mathematical narrative develops the high-temperature theory of the Sherrington--Kirkpatrick (SK) model:
\begin{enumerate}
\item The decoupled reference system, hypercube factorizations, exact first-moment calculations, and the first-moment upper bound on the free energy (Section~1.2).
\item Guerra's smart path Gaussian interpolation, the derivative formula with a negative square remainder, and Guerra's replica-symmetric (RS) upper bound minimized at the fixed-point equation $q = \mathbb{E}[\tanh^2(\beta z \sqrt{q} + h)]$ (Section~1.3).
\item R.~Lata\l{}a's high-temperature overlap concentration: the general second-order replica derivative formula along Gaussian interpolations, the monotonicity of the compensated exponential functional, exponential overlap concentration $\nu(\exp s N (R_{1,2} - q)^2) \le (1 - 2s - 4\beta^2)^{-1/2}$ for $\beta^2 < 1/4$, and the Aizenman--Lebowitz--Ruelle (ALR) high-temperature free energy formula $|p_N(\beta, h) - \operatorname{RS}(q)| \le K/N$ at a solution $q$ of the self-consistency equation (Section~1.4).
\item Spin decorrelation, asymptotic product measures on finite spin marginals, and high-dimensional random projections (Sections~1.4--1.5).
\end{enumerate}

\medskip\noindent\textbf{Conventions and Notation.}
The spin configuration space is $\Sigma_N = \{-1, +1\}^N$, represented in Lean as functions $\sigma : \mathrm{Fin}\,N \to \mathbb{Z}^\times$ where $\mathbb{Z}^\times = \{-1, +1\}$.
The Hamiltonian with inverse temperature $\beta \in \mathbb{R}$ and external field $h \in \mathbb{R}$ is
\[
-H_N(\sigma) = \frac{\beta}{\sqrt{N}} \sum_{1 \le i < j \le N} g_{ij} \sigma_i \sigma_j + h \sum_{i \le N} \sigma_i,
\]
where the disorder $(g_{ij})_{1 \le i < j \le N}$ consists of $\binom{N}{2} = \frac{N(N-1)}{2}$ independent standard Gaussian variables $\mathcal{N}(0, 1)$ indexed by $\mathrm{FinPairs}\,N = \{p \in \mathrm{Fin}\,N \times \mathrm{Fin}\,N \mid p.1 < p.2\}$.
The partition function is $Z_N(\beta) = \sum_\sigma \exp(-H_N(\sigma))$, the expected free energy is $p_N(\beta) = N^{-1}\mathbb{E}[\log Z_N(\beta)]$, $\langle\cdot\rangle$ denotes the Gibbs average (for several replicas, the product of independent copies), $\nu(\cdot)=\mathbb{E}[\langle\cdot\rangle]$ its average over the disorder, and the overlap of two configurations is $R_{1,2} = R(\sigma^1, \sigma^2) = N^{-1}\sum_{i \le N} \sigma_i^1 \sigma_i^2$.

\section{The SK Hamiltonian and Foundations (Talagrand Section~1.1)}\label{sec:sg-tal:foundations}

In Section~1.1, the SK Hamiltonian is defined on the discrete hypercube $\Sigma_N$.
Its covariance structure, Gaussian comparison properties, concentration of measure, and the thermodynamic limit are foundational.

\begin{remark}[Differences between the Lean statements and the book in Section~1.1]\label{rem:sg-tal:dev-11}
The theorem statements below are those of the Lean declarations. They differ from the setting of the text as follows.
\begin{itemize}
\item \emph{Slepian's lemma} is proved for abstract finite Gaussian families $(X_a,Y_a)_{a\in\alpha}$ on a probability space, with $X$ and $Y$ uncorrelated, not only for fields on $\Sigma_N$.
  Theorem~\ref{thm:sg-tal:slepian} is the book's general Proposition~1.3.3 for a $C^2$ test function with exponentially bounded value, gradient and Hessian instead of a $C^\infty$ function of moderate growth (Eq.~(A.18)). The free-energy comparison, the instance $F=-\log\sum_ae^{x_a+h_a}$ with the bias $h$ as a separate argument, is Theorem~\ref{thm:sg-tal:slepian-lse}. The Lean statements also ask that the two vectors be independent and jointly Gaussian (without loss of generality, since only the two covariance matrices matter). See Remark~\ref{rem:sg-tal:slepian-compare}.
\item \emph{Concentration} is proved for the mixed $p$-spin Hamiltonian with the constant $4\sum_p\beta_p^2$ in the exponent. The SK model is a special case, but the SK specialization is not stated as a separate Lean theorem.
\item \emph{The thermodynamic limit} is proved for the full-matrix Hamiltonian $N^{-1/2}\sigma\cdot(G\sigma)$ without external field, not for the pair disorder with a field $h$ used in Sections~1.2--1.4. The two disorders have different covariances ($NR^2$ versus $\frac N2(R^2-\frac1N)$), and no Lean statement relates the corresponding free energies; the high-temperature limit in Theorem~\ref{thm:sg-tal:alr} is therefore derived directly and does not use this theorem.
\item \emph{Covariance.} The SK pair field has covariance $\frac{N\beta^2}{2}(R^2-\frac1N)$ (Lean: \texttt{integral\_sk\_pair\_field\_mul\_eq\_overlap}).
\end{itemize}
\end{remark}

\begin{definition}[Disorder pairs and Hamiltonian]\label{def:sg-tal:hamiltonian}
  \lean{SpinGlasses.FinPairs, SpinGlasses.card_finPairs, SpinGlasses.cast_card_finPairs}
  \leanok
\talsource{Eq.~(1.12)}
For $N \in \mathbb{N}$, the type of disordered pairs is $\mathrm{FinPairs}\,N = \{p : \mathrm{Fin}\,N \times \mathrm{Fin}\,N \mid p.1 < p.2\}$.
Its cardinality is $|\mathrm{FinPairs}\,N| = \binom{N}{2} = \frac{N(N-1)}{2}$.
For disorder $g : \mathrm{FinPairs}\,N \to \mathbb{R}$ and configuration $\sigma : \mathrm{Fin}\,N \to \mathbb{Z}^\times$, the Hamiltonian is
\[
-H_N(g, \sigma) = \frac{\beta}{\sqrt{N}} \sum_{p \in \mathrm{FinPairs}\,N} g_p \sigma_{p.1} \sigma_{p.2} + h \sum_{i \le N} \sigma_i.
\]
\end{definition}

\begin{theorem}[Covariance of the SK Hamiltonian]\label{thm:sg-tal:covariance}
  \lean{SpinGlasses.integral_sk_pair_field_mul_eq_overlap}
  \leanok
\talsource{Eqs.~(1.4), (1.6); with $\beta$, Eq.~(1.63)}
Let $N\ge1$ and $\beta\in\mathbb R$. For any two configurations $\sigma^1, \sigma^2 \in \Sigma_N$, the Hamiltonian field has covariance
\[
\mathbb{E}\left[\left(\frac{\beta}{\sqrt{N}} \sum_{i < j} g_{ij} \sigma_i^1 \sigma_j^1\right)\left(\frac{\beta}{\sqrt{N}} \sum_{i < j} g_{ij} \sigma_i^2 \sigma_j^2\right)\right] = \frac{N \beta^2}{2}\left(R(\sigma^1, \sigma^2)^2 - \frac{1}{N}\right).
\]
\end{theorem}
\begin{proof}
  \leanok
By independence of the $g_{ij}$, the expectation is $\frac{\beta^2}{N} \sum_{i < j} \sigma_i^1 \sigma_j^1 \sigma_i^2 \sigma_j^2$.
Using $\sum_{i < j} x_i x_j = \frac{1}{2}((\sum_i x_i)^2 - \sum_i x_i^2)$ with $x_i = \sigma_i^1 \sigma_i^2$ (so $x_i^2 = 1$ and $\sum x_i = N R_{1,2}$),
the sum equals $\frac{1}{2}(N^2 R_{1,2}^2 - N) = \frac{N^2}{2}(R_{1,2}^2 - 1/N)$.
Multiplying by $\beta^2 / N$ gives $\frac{N\beta^2}{2}(R_{1,2}^2 - 1/N)$.
Formalized in \texttt{SpinGlasses/SKReplicaSymmetricBound.lean}.
\end{proof}

\begin{lemma}[Derivative of a Gaussian expectation along the trigonometric path]\label{lem:sg-tal:path-derivative}
  \lean{SpinGlasses.fderiv_apply_eq_sum_mul, SpinGlasses.hasDerivAt_integral_comp_trig}
  \leanok
\talsource{Lemma~1.3.1, Eq.~(1.40), with the trigonometric path in place of (1.38)}
Let $\kappa$ be a finite set, $(\Omega,P)$ a probability space and $X,Y:\kappa\to\Omega\to\mathbb R$ random variables such that the combined family $(X_l,Y_l)_{l\in\kappa}$ (indexed by $\kappa\sqcup\kappa$)
is a centered Gaussian vector with $\mathbb{E}[X_lY_k]=0$ for all $l,k$; write $A_{lk}=\mathbb{E}[X_lX_k]$ and $B_{lk}=\mathbb{E}[Y_lY_k]$. Let $F\in C^2(\mathbb R^\kappa)$ have \emph{exponential growth of order $\le2$}: there are constants
$C_0,c_0,c_1,c_2$ and $D_l$, $D_{kl}$ with
\[
|F(y)|\le C_0e^{c_0\|y\|_1},\qquad |\partial_lF(y)|\le D_le^{c_1\|y\|_1},\qquad |\partial_k\partial_lF(y)|\le D_{kl}e^{c_2\|y\|_1}\qquad(y\in\mathbb R^\kappa),
\]
where $\|y\|_1=\sum_l|y_l|$. Put $Z_s=\cos s\,X+\sin s\,Y$. Then $s\mapsto\mathbb{E}F(Z_s)$ is differentiable on $\mathbb R$, with
\[
\frac{d}{ds}\mathbb{E}F(Z_s)\Big|_{s=t}=\sin t\cos t\sum_{l,k\in\kappa}(B_{lk}-A_{lk})\,\mathbb{E}\,\partial_k\partial_lF(Z_t).
\]
\end{lemma}
\begin{proof}
  \leanok
  \uses{thm:sg:steinvec}
The path $Z_s$ replaces the book's $\sqrt t\,u+\sqrt{1-t}\,v$ of Eq.~(1.38) by $\cos s\,X+\sin s\,Y$: it has the same endpoints in the roles $v=X$ (at $s=0$) and $u=Y$ (at $s=\pi/2$), the covariance of $Z_s$ is $\cos^2s\,A+\sin^2s\,B$, and it is smooth on all of $\mathbb R$, so that no singular derivative has to be handled at the endpoints.
\emph{Step 1: differentiate under the expectation.} Pointwise, $\frac{d}{ds}F(Z_s)=\sum_lz_s(l)\,\partial_lF(Z_s)$ with $z_s(l)=-\sin s\,X_l+\cos s\,Y_l$ (chain rule, Eq.~(1.41)). The $\ell^1$ norm of $Z_s$ is at most that of $(X,Y)$, since $|\cos|,|\sin|\le1$,
and $|z_s(l)|\le\|(X,Y)\|_1\le e^{\|(X,Y)\|_1}$. Hence $|\sum_lz_s(l)\partial_lF(Z_s)|\le(\sum_lD_l)\,e^{(|c_1|+1)\|(X,Y)\|_1}$, a bound that does not depend on $s$ and is integrable, because a Gaussian vector has exponential moments of every order (\texttt{integrable\_comp\_of\_exp\_growth}). Dominated differentiation
(\texttt{hasDerivAt\_integral\_comp\_of\_exp\_growth}) gives $\frac{d}{ds}\mathbb{E}F(Z_s)=\sum_l\mathbb{E}[z_s(l)\,\partial_lF(Z_s)]$.
\emph{Step 2: Gaussian integration by parts.} The pair $(z_s,Z_s)$ is a centered jointly Gaussian family (a linear image of $(X,Y)$), and, because $\mathbb{E}[X_lY_k]=0$,
\[
\mathbb{E}[z_s(l)Z_s(k)]=-\sin s\cos s\,A_{lk}+\cos s\sin s\,B_{lk}=\sin s\cos s\,(B_{lk}-A_{lk}).
\]
Gaussian integration by parts (Eq.~(A.17), Theorem~\ref{thm:sg:steinvec}) applied to the function $\partial_lF$ gives $\mathbb{E}[z_s(l)\,\partial_lF(Z_s)]=\sum_k\mathbb{E}[z_s(l)Z_s(k)]\,\mathbb{E}\,\partial_k\partial_lF(Z_s)$.
Summing over $l$ gives the formula. The growth bounds on $\partial_lF$ and $\partial_k\partial_lF$ are what the integration by parts needs, to see that all expectations exist.
Formalized in \texttt{SpinGlasses/GaussianCovarianceInterpolation.lean}.
\end{proof}

\begin{theorem}[Slepian's lemma]\label{thm:sg-tal:slepian}
  \lean{SpinGlasses.integral_comp_le_of_covariance_le}
  \leanok
  \uses{lem:sg-tal:path-derivative}
\talsource{Proposition~1.3.3 (Slepian's lemma)}
Let $\kappa$ be a finite set, $(\Omega,P)$ a probability space and $X,Y:\kappa\to\Omega\to\mathbb R$ random variables such that $(X_l,Y_l)_{l\in\kappa}$ is a centered Gaussian vector with
$\mathbb{E}[X_lY_k]=0$ for all $l,k$ (that is, $X$ and $Y$ are independent centered Gaussian vectors). Assume
\begin{itemize}
\item $\mathbb{E}[X_l^2]=\mathbb{E}[Y_l^2]$ for all $l$, and $\mathbb{E}[X_lX_k]\le\mathbb{E}[Y_lY_k]$ for all $l\ne k$;
\item $F\in C^2(\mathbb R^\kappa)$ has exponential growth of order $\le2$ (Lemma~\ref{lem:sg-tal:path-derivative}) and $\partial_k\partial_lF\ge0$ for all $l\ne k$.
\end{itemize}
Then $\mathbb{E}F(X)\le\mathbb{E}F(Y)$. In the notation of the book, $u=Y$, $v=X$: if $\mathbb{E}u_i^2=\mathbb{E}v_i^2$, $\mathbb{E}u_iu_j\ge\mathbb{E}v_iv_j$ for $i\ne j$ and $\partial_i\partial_jF\ge0$ for $i\ne j$, then $\mathbb{E}F(u)\ge\mathbb{E}F(v)$.
\end{theorem}
\begin{proof}
  \leanok
\emph{Idea.} We do not compare $\mathbb{E}F(Y)$ and $\mathbb{E}F(X)$ directly. Instead we move continuously from the vector $X$ to the vector $Y$ along a path of Gaussian vectors, and show that the expectation of $F$ never decreases along the path. The covariance matrix of the vector at time $s$ is a mixture of $A$ and $B$, and the assumptions on $A$, $B$ and on the mixed second derivatives of $F$ are exactly what makes $\mathbb{E}F$ increase.

\emph{Notation.} Write $A_{lk}=\mathbb{E}[X_lX_k]$ and $B_{lk}=\mathbb{E}[Y_lY_k]$ (deterministic $\kappa\times\kappa$ matrices), and, for $s\in\mathbb R$,
\[
Z_s=\cos s\,X+\sin s\,Y,\qquad z_s=\frac{d}{ds}Z_s=-\sin s\,X+\cos s\,Y,\qquad \varphi(s)=\mathbb{E}F(Z_s).
\]

\textbf{Step 1: the two sides of the claim are the endpoints of the path.}
Since $\cos0=1$, $\sin0=0$, $\cos\frac\pi2=0$ and $\sin\frac\pi2=1$ we have $Z_0=X$ and $Z_{\pi/2}=Y$, hence
\[
\varphi(0)=\mathbb{E}F(X),\qquad \varphi(\tfrac\pi2)=\mathbb{E}F(Y).
\]
The inequality to be proved is therefore $\varphi(0)\le\varphi(\frac\pi2)$. The trigonometric path is used instead of the book's $\sqrt t\,u+\sqrt{1-t}\,v$ because it is smooth on all of $\mathbb R$ (no singular derivative at the endpoints), and it keeps the covariance of $Z_s$ equal to $\cos^2s\,A+\sin^2s\,B$ (Step 2, using $\mathbb{E}[XY]=0$), which interpolates between $A$ and $B$.

\textbf{Step 2: the vector is Gaussian along the whole path, and the covariances needed are computed.}
Because $(X,Y)$ is jointly Gaussian (Remark~\ref{rem:sg:hxy}), each of $Z_s$ and $z_s$ is a linear image of $(X,Y)$, so the pair $(z_s,Z_s)$ is again a centered jointly Gaussian family (\texttt{hasGaussianLaw\_sumElim\_lincomb}; centered by \texttt{integral\_lincomb\_eq\_zero}).
Expanding the product and using $\mathbb{E}[X_lY_k]=0$ for all $l,k$, so that every mixed term vanishes (\texttt{integral\_lincomb\_mul\_lincomb}), we get the two covariances
\[
\mathbb{E}[Z_s(l)Z_s(k)]=\cos^2s\,A_{lk}+\sin^2s\,B_{lk},
\qquad
\mathbb{E}[z_s(l)\,Z_s(k)]=-\sin s\cos s\,A_{lk}+\cos s\sin s\,B_{lk}=\sin s\cos s\,(B_{lk}-A_{lk}).
\]
The cross covariance $\mathbb{E}[z_s(l)Z_s(k)]$ is the quantity that appears in Gaussian integration by parts, and it is proportional to $B-A$.

\textbf{Step 3: $\varphi$ is differentiable, with derivative $\sum_l\mathbb{E}[z_s(l)\,\partial_lF(Z_s)]$.}
Pointwise in $\omega$, the chain rule gives $\frac{d}{ds}F(Z_s)=\sum_lz_s(l)\,\partial_lF(Z_s)$ (Eq.~(1.41) of the book). To differentiate under the expectation we need an integrable bound that is uniform in $s$.
Since $|\cos s|,|\sin s|\le1$, we have $\|Z_s\|_1\le\|(X,Y)\|_1$ and $|z_s(l)|\le\|(X,Y)\|_1\le e^{\|(X,Y)\|_1}$. The growth assumption $|\partial_lF(y)|\le D_le^{c_1\|y\|_1}$ then gives
\[
\Big|\sum_lz_s(l)\,\partial_lF(Z_s)\Big|\le\Big(\sum_lD_l\Big)e^{(|c_1|+1)\|(X,Y)\|_1},
\]
and the right side does not depend on $s$ and is integrable, because a Gaussian vector has exponential moments of every order (\texttt{integrable\_comp\_of\_exp\_growth}). The same growth assumption on $F$ shows that $F(Z_s)$ is integrable. By dominated differentiation (\texttt{hasDerivAt\_integral\_comp\_of\_exp\_growth}),
\[
\varphi'(s)=\sum_l\mathbb{E}\big[z_s(l)\,\partial_lF(Z_s)\big]\qquad\text{for every }s\in\mathbb R.
\]
In particular $\varphi$ is differentiable, hence continuous, on all of $\mathbb R$.

\textbf{Step 4: Gaussian integration by parts evaluates the derivative.}
Gaussian integration by parts (Eq.~(A.17) of the book, Theorem~\ref{thm:sg:steinvec}) says: if $(W,V)$ is a centered jointly Gaussian family, $G$ is differentiable with exponentially bounded value and partial derivatives, then
$\mathbb{E}[W\,G(V)]=\sum_k\mathbb{E}[WV_k]\,\mathbb{E}\,\partial_kG(V)$. We apply it with $W=z_s(l)$, $V=Z_s$ and $G=\partial_lF$, whose value and first partial derivatives $\partial_k\partial_lF$ are exponentially bounded by assumption. With the cross covariance from Step 2,
\[
\mathbb{E}\big[z_s(l)\,\partial_lF(Z_s)\big]=\sin s\cos s\sum_k(B_{lk}-A_{lk})\,\mathbb{E}\,\partial_k\partial_lF(Z_s),
\]
and summing over $l$,
\[
\varphi'(s)=\sin s\cos s\sum_{l,k}(B_{lk}-A_{lk})\,\mathbb{E}\,\partial_k\partial_lF(Z_s).
\]
This is Lemma~\ref{lem:sg-tal:path-derivative} (\texttt{hasDerivAt\_integral\_comp\_trig}), the analogue of Eq.~(1.40) of the book for the trigonometric path.
Note that $\mathbb{E}[z_s(l)\,\partial_lF(Z_s)]$ would be hard to bound directly, since $z_s$ is correlated with $Z_s$ and hence with $\partial_lF(Z_s)$; integration by parts replaces it by an expression involving only the deterministic covariances $A$ and $B$ and the second derivatives of $F$.

\textbf{Step 5: the derivative is nonnegative on $[0,\pi/2]$.} Use the assumptions term by term.
\begin{itemize}
\item \emph{Equal variances} $A_{ll}=B_{ll}$: the diagonal terms $l=k$ of the double sum vanish, since they carry the factor $B_{ll}-A_{ll}=0$. This is why no sign condition on the pure second derivatives $\partial_l^2F$ is needed.
\item \emph{Covariance order} $A_{lk}\le B_{lk}$ for $l\ne k$ and \emph{mixed derivatives} $\partial_k\partial_lF\ge0$ for $l\ne k$: each off-diagonal term is the product $(B_{lk}-A_{lk})\cdot\mathbb{E}\,\partial_k\partial_lF(Z_s)$ of two nonnegative factors, because the expectation of a pointwise nonnegative function is nonnegative.
\item \emph{The path factor:} for $s\in[0,\pi/2]$ both $\sin s\ge0$ and $\cos s\ge0$, so $\sin s\cos s\ge0$.
\end{itemize}
Hence, for $s\in[0,\pi/2]$,
\[
\varphi'(s)=\sin s\cos s\sum_{l\ne k}(B_{lk}-A_{lk})\,\mathbb{E}\,\partial_k\partial_lF(Z_s)\;\ge\;0.
\]
(Outside $[0,\pi/2]$ the sign of $\sin s\cos s$ can be negative, which is why the argument is restricted to this interval; only the endpoints $0$ and $\pi/2$ are needed.)

\textbf{Step 6: a nonnegative derivative gives monotonicity, hence the inequality.}
The function $\varphi$ is continuous on $[0,\pi/2]$ and differentiable on the interior with $\varphi'\ge0$ (Steps 3--5). By the mean value theorem, for any $0\le s<t\le\pi/2$ there is $\xi\in(s,t)$ with
\[
\varphi(t)-\varphi(s)=\varphi'(\xi)\,(t-s)\ge0,
\]
because $\varphi'(\xi)\ge0$ and $t-s>0$. So $\varphi$ is non-decreasing on $[0,\pi/2]$ (\texttt{monotoneOn\_of\_deriv\_nonneg}). Taking $s=0$ and $t=\pi/2$ gives $\varphi(0)\le\varphi(\frac\pi2)$, which by Step 1 is
\[
\mathbb{E}F(X)\le\mathbb{E}F(Y).
\]
In words: if the mixed second derivatives of $F$ are nonnegative, then $F$ is ``supermodular'', it rewards coordinates that move together, so a vector whose coordinates are more strongly correlated (larger $B$ against $A$, same variances) has the larger expectation of $F$.
Formalized in \texttt{SpinGlasses/GaussianCovarianceInterpolation.lean}.
\end{proof}

\begin{theorem}[Gaussian comparison for the free energy]\label{thm:sg-tal:slepian-lse}
  \lean{SpinGlasses.integral_log_sum_exp_le_of_covariance_le, SpinGlasses.sum_exp_add_pos,
        SpinGlasses.fderiv_log_sum_exp_single, SpinGlasses.fderiv_fderiv_log_sum_exp_single}
  \leanok
  \uses{thm:sg-tal:slepian}
\talsource{Proposition~1.3.3 (Slepian's lemma) for the function $F=-\log\sum_ae^{x_a+h_a}$; no counterpart in the book as a separate statement}
Let $\alpha$ be a finite nonempty set, $(\Omega,P)$ a probability space, and $X,Y:\alpha\to\Omega\to\mathbb R$ such that $(X_a,Y_a)_{a\in\alpha}$ is a centered Gaussian vector with $\mathbb{E}[X_aY_b]=0$ for all $a,b$, $\mathbb{E}[X_a^2] = \mathbb{E}[Y_a^2]$ and $\mathbb{E}[X_aX_b] \le \mathbb{E}[Y_aY_b]$ for all $a,b$.
Then for any bias $h:\alpha\to\mathbb R$,
\[
\mathbb{E}\Big[\log \sum_a \exp(Y_a + h_a)\Big] \le \mathbb{E}\Big[\log \sum_a \exp(X_a + h_a)\Big].
\]
\end{theorem}
\begin{proof}
  \leanok
\emph{Idea.} The statement is Slepian's lemma (Theorem~\ref{thm:sg-tal:slepian}) for one particular test function, the negative free energy. All the work is in checking the hypotheses on the test function: smoothness, growth, and the sign of the mixed second derivatives.
Put $Z(x)=\sum_be^{x_b+h_b}>0$, $F(x)=-\log Z(x)$ and let $w_a(x)=e^{x_a+h_a}/Z(x)$ be the Gibbs weights; they lie in $[0,1]$ (\texttt{gibbsWeight\_mem\_Icc}). We check the hypotheses of Theorem~\ref{thm:sg-tal:slepian}.
\begin{itemize}
\item \emph{Smoothness.} $x\mapsto Z(x)$ is $C^\infty$ and positive, and the logarithm is $C^\infty$ on $(0,\infty)$, so $F\in C^\infty\subseteq C^2$.
\item \emph{Gradient.} By the chain rule, $\partial_a\log Z=\partial_aZ/Z=e^{x_a+h_a}/Z=w_a$, that is, $\partial_aF=-w_a$ (\texttt{fderiv\_log\_sum\_exp\_single}). Hence $|\partial_aF|=w_a\le1$.
\item \emph{Hessian.} Write $w_l=e^{x_l+h_l}\,Z^{-1}$ and differentiate in the direction $k$ by the product rule: $\partial_ke^{x_l+h_l}=\mathbf 1[l=k]\,e^{x_l+h_l}$ and $\partial_kZ^{-1}=-Z^{-2}e^{x_k+h_k}$. Therefore
  \[
  \partial_k\partial_l\log Z=\partial_kw_l=\mathbf 1[l=k]\,w_l-w_lw_k=w_l\big(\mathbf 1[l=k]-w_k\big),
  \]
  that is, the Hessian of $\log Z$ is $\mathrm{diag}(w)-ww^{\top}$ (\texttt{fderiv\_fderiv\_log\_sum\_exp\_single}), and $\partial_k\partial_lF=w_lw_k-\mathbf 1[l=k]\,w_l$. Its absolute value is at most $1$, because $w_l\in[0,1]$ and $|\mathbf 1[l=k]-w_k|\le1$.
  \emph{The sign condition:} for $l\ne k$ the indicator vanishes and $\partial_k\partial_lF=w_lw_k\ge0$. The pure second derivatives $\partial_l^2F=w_l^2-w_l\le0$ have no sign condition imposed on them, and this is fine, because the diagonal terms drop out of the proof of Theorem~\ref{thm:sg-tal:slepian}.
  Note that the mixed derivatives of $\log Z$ itself are $-w_lw_k\le0$, the \emph{opposite} sign to the book's hypothesis. This is why the comparison is for $F=-\log Z$, and why the field with \emph{larger} correlations has the \emph{smaller} expected free energy.
\item \emph{Growth.} By the Lipschitz estimate for the free energy, $|\log\sum_ae^{x_a+h_a}|\le|\log\sum_ae^{h_a}|+\|x\|_1$ (\texttt{SpinGlasses.abs\_log\_sum\_exp\_le}, proved in the course of Theorem~\ref{thm:sg:expmoment}), and $\|x\|_1\le e^{\|x\|_1}$ and $1\le e^{\|x\|_1}$, we get $|F(x)|\le(|\log\sum_ae^{h_a}|+1)\,e^{\|x\|_1}$. The gradient and Hessian are bounded by $1=1\cdot e^{0\cdot\|x\|_1}$. So $F$ has exponential growth of order $\le2$.
\end{itemize}
The covariance hypotheses are those of Theorem~\ref{thm:sg-tal:slepian}: equal variances on the diagonal, and $\mathbb{E}[X_aX_b]\le\mathbb{E}[Y_aY_b]$ for all $a,b$, which contains the condition for $a\ne b$. Theorem~\ref{thm:sg-tal:slepian} gives $\mathbb{E}F(X)\le\mathbb{E}F(Y)$, that is, $-\mathbb{E}\log Z(X)\le-\mathbb{E}\log Z(Y)$, which is
$\mathbb{E}\log\sum_ae^{Y_a+h_a}\le\mathbb{E}\log\sum_ae^{X_a+h_a}$.

\emph{Sanity check.} Let $\alpha=\{1,2\}$, $X_1,X_2$ independent standard Gaussians and $Y_1=Y_2=Z$ standard Gaussian, with $Z$ independent of $X$. The hypotheses hold ($A=I$, $B=\begin{psmallmatrix}1&1\\1&1\end{psmallmatrix}$, so $B-A\ge0$ entrywise with zero diagonal). The left side is $\mathbb{E}[Z]+\log(e^{h_1}+e^{h_2})=\log(e^{h_1}+e^{h_2})$ and the right side
is $\mathbb{E}\log(e^{X_1+h_1}+e^{X_2+h_2})$, which is at least $\log(e^{h_1}+e^{h_2})$ by Jensen's inequality for the convex function $x\mapsto\log\sum_ae^{x_a+h_a}$ and $\mathbb{E}X=0$. This agrees with the theorem.
This is the form of Slepian's lemma used in the Guerra--Toninelli theorem (Theorem~\ref{thm:sg-tal:gt}).
Formalized in \texttt{SpinGlasses/GaussianCovarianceInterpolation.lean}.
\end{proof}

\begin{remark}[Comparison with the book's Proposition~1.3.3]\label{rem:sg-tal:slepian-compare}
The book states Slepian's lemma for a general test function: if $F:\mathbb R^M\to\mathbb R$ is infinitely differentiable with all partial derivatives of moderate growth (Eq.~(A.18): $F(x)e^{-a|x|^2}\to0$ for every $a>0$) and $\partial_i\partial_jF\ge0$ for all $i\ne j$, and $u,v$ are
centered jointly Gaussian vectors with $\mathbb{E}u_i^2=\mathbb{E}v_i^2$ for all $i$ and $\mathbb{E}u_iu_j\ge\mathbb{E}v_iv_j$ for all $i\ne j$, then $\mathbb{E}F(u)\ge\mathbb{E}F(v)$.
Theorem~\ref{thm:sg-tal:slepian} is this statement with the following differences.
\begin{itemize}
\item \emph{Regularity and growth.} The book asks $F\in C^\infty$ with all derivatives of moderate growth. Lean asks $F\in C^2$ (only derivatives up to order $2$ occur in the proof) with $F$ and its first and second partial derivatives bounded by $Ce^{c\|y\|_1}$. This is a restriction: functions that grow like $e^{a|y|^2}$ with small $a>0$ have moderate growth in the book's sense but are excluded. Polynomials, bounded functions and everything of exponential type are included. The restriction exists only because the Lean integration by parts is proved under exponential growth. The log-sum-exp function of Theorem~\ref{thm:sg-tal:slepian-lse} satisfies it.
\item \emph{Independence.} Lean assumes $(X,Y)$ jointly Gaussian and centered with $\mathbb{E}[X_aY_b]=0$, that is, $X$ and $Y$ independent Gaussian vectors. This is not a restriction, since the conclusion depends only on the two covariance matrices,
  and any two centered Gaussian laws can be realised as independent vectors on a product space; the book makes the same reduction before Eq.~(1.38), noting that one may assume $u$ and $v$ independent ``since we are comparing a function of the law of $u$ with a function of the law of $v$''. In the proof the independence is what makes the covariance of the interpolating vector $\cos^2s\,A+\sin^2s\,B$. The Lean theorem takes such a realisation as given.
\item \emph{Roles.} The Lean names are $v=X$ and $u=Y$, and the conclusion is written $\mathbb{E}F(X)\le\mathbb{E}F(Y)$. The covariance condition is stated for $l\ne k$ only (the diagonal is the equal-variance hypothesis).
\item \emph{The path.} The proof uses the path $\cos s\,X+\sin s\,Y$ in place of $\sqrt t\,u+\sqrt{1-t}\,v$, with the same endpoints; see Lemma~\ref{lem:sg-tal:path-derivative}.
\end{itemize}
The free-energy statement of Theorem~\ref{thm:sg-tal:slepian-lse} is the instance $F=-\log\sum_ae^{x_a+h_a}$: the mixed derivatives of $\log\sum e^{x}$ are $\le0$, so the book's lemma applies to its negative and the field with larger correlations has the smaller expected free energy.
\end{remark}

\begin{theorem}[Concentration of the free energy]\label{thm:sg-tal:concentration}
  \lean{SpinGlasses.mixedPSpin_free_energy_concentration}
  \leanok
\talsource{Theorem~1.3.4, Eq.~(1.47); Proposition~1.3.5, Eq.~(1.53); the SK instance is Eq.~(1.54)}

\begin{theorem}[The thermodynamic limit: Guerra--Toninelli theorem]\label{thm:sg-tal:gt}
  \lean{SpinGlasses.integral_log_sum_exp_add_le, SpinGlasses.sk_tendsto_free_energy}
  \leanok
\talsource{Theorem~1.3.9, Eqs.~(1.77)--(1.78)}
Lean proves this for the full-matrix SK Hamiltonian without external field, $H_N(G,\sigma)=N^{-1/2}\,\sigma\cdot(G\sigma)$ with $G$ an $N\times N$ matrix of independent standard Gaussians, and $F_N(\beta)=N^{-1}\mathbb{E}\big[\log\sum_\sigma\exp(\beta H_N(G,\sigma))\big]$.
\emph{Superadditivity (abstract form).} If $\xi$ is convex on $[-1,1]$ and $H_N$ are centered Gaussian fields on $\Sigma_N$ with $\mathbb{E}[H_N(\sigma)H_N(\tau)]=N\xi(R(\sigma,\tau))$, then for $N,M\ge1$
\[
\mathbb{E}\Big[\log\sum_\sigma e^{H_N(\sigma)}\Big]+\mathbb{E}\Big[\log\sum_\sigma e^{H_M(\sigma)}\Big]\le \mathbb{E}\Big[\log\sum_\sigma e^{H_{N+M}(\sigma)}\Big].
\]
\emph{Limit.} For every $\beta\in\mathbb R$,
\[
F_N(\beta)\longrightarrow \sup_{N\ge1}F_N(\beta)\qquad(N\to\infty).
\]
\end{theorem}
\begin{proof}
  \leanok
By Guerra--Toninelli cavity interpolation splitting a system of size $N+M$ into two independent subsystems of sizes $N$ and $M$; superadditivity of $N F_N$ and Fekete's lemma give the limit (the SK covariance is $N\xi(R)$ with $\xi(x)=\beta^2x^2$, which is convex).
Formalized in \texttt{SpinGlasses/GuerraToninelli.lean}.
\end{proof}

\section{Decoupled Systems, Hypercube Factorization and First Moments (Talagrand Section~1.2)}\label{sec:sg-tal:decoupled}

Section~1.2 of Talagrand establishes the non-interacting baseline: product factorization over the hypercube, local magnetizations under decoupled fields, exact calculation of $\mathbb{E}[Z_N]$, the Jensen upper bound, and the temperature derivative of the free energy.

\begin{remark}[Differences between the Lean statements and the book in Section~1.2]\label{rem:sg-tal:dev-12}
\begin{itemize}
\item The decoupled results are stated for an arbitrary finite index type $\iota$ (with $N=|\iota|$), not only $\mathrm{Fin}\,N$. The decoupled free energy requires the fields $Y_i$ to be identically distributed and $\log\cosh Y_{i_0}$ integrable, which is implicit in the book.

\end{itemize}
\end{remark}

\begin{lemma}[Hypercube product factorization]\label{lem:sg-tal:hypercube-prod}
  \lean{sum_units_int,
        hypercube_sum_prod_eq_prod_add,
        hypercube_sum_prod_erase_eq_prod_add}
  \leanok
\talsource{Eq.~(1.30)}
For a finite index set $\iota$, a commutative semiring $R$ and a family $a : \iota \to \mathbb{Z}^\times \to R$,
\[
\sum_{\sigma : \iota \to \mathbb{Z}^\times} \prod_{i \in \iota} a(i, \sigma_i) = \prod_{i \in \iota} (a(i, 1) + a(i, -1)),
\]
and, singling out a coordinate $j \in \iota$,
\[
\sum_{\sigma : \iota \to \mathbb{Z}^\times} \prod_{i \in \iota} a(i, \sigma_i) = (a(j, 1) + a(j, -1)) \prod_{i \neq j} (a(i, 1) + a(i, -1)).
\]
Also $\sum_{s\in\mathbb Z^\times}f(s)=f(1)+f(-1)$.
\end{lemma}
\begin{proof}
  \leanok
Interchange of sum and product over the finite product set (\texttt{Finset.prod\_univ\_sum}); the second identity splits off the factor $j$.
Formalized in \texttt{ForMathlib/Algebra/BigOperators/HypercubeProduct.lean}.
\end{proof}

\begin{theorem}[Decoupled partition function and local observables]\label{thm:sg-tal:decoupled-partition}
  \lean{Real.continuous_tanh,
        SpinGlasses.integrable_tanh_sq,
        SpinGlasses.decoupled_partition_function_eq,
        SpinGlasses.decoupled_partition_function_eq_mul_erase,
        SpinGlasses.decoupled_sum_spin_mul_exp_eq,
        SpinGlasses.decoupled_gibbs_spin_eq,
        SpinGlasses.decoupled_gibbs_replica_two_eq,
        SpinGlasses.decoupled_free_energy_eq}
  \leanok
\talsource{Eqs.~(1.29)--(1.36): partition function (1.32), free energy (1.33), local averages (1.34)--(1.36)}
For the decoupled cavity-free Hamiltonian $-H_0(\sigma) = \sum_{i \le N} \sigma_i Y_i$, the partition function factorizes as
\[
Z_0 = 2^N \prod_{i=1}^N \cosh Y_i.
\]
Splitting off the coordinate $j$,
\[
Z_0 = 2\cosh Y_j \prod_{i\neq j} 2\cosh Y_i,
\qquad
\sum_{\sigma}\sigma_j \exp\Big(\sum_i \sigma_i Y_i\Big) = 2\sinh Y_j \prod_{i\neq j} 2\cosh Y_i .
\]
The decoupled Gibbs expectation of a single spin satisfies $\langle \sigma_j \rangle_0 = \tanh Y_j$,
the two-replica overlap satisfies $\langle \sigma_j^1 \sigma_j^2 \rangle_0 = \tanh^2 Y_j$,
and, if the $Y_i$ are random variables with the same law and $\log\cosh Y_{i_0}$ is integrable, the decoupled free energy is
\[
\frac1N\,\mathbb{E}\Big[\log\sum_\sigma\exp\Big(\sum_i\sigma_iY_i\Big)\Big] = \log 2 + \mathbb{E}\big[\log\cosh Y_{i_0}\big].
\]
\end{theorem}
\begin{proof}
  \leanok
Apply Lemma~\ref{lem:sg-tal:hypercube-prod} with $a(i, \pm 1) = \exp(\pm Y_i)$ and use $\exp(Y) + \exp(-Y) = 2\cosh Y$ and $\exp(Y) - \exp(-Y) = 2\sinh Y$.
For the single-spin sum take $a(j,s)=s\,e^{sY_j}$ and $a(i,s)=e^{sY_i}$ for $i\ne j$; the ratio of the two sums is $\tanh Y_j$.
The integrability of $\tanh^2(Y)$ for any a.e.\ measurable $Y$ (it is bounded by $1$) is recorded separately because every expectation below uses it.
Formalized in \texttt{SpinGlasses/SKDecoupled.lean}.
\end{proof}

\begin{theorem}[Overlaps of the decoupled model]\label{thm:sg-tal:decoupled-overlap}
  \lean{SpinGlasses.decoupled_gibbs_overlap_eq,
        SpinGlasses.decoupled_replica_overlap_eq}
  \leanok
\talsource{Eqs.~(1.34)--(1.36); at a fixed point, Eq.~(1.104)}
Let $\iota$ be a finite nonempty index set of size $N$, $Y:\iota\to\mathbb R$ and let $\langle\cdot\rangle_0$ be the Gibbs average for $-H_0(\sigma)=\sum_i\sigma_iY_i$.
For two independent replicas the mean overlap is
\[
\Big\langle \frac1N\sum_{j}\sigma^1_j\sigma^2_j\Big\rangle_0=\frac1N\sum_{j}\tanh^2 Y_j .
\]
If moreover $Y_j$ are random variables on a probability space with the same law for every $j$, then
\[
\mathbb{E}[\frac1N\sum_{j}\tanh^2 Y_j]=\mathbb{E}[\tanh^2 Y_{j_0}],
\]
which for $Y_j=\beta z_j\sqrt q+h$ is the right-hand side $\mathbb{E}[\tanh^2(\beta z\sqrt q+h)]$ of the self-consistency equation (Eq.~(1.74)): at a fixed point $q$ the decoupled model has expected overlap exactly $q$.
\end{theorem}
\begin{proof}
  \leanok
Linearity of the integral over the product of the two (identical) Gibbs measures reduces the first claim to the single-site identity $\langle\sigma^1_j\sigma^2_j\rangle_0=\langle\sigma_j\rangle_0^2=\tanh^2Y_j$ of Theorem~\ref{thm:sg-tal:decoupled-partition}.
The second claim is linearity of the expectation together with $\mathbb{E}[\tanh^2Y_j]=\mathbb{E}[\tanh^2Y_{j_0}]$, which follows from identical distribution.
Formalized in \texttt{SpinGlasses/SKDecoupled.lean}.
\end{proof}

\begin{theorem}[Exact first moment of the SK partition function]\label{thm:sg-tal:first-moment}

\section{Guerra's Smart Path Interpolation and the RS Upper Bound (Talagrand Section~1.3)}\label{sec:sg-tal:guerra-rs}

Section~1.3 introduces Guerra's smart path interpolation between the fully interacting SK model and decoupled Gaussian spins, leading to the replica-symmetric upper bound.

\begin{remark}[Deviations from the book in Section~1.3]\label{rem:sg-tal:dev-13}
\begin{itemize}
\item \emph{Parametrization of the path.} Definition~\ref{def:sg-tal:smart-path} uses the trigonometric path $\cos s\,X+\sin s\,Y+h$, $s\in[0,\pi/2]$, with $s=0$ the SK model and $s=\pi/2$ the decoupled model, whereas the book uses $\sqrt t\,X+\sqrt{1-t}\,Y+h$ with $t=1$ the SK end; they are related by $t=\cos^2 s$ and all derivative formulas carry the factor $\sin s\cos s$ instead of the book's $dt$ (Remark~\ref{rem:sg-tal:book-path}).
\item \emph{Joint Gaussian space.} The pair disorder $g$ and the site disorder $z$ are realized on the product space $(\mathrm{FinPairs}\,N\to\mathbb R)\times(\mathrm{Fin}\,N\to\mathbb R)$ with the product measure; joint Gaussianity of $(X_\sigma,Y_\sigma)_\sigma$ is proved from independence (\texttt{HasGaussianLaw.sumElim\_prod\_of\_independent}).
\item \emph{The fixed point $q$: a definition in the book, a hypothesis in Lean.} In the book, $q$ is \emph{the} solution of $q=\mathbb{E}[\tanh^2(\beta z\sqrt q+h)]$ (Eq.~(1.74)), so statements such as ``$p_N\le\operatorname{SK}(\beta,h)$'' refer to one specific number. In Lean every theorem that uses $q$ (Theorem~\ref{thm:sg-tal:overlap-conc}, Theorem~\ref{thm:sg-tal:alr}, and the helper lemmas of Section~\ref{sec:sg-tal:latala-alr}) quantifies over an arbitrary real number $q$ together with the hypothesis \texttt{hfix} : $\mathbb{E}[\tanh^2(\beta\sqrt q\,z+h)]=q$. The existence of such a $q\in[0,1]$ is proved (Theorem~\ref{thm:sg-tal:fixed-point}); uniqueness is not. See Remark~\ref{rem:sg-tal:fixed-point-scope} for the intuition, the reasons, and the consequences. Guerra's bound itself (Theorem~\ref{thm:sg-tal:guerra-rs-bound}) holds for every $q\ge0$ and needs no fixed-point hypothesis.
\end{itemize}
\end{remark}

\begin{definition}[Guerra's smart path Hamiltonian]\label{def:sg-tal:smart-path}
\talsource{Eqs.~(1.59), (1.62), (1.72)}
For $s \in [0, \pi/2]$, independent standard Gaussians $(g_{ij})_{i<j}$ and $(z_i)_{i \le N}$, parameter $q \ge 0$, and bias $h$, the (trigonometric) smart path Hamiltonian is
\[
-H_s(\sigma) = \cos s \,\frac{\beta}{\sqrt{N}} \sum_{1 \le i < j \le N} g_{ij} \sigma_i \sigma_j + \sin s \,\beta \sqrt{q} \sum_{i \le N} z_i \sigma_i + h \sum_{i \le N} \sigma_i.
\]
The interpolating partition function is $Z_s = \sum_\sigma \exp(-H_s(\sigma))$ and $\varphi(s) = N^{-1}\mathbb{E}[\log Z_s]$. At $s = 0$ this is the SK model and at $s = \pi/2$ the decoupled model with fields $\beta\sqrt q\, z_i + h$.
\end{definition}

\begin{remark}[The book's definition of the smart path]\label{rem:sg-tal:book-path}
\talsource{Eqs.~(1.59), (1.62), (1.64), (1.65)}
Talagrand (Section~1.3) uses, for $t \in [0,1]$,
\[
-H_t^{\mathrm{book}}(\sigma)=\sqrt{t}\,\frac{\beta}{\sqrt{N}} \sum_{i<j} g_{ij}\sigma_i\sigma_j+\sqrt{1-t}\,\beta\sqrt{q}\sum_{i} z_i\sigma_i+h\sum_{i}\sigma_i ,
\]
with $Z_t=\sum_\sigma\exp(-H^{\mathrm{book}}_t(\sigma))$ and $\varphi_{\mathrm{book}}(t)=N^{-1}\mathbb{E}[\log Z_t]$; here $t=1$ is the SK model and $t=0$ the decoupled one. The two paths are related by $t=\cos^2 s$, i.e.\ $\sqrt t=\cos s$ and $\sqrt{1-t}=\sin s$, so $H^{\mathrm{book}}_{\cos^2 s}=H_s$ and $\varphi(s)=\varphi_{\mathrm{book}}(\cos^2 s)$. The Lean development uses $s$ because $s\mapsto\cos s$, $\sin s$ are smooth on all of $\mathbb R$, whereas $t\mapsto\sqrt t$ and $\sqrt{1-t}$ are not differentiable at the endpoints, which would force a separate argument for differentiability at $t=0,1$. By the chain rule $\frac{d}{ds}=-2\sin s\cos s\,\frac{d}{dt}$, so every book formula in $\frac{d}{dt}$ becomes the same formula in $\frac{d}{ds}$ with an extra factor $\sin s\cos s$ and the opposite orientation (the SK end is $s=0$, not $t=1$). Integrals $\int_0^1 \cdots\,dt$ become $\int_0^{\pi/2}\cdots\,2\sin s\cos s\,ds$.
In the book's variable the derivative formula (Eq.~(1.65)) reads
\[
\frac{d}{dt}\varphi_{\mathrm{book}}(t) = \frac{1}{2N} \left(\mathbb{E}\big[\langle U(\sigma, \sigma) \rangle_t\big] - \mathbb{E}\big[\langle U(\sigma^1, \sigma^2) \rangle_t\big]\right) = -\frac{\beta^2}{4} \mathbb{E}\big[\langle (R_{1,2} - q)^2 \rangle_t\big] + \frac{\beta^2}{4}(1 - q)^2 ,
\]
with the book's covariance difference $U=\mathbb{E}[uu]-\mathbb{E}[vv]$, where $u_\sigma=\frac{\beta}{\sqrt N}\sum_{i<j}g_{ij}\sigma_i\sigma_j$ and $v_\sigma=\beta\sqrt q\sum_iz_i\sigma_i$ (Eq.~(1.64)), i.e.\ $\frac1NU(\sigma^1,\sigma^2)=\frac{\beta^2}{2}\big((R_{1,2}-q)^2-q^2-\frac1N\big)$. The Lean development uses the opposite sign $V=-U$, because the pair field $X$ multiplies $\cos s$.
\end{remark}

\begin{lemma}[Covariance of the decoupled Gaussian cavity field]\label{lem:sg-tal:decoupled-covariance}
  \lean{SpinGlasses.integral_decoupled_gaussian_field_mul}
  \leanok
\talsource{Eq.~(1.62) for $v_\sigma$, entering Eq.~(1.64)}
For arbitrary finite real vectors $\sigma, \tau$ and independent standard Gaussian variables
$(z_i)$, the cavity-field part of Guerra's path satisfies
\[
\mathbb{E}\left[\left(\beta\sqrt{q}\sum_i z_i\sigma_i\right)
  \left(\beta\sqrt{q}\sum_i z_i\tau_i\right)\right]
= \beta^2 q \sum_i \sigma_i\tau_i,
\qquad q \geq 0.
\]
\end{lemma}
\begin{proof}
  \leanok
This is the covariance of two linear forms in an iid standard Gaussian vector.
The Lean statement is deliberately for arbitrary finite real vectors, not only hypercube spins,
so it is reusable for relaxed and spherical spin models.  Formalized in
\texttt{SpinGlasses/SKReplicaSymmetricBound.lean}.
\end{proof}

\begin{lemma}[Pair-indexed Gaussian covariance]\label{lem:sg-tal:pair-covariance}
  \lean{SpinGlasses.integral_normalized_gaussian_linear_form_mul,

For pair-indexed iid standard Gaussian disorder and two spin configurations, integrating out the

  \left(\frac{\beta}{\sqrt{N}}\sum_{i<j}

\begin{lemma}[Algebraic assembly of the smart-path covariance difference]
  \label{lem:sg-tal:path-covariance-algebra}
  \lean{SpinGlasses.sum_mul_comm_eq_dotProduct,
        SpinGlasses.inv_sqrt_mul_self,
        SpinGlasses.smart_path_half_covariance_difference_eq,
        SpinGlasses.sk_smart_path_half_covariance_difference_eq}
  \leanok
\talsource{Eqs.~(1.55), (1.64)}
For Ising configurations, the two concrete Gaussian fields have covariances
\[
C = \frac{N\beta^2}{2}\left(R(\sigma,\tau)^2 - \frac1N\right),
\qquad
D = \beta^2 q\, \sigma\mathbin{\cdot}\tau.
\]
Then their normalized half-difference is exactly
\[
\frac{C-D}{2N} = \frac{\beta^2}{4}
  \left((R(\sigma,\tau)-q)^2-q^2-\frac1N\right).
\]
\end{lemma}
\begin{proof}
  \leanok
The concrete theorem first invokes the already-formalized pair and cavity covariance calculations.
It then reuses the general algebraic bridge, substituting
$\sigma\mathbin{\cdot}\tau=N R(\sigma,\tau)$ and completing the square.
Two elementary facts are used throughout the covariance computations and are stated as separate lemmas: a linear form $\sum_k g_k a_k$ equals the dot product $a\cdot g$, and $(\sqrt n)^{-1}(\sqrt n)^{-1}=n^{-1}$ for $n\ge0$ (this is how the factor $N^{-1/2}$ in the Hamiltonian squares to the $1/N$ of the covariance).
\end{proof}

\begin{remark}[Scope of the fixed-point results: intuition, context, reasons and consequences]\label{rem:sg-tal:fixed-point-scope}
\talsource{Eqs.~(1.74)--(1.76), Proposition~1.3.8}
\emph{Context and intuition.} Guerra's bound (Theorem~\ref{thm:sg-tal:guerra-rs-bound}) is valid for every parameter $q\ge0$:
\[
p_N(\beta,h)\le \operatorname{RS}(q):=\log 2+\mathbb{E}[\log\cosh(\beta z\sqrt q+h)]+\tfrac{\beta^2}{4}(1-q)^2 .
\]
The parameter $q$ plays the role of a guessed overlap of two replicas. The bound is best when $q$ is chosen to minimize $\operatorname{RS}(q)$. The book computes (using the Gaussian integration by parts identity (1.75), $\frac{d}{dx}\mathbb{E}[f(\beta z\sqrt x+h)]=\frac{\beta^2}{2}\mathbb{E}[f''(\beta z\sqrt x+h)]$, with $f=\log\cosh$ and $(\log\cosh)''=1-\tanh^2$)
\[
\operatorname{RS}'(q)=\tfrac{\beta^2}{2}\big(q-\mathbb{E}[\tanh^2(\beta z\sqrt q+h)]\big),
\]
so the optimal $q$ is exactly a solution of (1.74). This also has a physical meaning: at the fixed point the decoupled model has expected overlap exactly $q$ (Theorem~\ref{thm:sg-tal:decoupled-overlap}), which is the reason the decoupled endpoint of the interpolation is the right comparison system. Proposition~1.3.8 then shows that the solution is unique when $\mathbb{E}[h^2]>0$ (Latala--Guerra, hard proof in the book's Appendix A.14), and for $\beta<1$ there is a short argument: with $f=\tanh^2$ one has $f''=2(1-t^2)(1-3t^2)\le2$ for $t=\tanh y$, so $\psi'\le\beta^2<1$, $\psi(x)-x$ is strictly decreasing and has at most one zero. From then on the book writes $q$ for this unique number and $\operatorname{SK}(\beta,h):=\operatorname{RS}(q)$.

\emph{What is formalized.} Only Theorem~\ref{thm:sg-tal:fixed-point}: a solution of (1.74) exists. \emph{What is not formalized:} (a) the derivative formula (1.75) for $x>0$ and hence the identity $\operatorname{RS}'(q)=\frac{\beta^2}{2}(q-\psi(q))$; (b) the consequence that some minimizer of $\operatorname{RS}$ over $q\ge0$ is a fixed point (a minimizer at $q=0$ forces $h=0$, in which case $0$ is itself a fixed point); (c) the bound $\psi'\le\beta^2$ and uniqueness of the fixed point for $\beta<1$; (d) the full Latala--Guerra monotonicity of Proposition~1.3.8. Note that ``every fixed point minimizes $\operatorname{RS}$'' is false in general (for $h=0$ and $\beta>1$ there are two fixed points), so the correct formal statement of (b) is about \emph{some} minimizer.

\emph{Why this was left out.} None of the later arguments uses minimality or uniqueness. Lata\l{}a's argument (Section~1.4) uses only the identity $q=\mathbb{E}[\tanh^2(\beta z\sqrt q+h)]$: through the decoupled overlap (Eq.~(1.104), $\nu_0(\sigma^1_i\sigma^2_i)=q$) it makes the decoupled model's overlap concentrate at $q$ (Proposition~\ref{prop:sg-tal:base-case}), and the smart-path derivative needs only the generic $q$ of Guerra's bound. Items (a)--(c) additionally need a differentiation-under-the-integral argument in the variance parameter $x$, with the $\sqrt x$ singularity at $x=0$ (handled by continuity, as the derivative is only needed for $x>0$), and (d) is a long separate argument that the book itself calls ``not really of importance''. The effort was therefore spent on the overlap concentration and the ALR formula.

\emph{Impact on the Lean code.} (i) Every Lean theorem that involves $q$ carries the hypothesis \texttt{hfix} and quantifies over all solutions; this is at least as strong as the book's statement, which is about one particular solution, and the theorems are not vacuous because Theorem~\ref{thm:sg-tal:fixed-point} produces a solution. (ii) For $\beta<1$ (hence in the whole regime $\beta^2<1/4$ where Theorems~\ref{thm:sg-tal:overlap-conc} and~\ref{thm:sg-tal:alr} are proved) the solution is unique in the book, so Lean's ``for every solution'' is the book's statement; but Lean does not prove this uniqueness. (iii) Consequently the limit in Theorem~\ref{thm:sg-tal:alr} is stated as $\operatorname{RS}(q)$ for the supplied $q$, and the Lean development does not define a function $(\beta,h)\mapsto\operatorname{SK}(\beta,h)$ or prove that the limit is independent of the choice of $q$. (iv) Nothing here affects the correctness of the proved statements; it only means that the identification ``$q$ is the minimizer of Guerra's bound, hence canonical'' lives in this blueprint text and not in Lean.
\end{remark}

\section{Lata\l{}a's Overlap Concentration and the ALR Formula (Talagrand Section~1.4)}\label{sec:sg-tal:latala-alr}

Section~1.4 proves R.~Lata\l{}a's exponential overlap concentration theorem and establishes the Aizenman--Lebowitz--Ruelle (ALR) high-temperature free energy formula.

\begin{remark}[Deviations from the book in Section~1.4]\label{rem:sg-tal:dev-14}
Every statement below is proved in Lean, but several proofs differ from Talagrand's, and a few statements are weaker or reorganized.
\begin{itemize}
\item \emph{Lemma~1.4.2: trigonometric form, no $\tfrac12$.} The derivative is taken in the trigonometric parameter $s$, with $U(\sigma,\tau)=\mathbb{E}[Y_\sigma Y_\tau]-\mathbb{E}[X_\sigma X_\tau]$ (no factor $1/2$) and prefactor $\sin s\cos s$. Via $t=\cos^2 s$ this is the book's Eq.~(1.90), with the opposite sign because $s=0$ is the SK end. Differentiation under $\mathbb{E}$ is justified explicitly by domination. The lemma is also proved for an observable that depends differentiably on $s$, which is what the compensated functional requires.
\item \emph{Eq.~(1.97) and Corollary~1.4.6 are not separate statements.} The case $n=2$ is expanded inside the proof of Lemma~\ref{lem:sg-tal:replica-decoupling}.
\item \emph{Lemma~1.4.5 by Chebyshev's inequality.} The book expands the exponential in a series, applies H\"older's inequality and uses replica symmetry. Lean applies Chebyshev's association inequality $\int a\int b\le\int ab$ for similarly ordered functions to $a=A$, $b=e^{uA}$, separately for each realization of the disorder. Lemma~1.4.5 is not stated on its own as an inequality between $\nu_t$-averages.
\item \emph{Corollary~1.4.7: direction and generality.} The compensated functional is nondecreasing in the trigonometric parameter $s$, equivalently non-increasing in the book's $t=\cos^2 s$. It is proved abstractly for any nonnegative symmetric overlap function $A$ with constant diagonal and $U=-\kappa(A-c)$, and then specialized to $A=(R_{1,2}-q)^2$.
\item \emph{Proposition~1.4.8.} The one-dimensional inequality $(\cosh u+q\sinh u)e^{-qu}\le e^{u^2/2}$ is obtained from Hoeffding's lemma (Mathlib) applied to an explicit two-point law, not from a direct calculus argument. The randomization $u\sim\mathcal N(0,2\lambda/N)$ is carried out as a Fubini exchange with a standard Gaussian, valid because the overlap is bounded.
\item \emph{Theorem~1.4.1 and the moment bounds.} The exponential bound is proved for the SK endpoint. The uniform bound $\nu_s((R_{1,2}-q)^2)\le K/N$ (Eqs.~(1.89), (1.103)) is derived from the compensated inequality and $e^x\ge1+x$ with an explicit constant; Eq.~(1.88) for all even moments is not formalized.
\item \emph{Theorem~1.4.10 by monotonicity.} Instead of integrating $\varphi'$ over $[0,\pi/2]$, Lean shows that $\varphi+c\sin^2$ is monotone for two choices of $c$ (using $0\le A\le K/N$), which gives the same two-sided $O(1/N)$ bound without any integrability check. The convergence $p_N\to\operatorname{SK}(\beta,h)$ is derived from this bound directly for the pair disorder with field, not from the thermodynamic limit of Theorem~1.3.9.
\item \emph{Hypotheses.} $q$ is a hypothesis (fixed-point equation), and the book's $\beta<1/2$ is stated as $\beta^2<1/4$.
\end{itemize}
\end{remark}

\begin{lemma}[Replica decoupling]\label{lem:sg-tal:replica-decoupling}
  \lean{MeasureTheory.integral_mul_integral_le_integral_mul,
        MeasureTheory.integral_comp_mul_integral_comp_le,
        MeasureTheory.integral_pi_fin_two,
        SpinGlasses.replica_decoupling_comb_nonneg}
  \leanok
\talsource{Lemma~1.4.5, Eq.~(1.98); Corollary~1.4.6, Eq.~(1.100); the $n=2$ derivative is Eq.~(1.97)}
Let $\mu$ be a Gibbs measure on a finite set $\alpha$, $A : \alpha\times\alpha\to[0,\infty)$ symmetric with constant diagonal, $U(\sigma,\tau) = -\kappa (A(\sigma,\tau) - c)$ with $\kappa\ge0$, and $u\ge 0$.
Let $\mu=\mu_f$ be the Gibbs measure of an energy $f:\alpha\to\mathbb R$, i.e.\ $\mu(\sigma)\propto e^{f(\sigma)}$, and $\Phi(\sigma^1,\sigma^2) = \exp(u A(\sigma^1,\sigma^2))$. Writing $\tau=(\sigma^1,\sigma^2,\sigma^3,\sigma^4)$ for the four replicas indexed by $\mathrm{Option}(\mathrm{Option}\,\mathrm{Fin}\,2)$, the combination of Lemma~\ref{lem:sg-tal:replica-deriv} (for $n=2$) and the moving-observable term for $u'=8\kappa\sin s\cos s$ is nonnegative:
\[
\int \Phi(\sigma^1,\sigma^2)\Big(\sum_{\ell,\ell'\le2} U(\sigma^\ell,\sigma^{\ell'}) - 4\sum_{\ell\le2} U(\sigma^\ell,\sigma^{3}) - 2\,U(\sigma^{3},\sigma^{3}) + 6\,U(\sigma^{3},\sigma^{4})\Big)\,d\mu^{\otimes4} + 8\kappa\int A(\sigma^1,\sigma^2)\,\Phi(\sigma^1,\sigma^2)\,d\mu^{\otimes2} \ge 0 .
\]
(For $A=(R_{1,2}-q)^2$ this contains the book's Lemma~1.4.5 and Corollary~1.4.6; see Remark~\ref{rem:sg-tal:dev-14}.)
\end{lemma}
\begin{proof}
\leanok
\emph{Differs from the book:} Talagrand proves Lemma~1.4.5 by expanding the exponential, H\"older's inequality and replica symmetry.
The Lean proof is shorter and works pointwise in the disorder: since $U$ is $-\kappa(A-c)$ with $A$ symmetric and of constant diagonal, the four-replica combination collapses to $-\kappa(2A_{12}-4A_{13}-4A_{23}+6A_{34})$ (the constants $c$ and the diagonal values cancel because the coefficients sum to zero).
The terms with $A_{13}, A_{23}$ are integrals of nonnegative functions; the term with $A_{34}$ factors by independence of the replicas $3,4$ from $1,2$ into $\langle\Phi\rangle\langle A\rangle$, and Chebyshev's association inequality (for two similarly ordered functions, $\int a\int b\le\int ab$; here $a=A$ and $b=e^{uA}$ with $u\ge0$) gives $\langle A\rangle\langle e^{uA}\rangle\le\langle A e^{uA}\rangle$.
Hence the combination is at least $\kappa(6\langle\Phi A\rangle - 6\langle\Phi\rangle\langle A\rangle)\ge0$.
\emph{Formalization note.} Lemma~1.4.5 is not stated on its own as an inequality between $\nu_t$-averages; only the combination above (Lemma~1.4.2 for $n=2$ plus the moving-observable term), which is what Corollary~\ref{cor:sg-tal:compensated-mono} uses, is a Lean statement.
Formalized in \texttt{SpinGlasses/SKLatalaCompensated.lean} (Chebyshev's inequality in \texttt{ForMathlib/MeasureTheory/Integral/ChebyshevAssociation.lean}).
\end{proof}

\leanok

\emph{Step 2 (replicas factorize).} For fixed fields $Y_i$, summing first over $\sigma^2$ and then over $\sigma^1$ with the hypercube factorization gives
$\langle\exp(u\sum_i(\sigma^1_i\sigma^2_i-q))\rangle=\prod_i e^{-uq}(\cosh u+\tanh^2(Y_i)\sinh u)$, because $\sum_{s,t=\pm1}e^{(s+t)y+u(st-q)}=4\cosh^2 y\;e^{-uq}(\cosh u+\tanh^2 y\sinh u)$.
\emph{Step 3 (disorder).} The coordinates $z_i$ are independent, so the expectation over $z$ is the $N$-th power of $e^{-uq}(\cosh u+\mathbb{E}[\tanh^2(Y)]\sinh u)=e^{-uq}(\cosh u+q\sinh u)$, which is at most $e^{Nu^2/2}$ by Step~1 (Eq.~(1.105)).
\emph{Step 4 (Gaussian linearization).} For $X=N(R_{1,2}-q)$ one has $\mathbb{E}[\langle e^{uX}\rangle]\le e^{Nu^2/2}$ for all $u$; writing $e^{\theta X^2/2}=\mathbb{E}_g[e^{\sqrt\theta\, g X}]$ with $g\sim\mathcal N(0,1)$, exchanging expectations (Fubini; $X$ is bounded) and using the bound with $u=\sqrt\theta g$ gives $\mathbb{E}[\langle e^{\theta X^2/2}\rangle]\le\mathbb{E}_g[e^{N\theta g^2/2}]=(1-N\theta)^{-1/2}$.
With $\theta=2\lambda/N$ this is the claim.
\emph{Formalization note.} The fixed-point equation is a hypothesis; only existence of a solution is formalized (Theorem~\ref{thm:sg-tal:fixed-point}, Remark~\ref{rem:sg-tal:fixed-point-scope}).
Formalized in \texttt{SpinGlasses/SKLatalaBase.lean}.
\end{proof}

\begin{theorem}[Exponential overlap concentration]\label{thm:sg-tal:overlap-conc}

\[

\]
Moreover, if $\beta^2<1/4$ then for every $s\in[0,\pi/2]$ the expected squared overlap under the smart-path energy at time $s$ satisfies (Eqs.~(1.89) and (1.103))
\[
\nu_s\big((R_{1,2}-q)^2\big)\le \frac{K}{N},\qquad K=\frac{(1/2-2\beta^2)^{-1/2}-1}{1/4-\beta^2}.

\begin{proof}
\leanok
Apply Corollary~\ref{cor:sg-tal:compensated-mono} with $\lambda=s+2\beta^2<1/2$ and then Proposition~\ref{prop:sg-tal:base-case}:
$\nu(\exp sN(R_{1,2}-q)^2)\le\nu_{\mathrm{dec}}(\exp\lambda N(R_{1,2}-q)^2)\le(1-2\lambda)^{-1/2}=(1-2s-4\beta^2)^{-1/2}$.
For the second claim take $\lambda=1/4+\beta^2$ and $s\in[0,\pi/2]$.
The compensated inequality at time $s$ gives $\nu_s(\exp(aN(R_{1,2}-q)^2))\le(1-2\lambda)^{-1/2}$ with $a=\lambda-2\beta^2\cos^2s\ge 1/4-\beta^2$.
From $e^x\ge1+x$ (packaged as the general inequality $1+c\,\mathbb{E}[X]\le\mathbb{E}[e^{cX}]$ on a probability space, applied first to the Gibbs average for each disorder and then to the disorder average) we get $1+aN\,\nu_t((R_{1,2}-q)^2)\le(1-2\lambda)^{-1/2}$, hence $\nu_s((R_{1,2}-q)^2)\le K/N$.

(ii) The book's assumption $\beta<1/2$ is stated as $\beta^2<1/4$ (in the first claim it is implied by $2s+4\beta^2<1$ with $s\ge0$).
(iii) Eq.~(1.88), the bound $(Kk/N)^k$ on all even moments, is not formalized; only the exponential bound and the second moment are.
Formalized in \texttt{SpinGlasses/SKOverlapConcentration.lean}.
\end{proof}

Formalized in \texttt{SpinGlasses/SKAizenmanLebowitzRuelle.lean}.
\end{proof}

\section{Spin Decorrelation, Asymptotic Product Measures and Projections (Talagrand Sections~1.4--1.5)}\label{sec:sg-tal:decorrelation}

The final component of the high-temperature theory connects overlap concentration to local spin independence and to the geometry of Gaussian projections.
Everything in this section is proved in Lean (Milestone T5).
The results on spin decorrelation and on the asymptotic product structure are proved first for an \emph{arbitrary} random Gibbs state on a finite configuration space whose law is symmetric between sites, and then specialized to the SK model with the overlap bound of Section~\ref{sec:sg-tal:latala-alr}.

\begin{remark}[Deviations from the book in Sections~1.4 (end) and~1.5]\label{rem:sg-tal:dev-15}
Each item states what differs from Talagrand's text and why the Lean proof was organized that way.
\begin{itemize}
\item \emph{General setting.} The book remarks that Proposition~1.4.14 only uses the symmetry between sites; Lean takes this literally. A \emph{site-symmetric random Gibbs state} consists of a finite set $\alpha$ of configurations, observables $\xi_1,\dots,\xi_N:\alpha\to[-1,1]$, weights $w_\omega(a)\ge0$ with $\sum_a w_\omega(a)=1$ depending measurably on the disorder $\omega$, and, for every permutation $\pi$ of the sites, a relabelling $a\mapsto\pi\cdot a$ of $\alpha$ with $\xi_i(\pi\cdot a)=\xi_{\pi(i)}(a)$ and a measure-preserving map $\varphi_\pi$ of the disorder with $w_{\varphi_\pi(\omega)}(a)=w_\omega(\pi\cdot a)$. For the SK model $\pi\cdot\sigma=\sigma\circ\pi$, and $\varphi_\pi$ relabels the Gaussian couplings $g_{ij}$ through the sorted pair $\{\pi(i),\pi(j)\}$ (Lean: \texttt{pairPerm}).
\emph{Why:} this is exactly the hypothesis the book's proof uses, it makes the statement reusable for other site-symmetric models, and it avoids any measurability theory of random measures: with finitely many configurations every Gibbs average is a finite sum of the weights, so measurability and integrability in $\omega$ follow from the continuity of the relevant expressions in the weight vector, which lies in the compact cube $[0,1]^\alpha$.

\item \emph{Proof of Proposition~1.4.14: pointwise in the disorder, site symmetry used last.}
In the book the chain is: (a) site symmetry turns $N(N-1)\cdots(N-p+1)\,\nu(\cdots)$ into a sum over tuples of distinct sites; (b) that sum is bounded by the sum over \emph{all} tuples, which equals $N^p\,\nu\big(R_{1,2}^{p-1}\,\dot{\boldsymbol\sigma}^1\!\cdot\dot{\boldsymbol\sigma}^2/N\big)$; (c) $R_{1,2}^{p-1}$ is replaced by $R_{1,2}^{p-1}-q^{p-1}$ because $\langle\dot{\boldsymbol\sigma}^1\!\cdot\dot{\boldsymbol\sigma}^2\rangle=0$; (d) the Cauchy--Schwarz inequality is applied for the measure $\nu$ (that is, after averaging over the disorder), together with a bound of $\nu\big((\dot{\boldsymbol\sigma}^1\!\cdot\dot{\boldsymbol\sigma}^2/N)^2\big)$.
Lean performs (b)--(d) \emph{before} averaging over the disorder, for each fixed weight vector $w$ (Lemma~\ref{lem:sg-tal:decorr-det}): the sum over all $(k+1)$-tuples of the squared centered correlation is at most $4kN^{k+1}\langle(R_{1,2}-q)^2\rangle$, where only the elementary inequality $|ab|\le 2a^2+b^2/8$ is used in place of Cauchy--Schwarz for $\nu$. Only afterwards is the disorder integrated, and site symmetry (a) is used to compare the $N(N-1)\cdots(N-k)$ tuples of distinct sites, which all carry the same expected value, with the sum over all tuples.
\emph{Why:} (i) The deterministic statement holds for every probability vector, so it needs no hypothesis on the law of the disorder and no square roots of expectations (hence no integrability bookkeeping for products of square roots); the only place where the randomness and the symmetry enter is the last step, which is a short counting argument. (ii) The result has the same shape as the book's, since in both cases the final bound is a constant times $\nu((R_{1,2}-q)^2)$: Cauchy--Schwarz followed by $\nu(b^2)^{1/2}\le c\,\nu((R_{1,2}-q)^2)^{1/2}$ gives $\nu\big(|R_{1,2}-q|\,|b|\big)\le c\,\nu((R_{1,2}-q)^2)$, and the weighted AM--GM inequality gives the same with a different constant. (iii) Nonnegativity of each term in the step ``distinct tuples $\le$ all tuples'' is automatic, because each term is the square of a Gibbs average.

\item \emph{The constant $16$ instead of $4$ in the four-replica step.}
The book writes $\langle(\mathbf R_{1,2}-R_{1,4}-R_{3,2}+R_{3,4})^2\rangle\le4\langle(R_{1,2}-q)^2\rangle$ after ``using that $(\sum_{i\le4}x_i)^2\le4\sum_{i\le4}x_i^2$''. Here $x_1,\dots,x_4$ are $\pm(R_{\ell\ell'}-q)$ for the four pairs of replicas, and the $q$'s cancel because the signs sum to zero. The quoted inequality gives $4\sum_{i\le4}x_i^2$, and each $\langle x_i^2\rangle$ equals $\langle(R_{1,2}-q)^2\rangle$ (replicas are independent and identically distributed), so the bound obtained from the quoted inequality is $4\cdot4=16$, not $4$. Lean proves exactly this: $\langle b^2\rangle\le16\langle(R_{1,2}-q)^2\rangle$ (Lemma~\ref{lem:sg-tal:decorr-det}, Step~4), and I did not try to improve the constant by other means.
\emph{Why it does not matter:} the constant enters only through $K(p)$, which is an unspecified number depending on $p$ alone in the book; the proof structure and the conclusion are unchanged.

\item \emph{Explicit constants and tuples instead of sets.}
$K(p)$ is the explicit number defined by $K(0)=0$, $K(p+1)=8p(p+1)^{p+1}+2K(p)$ (Lean: \texttt{spinDecorrConst}), obtained from $N^p\le p^p\,N(N-1)\cdots(N-p+1)$ for $p\le N$ (each factor $N-i$, $i<p$, is at least $N/p$) in place of the book's $\sup_{N\ge p}N^p/(N(N-1)\cdots(N-p+1))<\infty$. The induction is over tuples of distinct sites, because that is what the counting argument naturally controls; the statement for a set $S$ is deduced by enumerating $S$.
\emph{Why:} explicit constants make the statement checkable and let Theorem~\ref{thm:sg-tal:asymp-product} carry an explicit constant; a constant ``depending on $p$ only'' would require quantifying over a family of models inside an existential, which is awkward in Lean.

\item \emph{Total variation distance.} In Theorem~\ref{thm:sg-tal:asymp-product} the distance $\|G_{N,n}-\mu_n\|$ is the sum $\sum_\eta|G_{N,n}(\{\eta\})-\mu_n(\{\eta\})|$ (the book's Eq.~(A.79)), the first $n$ spins are the sites $0,\dots,n-1$, and $\mu_n(\{\eta\})=\prod_{i<n}(1+\eta_i\langle\sigma_i\rangle)/2$. The constant is $4^nK(n)$ times the overlap constant: the book's chain gives $2^n\sum_{I\subseteq[n]}\mathbb{E}(\langle\sigma_I\rangle-\prod_{i\in I}\langle\sigma_i\rangle)^2$, and each of the $2^n$ terms is bounded by $K(n)\nu((R_{1,2}-q)^2)$, using that $K$ is nondecreasing. \emph{Why:} the bound by the maximum over subsets is the simplest uniform one, and the book itself only claims a constant $K(n)$.

\item \emph{Section~1.5: a quantitative statement that is linear, not quadratic, in the overlap deviation.}
The book gives no formal statement in Section~1.5 (``we shall never need a formal statement''); it argues that $\mathbb{E}U^2$, $\mathbb{E}UV$ and $\mathbb{E}V^2$ all tend to the same Gaussian quantity because the Gaussian pair $(\mathbf g\cdot\mathbf x^1,\mathbf g\cdot\mathbf x^2)/\sqrt N$ has covariance $\simeq\left(\begin{smallmatrix}\rho&q\\q&\rho\end{smallmatrix}\right)$ and the law of a Gaussian vector depends on its covariance only. To turn this into a theorem one needs a quantitative dependence of $\mathbb{E}F(Z)$ on the covariance of $Z$. The standard tool, and the one proved in Lean (Lemma~\ref{lem:sg-tal:gaussian-interp}), is Gaussian interpolation: $\frac{d}{ds}\mathbb{E}F(\cos s\,X+\sin s\,Y)=\sin s\cos s\sum_{l,k}(B_{lk}-A_{lk})\,\mathbb{E}\,\partial_k\partial_lF$, which yields a bound that is \emph{linear} in the entries of the covariance difference, namely $\tfrac12\sum_{l,k}|B_{lk}-A_{lk}|\,\|\partial_k\partial_lF\|_\infty$. Here the covariance differences are the overlap deviations $R-q$ and $\|\mathbf x\|^2/N-\rho$ themselves, not their squares. Hence the proved bound is $\mathbb{E}_{\mathbf g}(U-V)^2\le C(f)\big(\nu|R_{1,2}-q|+\langle|\|\mathbf x\|^2/N-\rho|\rangle\big)$ with $C(f)=2(2M_1^2+M_0M_2)$, and for the SK model the Cauchy--Schwarz inequality $\nu|R_{1,2}-q|\le\nu((R_{1,2}-q)^2)^{1/2}$ gives the rate $\sqrt{K_{\mathrm{ov}}/N}$. The earlier planning document stated a bound of the form $L(f)^2\,\nu((R_{1,2}-q)^2)$ (rate $1/N$); that form is a guess of the plan and not a statement of the book. It would require the first-order terms in the covariance deviation to cancel in $\mathbb{E}U^2-2\,\mathbb{E}UV+\mathbb{E}V^2$, which the interpolation estimate does not show (each of the three terms is compared with the same reference value $T$ separately, so only the triangle inequality is available), and I did not establish such a cancellation. I do not claim the quadratic bound is false; it is simply not proved here.
\emph{Regularity of $f$.} The book allows ``a bounded, continuous function $f$''. The interpolation formula involves the second derivatives of $F(y_0,y_1)=f(y_0)f(y_1)$, so Lean requires $f\in C^2$ with $f$, $f'$, $f''$ bounded. Extending to bounded continuous $f$ would need an additional smoothing and approximation argument, which is not carried out.
\emph{Averaging.} The statement in Lean is for the mean square over $\mathbf g$ (and, for SK, over the disorder), not for ``typical $\mathbf g$ outside an exceptional set'' as in (1.137); the latter follows by Chebyshev's inequality but is not stated.

\item \emph{The earlier outline of this section is not formalized.} Before Milestone T5 this section contained a sketch based on the conditional exponential moment $\langle\exp(t\,\mathbf g\cdot(\sigma-\langle\sigma\rangle)/\sqrt N)\rangle\to\exp(t^2(1-q)/2)$ and a second-order Taylor argument. That sketch is not the argument of Talagrand's Section~1.5 (which compares $U$ and $V$ through Gaussian pairs), it was never a statement of the book, and it has been replaced by Theorem~\ref{thm:sg-tal:random-projection}. The exponential-moment convergence is not proved.

\item \emph{Hypotheses.} As in Section~\ref{sec:sg-tal:latala-alr}, the fixed point $q$ is a hypothesis (the equation $q=\mathbb{E}[\tanh^2(\beta\sqrt q\,z+h)]$ is assumed, and only the existence of a solution is proved), because Lean does not prove the uniqueness of the solution that makes ``the'' $q$ of the book well defined. The condition $\beta<1/2$ is stated as $\beta^2<1/4$, which is the same condition for $\beta\ge0$ and is the form in which the overlap bound is proved (it also allows $\beta<0$). The abstract results of this section need only $|q|\le1$ (for Proposition~\ref{prop:sg-tal:spin-decorr}) or $0\le q\le\rho$ (for Theorem~\ref{thm:sg-tal:random-projection}); the SK corollaries use $q\in[0,1]$ from the fixed-point equation.

\item \emph{Not formalized.} Models with non-Ising spins: Proposition~\ref{prop:sg-tal:spin-decorr} and the abstract part of Theorem~\ref{thm:sg-tal:random-projection} are stated for $[-1,1]$-valued observables, respectively for arbitrary vectors in $\mathbb{R}^N$ with an arbitrary self-overlap $\rho$, but Theorem~\ref{thm:sg-tal:asymp-product} uses $\sigma_i^2=1$ (the Walsh expansion of cylinders) and no non-Ising SK-type model is instantiated. Also not formalized are the uniqueness of the fixed point and Eq.~(1.88) for all even moments.
\end{itemize}
\end{remark}

\[
\sum_{u\in[N]^{k+1}}\Big\langle\dot\xi_{u_0}\,\xi_{u_1}\cdots\xi_{u_k}\Big\rangle^2\le 4k\,N^{k+1}\Big\langle (R_{1,2}-q)^2\Big\rangle ,
\]
where $\langle(R_{1,2}-q)^2\rangle=\sum_{a,b}w_aw_b(R(a,b)-q)^2$.
\end{lemma}
\begin{proof}
\leanok
\emph{Step 1 (replicas).} Expanding the squares, the left side is $\langle\!\langle(\sum_i\xi_i^1\xi_i^2)^k\sum_j\dot\xi_j^1\dot\xi_j^2\rangle\!\rangle$, because a sum over tuples of a product is a power of a sum.
\emph{Step 2 (centering).} Since $\langle\dot\xi_j\rangle=0$, $\langle\!\langle\sum_j\dot\xi^1_j\dot\xi^2_j\rangle\!\rangle=0$, so $(NR)^k$ may be replaced by $(NR)^k-(Nq)^k$.
\emph{Step 3 (Lipschitz bound).} For $|x|,|y|\le1$, $|x^k-y^k|\le k|x-y|$, and $|ab|\le2a^2+b^2/8$ with $a=R_{1,2}-q$, $b=\dot\xi^1\cdot\dot\xi^2/N$ bounds the left side by $kN^{k+1}\big(2\langle(R_{1,2}-q)^2\rangle+\tfrac18\langle b^2\rangle\big)$.
\emph{Step 4 (four replicas).} By direct expansion, $b=(R_{1,2}-q)-\langle R_{1,3}-q\rangle_3-\langle R_{3,2}-q\rangle_3+\langle R_{3,4}-q\rangle_{3,4}$, where the subscripts name the averaged replicas. Hence $b^2\le 4\sum$ of the squares of the four terms and, by Jensen's inequality for each averaged replica, $\langle b^2\rangle\le16\langle(R_{1,2}-q)^2\rangle$.
Combining the steps gives $kN^{k+1}(2+2)\langle(R_{1,2}-q)^2\rangle$.
Formalized in \texttt{SpinGlasses/SKSpinDecorrelation.lean}.
\end{proof}

\begin{proposition}[Spin decorrelation from overlap concentration]\label{prop:sg-tal:spin-decorr}

        SpinGlasses.integral_comp_weights_equiv,
        SpinGlasses.sq_sum_comp_equiv,
        SpinGlasses.integral_sq_centered_perm,
        Nat.pow_le_pow_mul_descFactorial,
        Fin.descFactorial_mul_le_sum,
        SpinGlasses.integral_sq_centered_le,

        SpinGlasses.spinDecorrConst_nonneg,
        SpinGlasses.spinDecorrConst_mono,
        SpinGlasses.integral_sq_prod_sub_prod_le,
        SpinGlasses.integral_sq_prod_sub_prod_le_finset}
  \leanok
\talsource{Proposition~1.4.14, Eqs.~(1.127)--(1.128)}

\[
\mathbb{E}\Big(\Big\langle\prod_{i\in S}\xi_i\Big\rangle-\prod_{i\in S}\langle\xi_i\rangle\Big)^2\le K(|S|)\,\nu\big((R_{1,2}-q)^2\big),
\]
with $K(0)=0$, $K(p+1)=8p(p+1)^{p+1}+2K(p)$. The same holds with $\prod_{l\le p}\xi_{u_l}$ for any $p$ distinct sites $u_1,\dots,u_p$.
\end{proposition}
\begin{proof}
\leanok
\emph{Step 1 (one centered site).} Let $u_0,\dots,u_k$ be distinct sites and $T(u)=\mathbb{E}\langle\dot\xi_{u_0}\xi_{u_1}\cdots\xi_{u_k}\rangle^2$. By site symmetry $T(\pi\circ u)=T(u)$ for every permutation $\pi$ (the squared correlation is a continuous functional of the weights, and the weights of $\varphi_\pi(\omega)$ are those of $\omega$ relabelled). Any two tuples of distinct sites differ by a permutation, so $T$ is constant on the $N(N-1)\cdots(N-k)$ injective tuples. Integrating Lemma~\ref{lem:sg-tal:decorr-det} over the disorder and dropping the non-injective tuples ($T\ge0$) gives $N(N-1)\cdots(N-k)\,T(u)\le4kN^{k+1}\nu((R_{1,2}-q)^2)$. With $N^{k+1}\le(k+1)^{k+1}N(N-1)\cdots(N-k)$ (each factor $N-i$ is at least $N/(k+1)$) we get $T(u)\le4k(k+1)^{k+1}\nu((R_{1,2}-q)^2)$.
\emph{Step 2 (induction on $p$).} Write $\langle\xi_{u_0}\cdots\xi_{u_k}\rangle-\prod_{l}\langle\xi_{u_l}\rangle=\langle\dot\xi_{u_0}\xi_{u_1}\cdots\xi_{u_k}\rangle+\langle\xi_{u_0}\rangle\big(\langle\xi_{u_1}\cdots\xi_{u_k}\rangle-\prod_{l\ge1}\langle\xi_{u_l}\rangle\big)$. Since $|\langle\xi_{u_0}\rangle|\le1$ and $(x+y)^2\le2x^2+2y^2$, the induction gives the constant $2\cdot4k(k+1)^{k+1}+2K(k)=K(k+1)$.
\emph{Step 3 (sets).} Enumerate $S$ and use that $\prod_{i\in S}f(i)=\prod_{l\le|S|}f(u_l)$.
Formalized in \texttt{SpinGlasses/SKSpinDecorrelation.lean}.
\end{proof}

\[

\begin{theorem}[A kind of central limit theorem for a finite Gibbs state]\label{thm:sg-tal:random-projection}

\[
\mathbb{E}\,\mathbb{E}_{\mathbf g}(U-V)^2\le2\,(2M_1^2+M_0M_2)\sqrt{\frac{K_{\mathrm{ov}}}N}.
\]
\end{theorem}
\begin{proof}
\leanok
Expand $(U-V)^2=U^2-2UV+V^2$. Each of $\mathbb{E}U^2=\sum_{a,a'}w_aw_{a'}\mathbb{E}f(\frac{\mathbf g\cdot x_a}{\sqrt N})f(\frac{\mathbf g\cdot x_{a'}}{\sqrt N})$, $\mathbb{E}UV$ and $\mathbb{E}V^2$ is an expectation $\mathbb{E}f(u_1\cdot\mathbf g+u_2\cdot\boldsymbol\xi)f(v_1\cdot\mathbf g+v_2\cdot\boldsymbol\xi)$ with $\boldsymbol\xi\sim\mathcal N(0,I_2)$ (for $UV$ and $V^2$ the average over $\xi$ is written as an integral over the coordinates of $\boldsymbol\xi$ by Fubini). Formalized in \texttt{SpinGlasses/SKRandomProjectionCLT.lean}.
\end{proof}

\begin{lemma}[Split of the SK pair sum and Gaussian disorder]

\begin{theorem}[The cavity path derivative (Lemma 1.6.3)]
\label{thm:sg-tal:cavity-path-derivative}
  \lean{SpinGlasses.hasDerivAt_integral_sk_cavity_path_angle,
        SpinGlasses.hasDerivAt_integral_sk_cavity_path_angle_offDiag,
        SpinGlasses.hasDerivAt_integral_sk_cavity_path_square,
        HasDerivAt.comp_arccos_sqrt_of_sin_cos,
        Real.arccos_sqrt_cos_sq}
  \leanok
\talsource{Lemma~1.6.3, Eq.~(1.151)}
Let $N,n\ge0$, $\beta,h\in\mathbb R$, $q\in[0,1]$, and let $f$ be any real function of $n$
replicas of $N+1$ Ising spins. For the energy
$E_s=\cos s\,X+\sin s\,Y+B$ of Lemma~\ref{lem:sg-tal:cavity-conditional-deriv}, whose base
energy $B$ is the old-pair energy with the prefactor $(N+1)^{-1/2}\beta$ and the field $h$,
put $\nu_t(f)=\mathbb E\langle f\rangle_{E_s}^{\otimes n}$ for $t=\cos^2 s$, the expectation
being over the old pairs, the new edges and the scalar Gaussian. Then for $0<t<1$
\[
\nu_t'(f)=\beta^2\sum_{\ell<\ell'\le n}\nu_t\bigl(f\varepsilon_\ell\varepsilon_{\ell'}
 (R^-_{\ell,\ell'}-q)\bigr)
-\beta^2 n\sum_{\ell\le n}\nu_t\bigl(f\varepsilon_\ell\varepsilon_{n+1}(R^-_{\ell,n+1}-q)\bigr)
+\beta^2\frac{n(n+1)}2\nu_t\bigl(f\varepsilon_{n+1}\varepsilon_{n+2}(R^-_{n+1,n+2}-q)\bigr),
\]
where $\varepsilon_\ell$ is the last spin of the replica $\ell$, $R^-$ is the truncated overlap
with the normalization $N+1$, and the Gibbs averages on the right run over $n+2$ replicas.
\end{theorem}
\begin{proof}
  \leanok
The angle form (the base energy being random is handled by
Lemma~\ref{lem:sg-tal:cavity-outer-derivative} specialized to the product Gaussian law on the
old pairs) is rewritten with the constant diagonal cancelled and the symmetric off-diagonal
pairs counted once (Lemma~\ref{lem:sg-tal:cavity-diagonal-cancel}). The chain rule for
$t=\cos^2 s=u$, $s=\arccos\sqrt u$, turns the factor $\sin s\cos s$ into $-\tfrac12$
(Lean: \texttt{HasDerivAt.comp\_arccos\_sqrt\_of\_sin\_cos}). The factor $-\tfrac12$ is pushed under
the three integrals; it turns the kernel $\beta^2\varepsilon\varepsilon'(q-R^-)$ into the book's
$\tfrac{\beta^2}2\varepsilon\varepsilon'(R^--q)$ and the covariance combination into the three
families of (1.151) (Lean: \texttt{neg\_half\_replica\_covariance\_offDiag},
\texttt{neg\_sk\_cavity\_covariance}). In the Lean statement the observable is not marginalized: the
three families are written as integrals over $n+2$ replicas of $f(\tau^1,\dots,\tau^n)$ times the
kernel, which is equal to the $\nu_t$ of the statement by
\texttt{integral\_pi\_option\_count\_tilted\_marginal}.
\emph{Not formalized:} the second form (1.152) with $R$ in place of $R^-$ (it follows from
Lemma~\ref{lem:sg-tal:cavity-overlap-split} since $\varepsilon_\ell^2=1$ and the three
families have total weight $\binom n2-n^2+\binom{n+1}2=0$).
\end{proof}

\begin{lemma}[Decoupled last spin in the finite Gibbs average]
\label{lem:sg-tal:decoupled-spin}

\[

=(\tanh Y)^{|I|}\;\mathbb E\,F(a^1,\dots,a^n),
\]

\end{lemma}

over the signs by the hypercube product formula. Each sign factor equals $\tanh Y$ if its replica is in
$I$ and $1$ otherwise, since $\sum_{\varepsilon=\pm1}\varepsilon e^{\varepsilon Y}/(e^Y+e^{-Y})=\tanh Y$.
\end{proof}

(symmetry between sites), and if $q=\mathbb E\tanh^2(\beta\sqrt q\,z+h)$ then
$\nu_0((\varepsilon_1\varepsilon_2-q)\varphi(R-q))=(1-q^2)\,\nu_0\bigl(\tfrac12[\varphi(D+x)-\varphi(D-x)]\bigr)$,
with $D=R^--q$ and $x=1/(N+1)$.
\end{theorem}
\begin{proof}
  \leanok
The first identity is Eq.~(1.157): the average of $(\sigma^1_i\sigma^2_i-q)\varphi(R-q)$ does not depend on $i$
(relabel the sites) and the $N+1$ of them add up to $(N+1)(R-q)\varphi(R-q)$. For the second, $R-q=D+\varepsilon x$ with
$\varepsilon=\varepsilon_1\varepsilon_2=\pm1$, so $\varphi(D+\varepsilon x)=S+\varepsilon T$ with $S,T$ the even and odd
parts of $\varphi(D+\cdot x)$, and $(\varepsilon-q)(S+\varepsilon T)=\varepsilon S-qS+T-q\varepsilon T$. Lemma~1.6.2 gives
$\nu_0(\varepsilon\,F)=q\,\nu_0(F)$ and $\nu_0(F)=\nu_0^-(F)$ for functions of the first $N$ coordinates, so the $S$
terms cancel and the $T$ terms add up to $(1-q^2)\nu_0(T)$.
\end{proof}

\begin{theorem}[Contraction of the overlap variance (Proposition 1.6.6)]
\label{thm:sg-tal:overlap-contraction}
  \lean{SpinGlasses.sk_overlap_pow_le_two_nu_zero,
        SpinGlasses.sk_overlap_pow_le_two_odd_part,
        SpinGlasses.sk_overlap_sq_le_two_div}
  \leanok
\talsource{Proposition~1.6.6, Eqs.~(1.157)--(1.159)}
Let $q\in[0,1]$ satisfy $q=\mathbb E\tanh^2(\beta\sqrt q\,z+h)$ and $16\beta^2e^{64\beta^2}\le\frac12$. For the
$(N+1)$-spin SK model, $\nu((R_{1,2}-q)^2)\le 2/(N+1)$. More generally,
$\nu((R-q)^{2k+2})\le2(1-q^2)\,\nu_0\bigl(\tfrac12[(D+x)^{2k+1}-(D-x)^{2k+1}]\bigr)$ for every $k\ge0$.
\end{theorem}
\begin{proof}
  \leanok
By Theorem~\ref{thm:sg-tal:overlap-odd-part}, $\nu((R-q)^{2k+2})=\nu(f)$ with
$f=(\varepsilon_1\varepsilon_2-q)(R-q)^{2k+1}$. By Theorem~\ref{thm:sg-tal:cavity-holder} with $n=2$, $p=(2k+2)/(2k+1)$ and
$r=2k+2$, and $|f|^p\le2^p(R-q)^{2k+2}$, $|\nu(f)-\nu_0(f)|\le16\beta^2e^{64\beta^2}\nu((R-q)^{2k+2})\le
\frac12\nu((R-q)^{2k+2})$, so $\nu((R-q)^{2k+2})\le2\nu_0(f)$ (this is the absorption of Eq.~(1.159)). The second part of
Theorem~\ref{thm:sg-tal:overlap-odd-part} with $\varphi(a)=a^{2k+1}$ evaluates $\nu_0(f)$. For $k=0$ the right-hand
side is $2(1-q^2)x\le2/(N+1)$.
\emph{Deviation:} the smallness condition has the constant $64$ rather than the book's $16$ in the exponent, as in
Theorem~\ref{thm:sg-tal:cavity-holder}, and $\beta_0$ is replaced by this explicit hypothesis.
\end{proof}

\begin{theorem}[Higher moments of the overlap (Proposition 1.6.7)]
\label{thm:sg-tal:overlap-moments}
  \lean{SpinGlasses.sk_overlap_moment_arith,
        SpinGlasses.abs_sk_overlap_sub_le_two,
        SpinGlasses.sk_overlap_moment_step,
        SpinGlasses.sk_overlap_moment_le}
  \leanok
\talsource{Proposition~1.6.7, Eq.~(1.160)}
Under the hypotheses of Theorem~\ref{thm:sg-tal:overlap-contraction}, for every $k\ge1$ and every $N$,
$\nu((R_{1,2}-q)^{2k})\le(64k/(N+1))^k$.
\end{theorem}
\begin{proof}
  \leanok
Induction on $k$; $k=1$ is Proposition~1.6.6. Let $x=1/(N+1)$. If $16(k+1)x>1$ the claim is trivial, because
$|R-q|\le2$ and $64(k+1)x>4$. Otherwise Theorem~\ref{thm:sg-tal:overlap-contraction} reduces
$\nu((R-q)^{2k+2})$ to $2(1-q^2)\nu_0(T)$ with $T=\frac12[(D+x)^{2k+1}-(D-x)^{2k+1}]$. By Lemma~\ref{lem:sg-tal:add-pow-estimates},
$0\le T\le(2k+1)x(|D|+x)^{2k}\le(2k+1)x(|R-q|+2x)^{2k}=:G$, since $D=R-q-\varepsilon x$. Lemma~1.6.4 for the two
replicas gives $\nu_0(G)\le e^{16\beta^2}\nu(G)\le2\nu(G)$, and the weighted Young inequality bounds
$\nu(G)\le(2k+1)x\,[\,2\nu((R-q)^{2k})+(1+4k)^{2k-1}(2x)^{2k}]$. With $\nu((R-q)^{2k})\le(64kx)^k$ and $16(k+1)x\le1$ the
right-hand side is at most $(64(k+1)x)^{k+1}$.
\emph{Deviation:} $\beta_0$ is replaced by the explicit smallness hypothesis; the book's
$\beta_0$ is not made explicit.
\end{proof}
