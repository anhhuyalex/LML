\chapter{The Sherrington--Kirkpatrick model and mixed $p$-spin models}\label{chap:sg}

This chapter formalizes Section~1.1 of Panchenko, \emph{The Sherrington--Kirkpatrick Model} (2013): the random
Hamiltonians of the Sherrington--Kirkpatrick (SK) model and of the pure and mixed $p$-spin models, their
covariance, the free energy and its smooth approximation of the ground-state energy, the monotonicity of the
free energy in the temperature, the zero-temperature limit, the Gibbs measure and the temperature derivative of
the free energy. The existence of the thermodynamic limit of the free energy (the Guerra--Toninelli theorem) is
the hypothesis \texttt{h\_lim} of the zero-temperature results of Section~\ref{sec:sg:thermo}; it is proved, for the
SK model and for the mixed $p$-spin models with only even $p$ and $p=1$, in Section~\ref{sec:sg:gt}. The Lean files are in
\texttt{SpinGlasses/} (modules \texttt{Basic}, \texttt{SKModel}, \texttt{FreeEnergy}, \texttt{ThermodynamicLimit},
\texttt{GibbsMeasure}, \texttt{MixedPSpin}), together with the general tools
\texttt{ForMathlib/Probability/Moments/GaussianPiLinearForm.lean}, \texttt{GaussianSeries.lean},
\texttt{ForMathlib/Probability/Independence/IntegralTsumMul.lean},
\texttt{ForMathlib/Topology/UniformSpace/LimitInterchange.lean} and
\texttt{ForMathlib/MeasureTheory/MeasurableSpace/Units.lean}.

The second part of the chapter (Sections~\ref{sec:sg:stein}--\ref{sec:sg:countable}) formalizes the Gaussian tools of
Section~1.2: Stein's lemma and Gaussian integration by parts for jointly Gaussian vectors, the replica covariance
formula for the Gibbs measure (Panchenko's Lemma~1.1 and Exercise~1.1) and its extension to a countable
configuration space (Lemma~1.2). The Lean files are \texttt{ForMathlib/MeasureTheory/Measure/PiUpdate.lean},
\texttt{ForMathlib/Probability/Moments/GaussianStein.lean}, \texttt{GaussianVector.lean}, \texttt{GaussianExp.lean},

\medskip\noindent\textbf{Notation.}
$\iota$ is a finite index set and $N=|\iota|$ (the number of spins; $\iota=\mathrm{Fin}\,N$ in the SK model, but all
statements hold for any finite type). A \emph{configuration} is $\sigma\in\Sigma_\iota=\{-1,+1\}^\iota$, a function
$\sigma:\iota\to\mathbb Z^\times$ in Lean (here $\mathbb Z^\times=\{\pm1\}$), read as the real vector
$(\sigma_i)_{i\in\iota}$ when sums are formed. The \emph{overlap} of two configurations is
$R(\sigma^1,\sigma^2)=N^{-1}\sum_i\sigma^1_i\sigma^2_i=N^{-1}\,\sigma^1\!\cdot\!\sigma^2$. The disorder
$G=(g_{ij})$ has i.i.d.\ $\mathcal N(0,1)$ entries, the law
\texttt{Measure.pi fun \_ => Measure.pi fun \_ => gaussianReal 0 1}, and the SK Hamiltonian is
$H_N(\sigma)=N^{-1/2}\sum_{i,j}g_{ij}\sigma_i\sigma_j=N^{-1/2}\,\sigma^\top G\sigma$. For an energy
$H:\Sigma\to\mathbb R$ and $\beta>0$ the partition function is $Z(\beta)=\sum_\sigma e^{\beta H(\sigma)}$ and the free
energy is $F_N(\beta)=N^{-1}\mathbb E\log Z_N(\beta)$.

\medskip\noindent\textbf{Which definitions are not introduced.}
Following the repository's convention this chapter introduces \emph{no} new Lean definition. The overlap, the
Hamiltonians, the partition function, the free energy, the covariance function $\xi$, the Gibbs weights and the
Gibbs measure are all written out explicitly in the statements: the overlap is
$N^{-1}\,\sigma\cdot\tau$ (an expression in \texttt{dotProduct}), the partition function is the sum
$\sum_\sigma e^{\beta H(\sigma)}$, the maximum is the repository's \texttt{Function.max}, and the Gibbs measure is
Mathlib's exponentially tilted counting measure \texttt{Measure.count.tilted} with tilt $\beta H$. Hamiltonians appear as an arbitrary function $H$ together with its
defining equation as a hypothesis (written \texttt{hH}).

\section{Configurations and overlap}\label{sec:sg:basic}

\begin{lemma}[Overlap]\label{lem:sg:overlap}
  \lean{SpinGlasses.card_inv_mul_dotProduct_self_eq_one, SpinGlasses.abs_card_inv_mul_dotProduct_le_one,
        SpinGlasses.card_spin_configurations, Matrix.mulVec_dotProduct_mulVec_of_transpose_mul_self}
  \leanok
Let $\iota$ be a nonempty finite type. (1) If $\sigma_i^2=1$ for all $i$ then $R(\sigma,\sigma)=1$.
(2) If $|\sigma_i|\le1$ and $|\tau_i|\le1$ for all $i$ then $|R(\sigma,\tau)|\le1$.
(3) $|\Sigma_\iota|=2^N$. (4) If $A^\top A=1$ for a matrix $A$ then $(Ax)\cdot(Ay)=x\cdot y$; in particular
$R(O\sigma,O\tau)=R(\sigma,\tau)$ for every orthogonal $O$.
\end{lemma}

\begin{proof}
  \leanok
(1) $\sum_i\sigma_i^2=N$. (2) $|\sum_i\sigma_i\tau_i|\le\sum_i|\sigma_i||\tau_i|\le N$ termwise, so no Cauchy--Schwarz
is needed. (3) $|\{\pm1\}^\iota|=2^{|\iota|}$ (\texttt{Fintype.card\_fun}). (4) $(Ax)\cdot(Ay)=x\cdot(A^\top A)y=x\cdot y$.
\end{proof}

\section{The SK Hamiltonian and its covariance}\label{sec:sg:sk}

\begin{lemma}[Covariance of linear forms of a Gaussian matrix]\label{lem:sg:bilinear}
  \lean{ProbabilityTheory.dotProduct_mulVec_of_eq_sum, ProbabilityTheory.memLp_linearForm_gaussianInit,
        ProbabilityTheory.integral_linearForm_gaussianInit, ProbabilityTheory.integral_gaussianMatrix_bilinear_mul_bilinear}
  \leanok
Let $G\in\mathbb R^{\iota\times\kappa}$ have i.i.d.\ $\mathcal N(0,1)$ entries and let $u,x\in\mathbb R^\iota$,
$v,y\in\mathbb R^\kappa$ be deterministic. Then $u^\top Gv=\sum_{i,j}u_iv_jg_{ij}$ is square-integrable and centered, and
\[\mathbb E\big[(u^\top Gv)(x^\top Gy)\big]=(u\cdot x)(v\cdot y).\]
\end{lemma}

\begin{proof}
  \leanok
Write $G$ as the random vector $q\mapsto g_q$ indexed by $q=(i,j)\in\iota\times\kappa$. Its coordinates are
independent standard Gaussians, so the second moment matrix is the identity
(\texttt{integral\_gaussianInit\_entry\_mul\_entry}), and for any random vector $X$ with square-integrable
coordinates $\mathbb E[(X\cdot a)(X\cdot b)]=a^\top\Gamma b$ with $\Gamma=\mathbb E[XX^\top]$
(\texttt{ProbabilityTheory.integral\_dotProduct\_mul\_dotProduct}). With $a_{(i,j)}=u_iv_j$ and
$b_{(i,j)}=x_iy_j$ we get $\sum_{i,j}u_iv_jx_iy_j=(u\cdot x)(v\cdot y)$.
\end{proof}

\begin{theorem}[Covariance of the SK Hamiltonian, Panchenko Eq.~(1.3)]\label{thm:sg:skcov}
  \lean{SpinGlasses.integral_skHamiltonian_mul_skHamiltonian}
  \leanok
For all real vectors $\sigma,\tau\in\mathbb R^\iota$,
$\mathbb E[H_N(\sigma)H_N(\tau)]=N\,R(\sigma,\tau)^2$.
\end{theorem}

\begin{proof}
  \leanok
  \uses{lem:sg:bilinear}
Apply Lemma~\ref{lem:sg:bilinear} with $u=v=\sigma$ and $x=y=\tau$: the covariance of the unnormalized forms is
$(\sigma\cdot\tau)^2$. The normalization $N^{-1/2}$ appears twice, giving
$N^{-1}(\sigma\cdot\tau)^2=N\,(N^{-1}\sigma\cdot\tau)^2$.
\end{proof}

\begin{lemma}[Dean's comfort identity]\label{lem:sg:comfort}
  \lean{SpinGlasses.sum_lt_mul_units_eq_sub}
  \leanok
For a totally ordered index set and $\sigma\in\{\pm1\}^\iota$, with the nested sums over $i<j$,
\[\sum_{i<j}g_{ij}\sigma_i\sigma_j=\sum_{i<j,\ \sigma_i=\sigma_j}g_{ij}-\sum_{i<j,\ \sigma_i\ne\sigma_j}g_{ij}.\]
\end{lemma}

\begin{proof}
  \leanok
For $\sigma_i,\sigma_j\in\{\pm1\}$ the product $\sigma_i\sigma_j$ equals $+1$ if $\sigma_i=\sigma_j$ and $-1$
otherwise; split each term accordingly.
\end{proof}

\section{Free energy and the smooth approximation of the ground-state energy}\label{sec:sg:free}

The estimate of this section is deterministic; it is proved first for a finite set, then integrated, then specialized
to the hypercube, so that it applies to any finite Gibbs system and not only to SK.

\begin{lemma}[Log-sum-exp sandwich]\label{lem:sg:lse}
  \lean{SpinGlasses.max_le_inv_mul_log_sum_exp, SpinGlasses.inv_mul_log_sum_exp_le,
        SpinGlasses.abs_inv_mul_log_sum_exp_sub_max_le}
  \leanok
For a finite nonempty set $\alpha$, $f:\alpha\to\mathbb R$ and $\beta>0$,
\[\max_a f(a)\le\frac1\beta\log\sum_a e^{\beta f(a)}\le\frac{\log|\alpha|}{\beta}+\max_a f(a).\]
\end{lemma}

\begin{proof}
  \leanok
The sum contains the term $e^{\beta\max f}$, and it is at most $|\alpha|e^{\beta\max f}$. Take logarithms and divide
by $\beta>0$.
\end{proof}

\begin{lemma}[Integrability of the random landscape]\label{lem:sg:integrable}
  \lean{SpinGlasses.integrable_max, SpinGlasses.measurable_log_sum_exp, SpinGlasses.integrable_log_sum_exp}
  \leanok
Let $H:\Omega\to\mathbb R^\alpha$ have measurable, integrable coordinates $\omega\mapsto H(\omega)_a$ on a finite
measure space, $\alpha$ finite nonempty and $\beta>0$. Then $\omega\mapsto\max_aH(\omega)_a$ and
$\omega\mapsto\log\sum_ae^{\beta H(\omega)_a}$ are integrable.
\end{lemma}

\begin{proof}
  \leanok
  \uses{lem:sg:lse}
$|\max_aH_a|\le\sum_a|H_a|$, which is integrable. By Lemma~\ref{lem:sg:lse},
$|\log Z|\le\beta|\max H|+\log|\alpha|$, again integrable on a finite measure space; measurability is that of a
finite composition.
\end{proof}

\begin{theorem}[Smooth approximation, Panchenko Eq.~(1.7)]\label{thm:sg:smooth}
  \lean{SpinGlasses.integral_max_le_inv_mul_integral_log_sum_exp,
        SpinGlasses.inv_mul_integral_log_sum_exp_le,
        SpinGlasses.abs_inv_mul_integral_log_sum_exp_sub_integral_max_le,
        SpinGlasses.log_card_spin_configurations, SpinGlasses.smooth_approximation_sandwich,
        SpinGlasses.smooth_approximation_error_bound}
  \leanok
Let $H$ be a random energy on $\Sigma_\iota$ with measurable integrable coordinates under a probability measure, and
$\beta>0$. Then
\[\frac1N\mathbb E\max_\sigma H(\sigma)\le\frac{1}{N\beta}\mathbb E\log Z(\beta)\le
\frac{\log2}{\beta}+\frac1N\mathbb E\max_\sigma H(\sigma),\]
hence $\big|\frac1{N\beta}\mathbb E\log Z(\beta)-\frac1N\mathbb E\max_\sigma H(\sigma)\big|\le\frac{\log2}\beta$.
(For $N=0$ both sides vanish by the convention $0^{-1}=0$.)
\end{theorem}

\begin{proof}
  \leanok
  \uses{lem:sg:lse, lem:sg:integrable}
Integrate Lemma~\ref{lem:sg:lse} (Bochner integrals are monotone, and the integrands are integrable by
Lemma~\ref{lem:sg:integrable}), then use $\log|\Sigma_\iota|=N\log2$ (Lemma~\ref{lem:sg:overlap}(3)) and divide
by $N$.
\end{proof}

\begin{theorem}[Smooth approximation for the SK model]\label{thm:sg:sksmooth}
  \lean{SpinGlasses.sk_measurable_integrable, SpinGlasses.sk_smooth_approximation_sandwich,
        SpinGlasses.sk_smooth_approximation_error_bound}
  \leanok
For the SK Hamiltonian under i.i.d.\ Gaussian disorder, the hypotheses of Theorem~\ref{thm:sg:smooth} hold, so its
conclusion holds with $H=H_N$.
\end{theorem}

\begin{proof}
  \leanok
  \uses{thm:sg:smooth, lem:sg:bilinear}
For fixed $\sigma$, $G\mapsto H_N(\sigma)$ is a finite linear combination of the coordinates $g_{ij}$, hence
measurable, and it is integrable because it is square-integrable (Lemma~\ref{lem:sg:bilinear}).
\end{proof}

\section{Monotonicity in the temperature and the zero-temperature limit}\label{sec:sg:thermo}

\begin{lemma}[Lyapunov monotonicity of the log-partition function]\label{lem:sg:lyapunov}
  \lean{monotoneOn_inv_mul_log_sum_mul_exp_sub_log_sum,
        SpinGlasses.monotoneOn_inv_mul_log_sum_exp_sub_log_card,
        SpinGlasses.monotoneOn_inv_mul_integral_log_sum_exp_sub_log_card,
        SpinGlasses.monotoneOn_inv_mul_free_energy_sub_log_two}
  \leanok
For a finite nonempty set $\alpha$ and $f:\alpha\to\mathbb R$, the function
$\beta\mapsto\beta^{-1}\big(\log\sum_ae^{\beta f(a)}-\log|\alpha|\big)$ is non-decreasing on $(0,\infty)$. The same
holds for $\beta\mapsto\beta^{-1}\big(\mathbb E\log\sum_ae^{\beta H_a}-\log|\alpha|\big)$ for a random energy as in
Lemma~\ref{lem:sg:integrable}, and for a random energy on the hypercube the function
$\beta\mapsto\beta^{-1}\big(F_N(\beta)-\log2\big)$ is non-decreasing on $(0,\infty)$. More generally, for nonnegative weights
$w_a$ with $\sum_aw_a>0$ the function $\beta\mapsto\beta^{-1}\big(\log\sum_aw_ae^{\beta f(a)}-\log\sum_aw_a\big)$ is
non-decreasing on $(0,\infty)$; the uniform statement is the case $w\equiv1$.
\end{lemma}

\begin{proof}
  \leanok
  \uses{lem:sg:integrable}
Let $\mu_0$ be the uniform probability measure on $\alpha$. Then $\beta^{-1}(\log\sum_ae^{\beta f}-\log|\alpha|)=
\log\|e^f\|_{L^\beta(\mu_0)}$, and the claim is that the $L^p$ norm increases with $p$. Directly: for
$0<\beta_1\le\beta_2$ set $p=\beta_2/\beta_1\ge1$ and apply the power-mean inequality
$\big(\sum_a|\alpha|^{-1}z_a\big)^p\le\sum_a|\alpha|^{-1}z_a^p$ to $z_a=e^{\beta_1f(a)}$. Taking logarithms gives
$\frac{\beta_2}{\beta_1}\log\mathrm{mean}(e^{\beta_1f})\le\log\mathrm{mean}(e^{\beta_2f})$; divide by $\beta_2$. The
random version integrates the pointwise inequality, and the hypercube version divides by $N$ and uses
$\log|\Sigma_\iota|=N\log2$.
\end{proof}

\begin{theorem}[Thermodynamic limit and zero-temperature limit, Panchenko Eq.~(1.10)]\label{thm:sg:zerotemp}
  \lean{exists_tendsto_of_dist_le_of_tendsto, SpinGlasses.monotoneOn_inv_mul_limit_sub_log_two,
        SpinGlasses.exists_tendsto_ground_state_energy, SpinGlasses.sk_monotoneOn_inv_mul_limit_sub_log_two,
        SpinGlasses.sk_exists_tendsto_ground_state_energy}
  \leanok
Consider a family of hypercube models indexed by $k\to\infty$ (for SK, $k=N$ and $\iota=\mathrm{Fin}\,N$), with
measurable, integrable coordinates, and suppose that for every $\beta>0$ the free energy converges,
$F_k(\beta)\to F(\beta)$. Then:
(1) $\beta\mapsto\beta^{-1}(F(\beta)-\log2)$ is non-decreasing on $(0,\infty)$;
(2) the normalized ground-state energies $E_k=|\iota_k|^{-1}\mathbb E\max_\sigma H_k(\sigma)$ converge to some $E_0$, and
$\beta^{-1}F(\beta)\to E_0$ as $\beta\to\infty$.
\end{theorem}

\begin{proof}
  \leanok
  \uses{lem:sg:lyapunov, thm:sg:smooth}
(1) A pointwise limit of monotone functions is monotone (Mathlib's
\texttt{monotoneOn\_of\_frequently\_monotoneOn\_of\_tendsto}) applied to Lemma~\ref{lem:sg:lyapunov}.
(2) By Theorem~\ref{thm:sg:smooth}, $|\beta^{-1}F_k(\beta)-E_k|\le\varepsilon(\beta):=\log2/\beta$ uniformly in $k$. The
limit $g(\beta)=\lim_kF_k(\beta)/\beta$ then satisfies $|g(\beta)-g(\beta')|\le\varepsilon(\beta)+\varepsilon(\beta')$ (let
$k\to\infty$ in the triangle inequality), so $g$ is Cauchy as $\beta\to\infty$ and has a limit $E_0$; the uniform
approximation then makes $E_k$ converge to the same $E_0$ (interchange of limits, in the abstract form of
\texttt{exists\_tendsto\_of\_dist\_le\_of\_tendsto}). In particular, unlike the statement in Panchenko, the
convergence of the ground-state energy is a conclusion and not an assumption.
\end{proof}

\section{The Gibbs measure}\label{sec:sg:gibbs}

For $\beta\in\mathbb R$ and $H:\alpha\to\mathbb R$ on a finite set, the \emph{Gibbs measure}
$G_\beta(a)=e^{\beta H(a)}/Z(\beta)$ is the exponential tilt of the counting measure by $\beta H$; in Lean it is
\texttt{Measure.count.tilted} with tilt $\beta H$, and it is a probability measure whenever $\alpha$ is nonempty. We write
$\langle g\rangle_\beta=\int g\,dG_\beta$.

\begin{lemma}[Gibbs averages]\label{lem:sg:gibbsavg}
  \lean{SpinGlasses.integral_count_eq_sum, SpinGlasses.mgf_count, SpinGlasses.integral_count_tilted,
        SpinGlasses.real_count_tilted_singleton, SpinGlasses.norm_integral_count_tilted_le}
  \leanok
(1) $Z(\beta)=\sum_ae^{\beta H(a)}$ is the moment-generating function of $H$ for the counting measure.
(2) $\langle g\rangle_\beta=\sum_a\frac{e^{\beta H(a)}}{Z(\beta)}\,g(a)$ and $G_\beta(\{a\})=e^{\beta H(a)}/Z(\beta)$.
(3) $|\langle g\rangle_\beta|\le\sum_a|g(a)|$.
\end{lemma}

\begin{proof}
  \leanok
(1) The integral against counting measure on a finite set is the sum. (2) Mathlib's
\texttt{integral\_tilted\_mul\_eq\_mgf}, and the indicator of $\{a\}$ for the mass of the singleton.
(3) $G_\beta$ is a probability measure and $|g|\le\sum_a|g(a)|$ pointwise.
\end{proof}

\begin{theorem}[Temperature derivative of the free energy]\label{thm:sg:deriv}
  \lean{SpinGlasses.hasDerivAt_log_sum_exp, SpinGlasses.hasDerivAt_integral_log_sum_exp,
        SpinGlasses.sk_hasDerivAt_free_energy}
  \leanok
(1) For every $H:\alpha\to\mathbb R$, $\frac{d}{d\beta}\log Z(\beta)=\langle H\rangle_\beta$.
(2) For a random energy with measurable, integrable coordinates and $\beta>0$,
$\frac{d}{d\beta}\mathbb E\log Z(\beta)=\mathbb E\langle H\rangle_\beta$.
(3) For the SK model, $\frac d{d\beta}F_N(\beta)=N^{-1}\mathbb E\langle H_N\rangle_\beta$ for $\beta>0$.
\end{theorem}

\begin{proof}
  \leanok
  \uses{lem:sg:gibbsavg, lem:sg:integrable}
(1) $\log Z=\mathrm{cgf}_H$, the cumulant-generating function of $H$ for the counting measure; every exponential moment
is finite on a finite set, so Mathlib's \texttt{integral\_tilted\_mul\_self} gives
$\mathrm{cgf}_H'(\beta)=\int H\,d(\mu.\mathrm{tilted}(\beta H))$. (2) Differentiate under the integral sign
(dominated differentiation): the pointwise derivative is $\langle H(\omega)\rangle_\beta$ by (1), and
$|\langle H(\omega)\rangle_\beta|\le\sum_a|H(\omega)_a|$ uniformly in $\beta$ by Lemma~\ref{lem:sg:gibbsavg}(3), which
is integrable by assumption. (3) is (2) with the SK landscape (Theorem~\ref{thm:sg:sksmooth}), multiplied by $N^{-1}$.
\end{proof}

\begin{theorem}[Zero-temperature concentration]\label{thm:sg:concentration}
  \lean{SpinGlasses.tendsto_real_count_tilted_singleton_of_lt, SpinGlasses.tendsto_integral_count_tilted_max}
  \leanok
Let $\alpha$ be finite nonempty and $H:\alpha\to\mathbb R$. (1) If $H(a)<\max H$ then $G_\beta(\{a\})\to0$ as
$\beta\to\infty$; indeed $G_\beta(\{a\})\le e^{-\beta(\max H-H(a))}$. (2) $\langle H\rangle_\beta\to\max H$.
\end{theorem}

\begin{proof}
  \leanok
  \uses{lem:sg:gibbsavg}
(1) $Z(\beta)\ge e^{\beta\max H}$ since it contains the term of a maximizer, hence
$e^{\beta H(a)}/Z(\beta)\le e^{-\beta(\max H-H(a))}\to0$. (2) $\max H-\langle H\rangle_\beta=\sum_aG_\beta(\{a\})(\max H-H(a))$
because the weights sum to one; every term with $H(a)<\max H$ tends to $0$ by (1) and the others vanish identically.
\end{proof}

\begin{remark}
The convergence of the Gibbs weights of the ground states to $1/|\mathrm{argmax}\,H|$ is not formalized, since it
is not used anywhere in this chapter.
\end{remark}

\section{Pure and mixed $p$-spin models}\label{sec:sg:pspin}

For $p\ge1$ the pure $p$-spin Hamiltonian is
\[H_{N,p}(\sigma)=N^{-(p-1)/2}\sum_{i_1,\dots,i_p\in\iota}g_{i_1\dots i_p}\sigma_{i_1}\cdots\sigma_{i_p},\]
with i.i.d.\ standard Gaussian disorder $g:(\mathrm{Fin}\,p\to\iota)\to\mathbb R$ (the case $p=2$ is the SK
Hamiltonian). The mixed $p$-spin Hamiltonian is $H_N(\sigma)=\sum_{p\ge1}\beta_pH_{N,p}(\sigma)$ with independent
disorders $g_p$, i.e.\ $g\sim\bigotimes_p\mathcal N(0,1)^{\iota^p}$ (\texttt{Measure.infinitePi}), and the covariance
function is $\xi(x)=\sum_{p\ge1}\beta_p^2x^p$. In Lean the condition $\beta_0=0$ replaces ``the sum starts at $p=1$''.

\begin{lemma}[Gaussian linear forms and series]\label{lem:sg:gaussseries}
  \lean{ProbabilityTheory.integral_dotProduct_mul_dotProduct_pi_gaussianReal,
        ProbabilityTheory.integral_dotProduct_pi_gaussianReal,
        ProbabilityTheory.memLp_dotProduct_pi_gaussianReal,
        ProbabilityTheory.integral_tsum_mul_tsum_of_indepFun,
        ProbabilityTheory.integrable_tsum_of_tsum_eLpNorm_ne_top,
        ProbabilityTheory.eLpNorm_two_le_ofReal_of_integral_sq_le,
        ProbabilityTheory.integral_tsum_dotProduct_mul_tsum_dotProduct,
        ProbabilityTheory.measurable_tsum_dotProduct, ProbabilityTheory.integrable_tsum_dotProduct}
  \leanok
(1) For $g$ with i.i.d.\ $\mathcal N(0,1)$ coordinates indexed by a finite set $\kappa$ and $a,b\in\mathbb R^\kappa$, the
linear forms $a\cdot g$ are square-integrable, centered, and $\mathbb E[(a\cdot g)(b\cdot g)]=a\cdot b$.
(2) Let $X_p,Y_p$ ($p$ in a countable set) be square-integrable with $\sum_p\|X_p\|_{L^2}<\infty$,
$\sum_p\|Y_p\|_{L^2}<\infty$, each $X_p$ centered and independent of $Y_q$ for $p\ne q$. Then
$\sum_pX_p$ is integrable and $\mathbb E[(\sum_pX_p)(\sum_qY_q)]=\sum_p\mathbb E[X_pY_p]$.
(3) Let $g_p$ ($p$ countable) be independent with i.i.d.\ $\mathcal N(0,1)$ coordinates and let $a_p,b_p$ be coefficient
vectors with $\sum_p\|a_p\|_2<\infty$ and $\sum_p\|b_p\|_2<\infty$. Then $\sum_pa_p\cdot g_p$ is measurable and integrable and
$\mathbb E[(\sum_pa_p\cdot g_p)(\sum_pb_p\cdot g_p)]=\sum_pa_p\cdot b_p$.
\end{lemma}

\begin{proof}
  \leanok
(1) The law of $g$ is the standard Gaussian of $\mathbb R^\kappa$, whose covariance bilinear form is the inner product,
and $\mathbb E g=0$. (2) The series converge absolutely almost surely, since $\sum_p\|X_p\|_{L^1}\le\sum_p\|X_p\|_{L^2}$
(Mathlib's \texttt{summable\_norm\_of\_tsum\_eLpNorm\_ne\_top}), so $(\sum_pX_p)(\sum_qY_q)=\sum_{(p,q)}X_pY_q$ almost
everywhere. By H\"older, $\mathbb E|X_pY_q|\le\|X_p\|_{L^2}\|Y_q\|_{L^2}$, and $\sum_{p,q}\|X_p\|_{L^2}\|Y_q\|_{L^2}<\infty$, so the
double series may be integrated termwise (\texttt{integral\_tsum}). For $p\ne q$, independence and centering give
$\mathbb E[X_pY_q]=\mathbb E[X_p]\mathbb E[Y_q]=0$, which leaves the diagonal. (3) Apply (2) to $X_p=a_p\cdot g_p$ and
$Y_p=b_p\cdot g_p$: they are independent for $p\ne q$ because they are functions of different coordinates of the
infinite product, $\|X_p\|_{L^2}=\|a_p\|_2$ by (1), and $\mathbb E[X_pY_p]=a_p\cdot b_p$ by (1).
\end{proof}

\begin{theorem}[Covariance of the pure $p$-spin Hamiltonian, Panchenko Eq.~(1.13)]\label{thm:sg:purecov}
  \lean{SpinGlasses.sum_prod_mul_prod_eq_pow, SpinGlasses.pureCoeff_dotProduct_pureCoeff,
        SpinGlasses.integral_purePSpin_mul_purePSpin}
  \leanok
For $p\ge1$, a nonempty finite $\iota$ and any $\sigma,\tau\in\mathbb R^\iota$,
$\mathbb E[H_{N,p}(\sigma)H_{N,p}(\tau)]=N\,R(\sigma,\tau)^p$.
\end{theorem}

\begin{proof}
  \leanok
  \uses{lem:sg:gaussseries}
$H_{N,p}(\sigma)=u(\sigma)\cdot g$ with $u(\sigma)_{i_1\dots i_p}=N^{-(p-1)/2}\prod_k\sigma_{i_k}$. By
Lemma~\ref{lem:sg:gaussseries}(1) the covariance is $u(\sigma)\cdot u(\tau)=N^{-(p-1)}\sum_{i_1\dots i_p}\prod_k\sigma_{i_k}\tau_{i_k}
=N^{-(p-1)}(\sigma\cdot\tau)^p$, since the sum over multi-indices of a product factorizes
(\texttt{Fintype.sum\_pow}). This equals $N(N^{-1}\sigma\cdot\tau)^p$ for $p\ge1$.
\end{proof}

\begin{lemma}[Summability]\label{lem:sg:summable}
  \lean{SpinGlasses.summable_abs_of_summable_two_pow_mul_sq, SpinGlasses.summable_sq_of_summable_abs,
        SpinGlasses.summable_sq_mul_pow_of_summable_abs}
  \leanok
(1) If $\sum_p2^p\beta_p^2<\infty$ (Panchenko's condition) then $\sum_p|\beta_p|<\infty$.
(2) If $\sum_p|\beta_p|<\infty$ then $\sum_p\beta_p^2<\infty$ and $\xi(x)=\sum_p\beta_p^2x^p$ converges absolutely for
$|x|\le1$.
\end{lemma}

\begin{proof}
  \leanok
(1) $2|\beta_p|\le2^p\beta_p^2+2^{-p}$, since $(2^p|\beta_p|-1)^2\ge0$, and $\sum_p2^{-p}<\infty$. (2) $\beta_p^2\le
(\sum_q|\beta_q|)\,|\beta_p|$ and $|\beta_p^2x^p|\le\beta_p^2$.
\end{proof}

\begin{theorem}[Covariance of the mixed $p$-spin Hamiltonian, Panchenko Eq.~(1.14)]\label{thm:sg:mixedcov}
  \lean{SpinGlasses.mixedPSpin_measurable, SpinGlasses.mixedPSpin_integrable,
        SpinGlasses.integral_mixedPSpin_mul_mixedPSpin}
  \leanok
Let $\iota$ be nonempty and finite, $\beta_0=0$, $\sum_p|\beta_p|<\infty$ and $\sigma,\tau\in[-1,1]^\iota$. Then the
series $H_N(\sigma)=\sum_p\beta_pH_{N,p}(\sigma)$ is measurable and integrable in the disorder, and
\[\mathbb E[H_N(\sigma)H_N(\tau)]=N\,\xi\big(R(\sigma,\tau)\big).\]
\end{theorem}

\begin{proof}
  \leanok
  \uses{lem:sg:gaussseries, thm:sg:purecov, lem:sg:summable}
$H_N(\sigma)=\sum_pa_p(\sigma)\cdot g_p$ with $a_p(\sigma)=\beta_pu_p(\sigma)$. By Theorem~\ref{thm:sg:purecov},
$\|a_p(\sigma)\|_2^2=\beta_p^2NR(\sigma,\sigma)^p\le\beta_p^2N$ because $0\le R(\sigma,\sigma)\le1$ for
$\sigma\in[-1,1]^\iota$; so $\|a_p(\sigma)\|_2\le|\beta_p|\sqrt N$ is summable in $p$. Lemma~\ref{lem:sg:gaussseries}(3) then gives
measurability, integrability and $\mathbb E[H_N(\sigma)H_N(\tau)]=\sum_pa_p(\sigma)\cdot a_p(\tau)
=\sum_p\beta_p^2NR(\sigma,\tau)^p$ (for $p=0$ both sides vanish because $\beta_0=0$).
\end{proof}

\begin{theorem}[Free energy of the mixed $p$-spin model]\label{thm:sg:mixedfree}
  \lean{SpinGlasses.mixedPSpin_smooth_approximation_error_bound,

Under the hypotheses of Theorem~\ref{thm:sg:mixedcov}, for $t>0$,

\begin{proof}
  \leanok
  \uses{thm:sg:mixedcov, thm:sg:smooth, thm:sg:deriv}
For a hypercube configuration the coordinates are in $[-1,1]$, so Theorem~\ref{thm:sg:mixedcov} supplies the
measurability and integrability of $G\mapsto H_N(\sigma)$ required by Theorems~\ref{thm:sg:smooth} and~\ref{thm:sg:deriv}.

\end{proof}

\section{Stein's lemma and Gaussian integration by parts}\label{sec:sg:stein}

\begin{theorem}[Stein's lemma, Panchenko Eq.~(1.19)]\label{thm:sg:stein1d}
  \lean{ProbabilityTheory.integrable_of_integrable_id_mul, ProbabilityTheory.hasDerivAt_gaussianPDFReal_zero,
        ProbabilityTheory.integrable_gaussianReal_iff_smul, ProbabilityTheory.integrable_pow_gaussianReal,
        ProbabilityTheory.integral_id_mul_gaussianReal_eq}
  \leanok
Let $v\ge0$, let $g\sim\mathcal N(0,v)$, and let $F:\mathbb R\to\mathbb R$ be differentiable with derivative $F'$ such
that $gF(g)$ and $F'(g)$ are integrable. Then $\mathbb E[g\,F(g)]=v\,\mathbb E[F'(g)]$.
\end{theorem}

\begin{proof}
  \leanok
For $v=0$ both sides vanish ($g=0$). Let $v>0$ and let $p$ be the density of $g$, so $p'(x)=-(x/v)p(x)$. Since $F$ is
continuous and $|F(x)|\le|xF(x)|$ for $|x|\ge1$, integrability of $xF(x)$ against $p$ gives integrability of $F$ against
$p$. Mathlib's integration by parts on $\mathbb R$ (all three products $Fp'$, $F'p$, $Fp$ integrable) gives
$\int Fp'=-\int F'p$, that is $-v^{-1}\int xF(x)p(x)\,dx=-\int F'p$. As a check, the case $F(x)=x^n$ recovers the moment recursion
$\mathbb E[g^{n+1}]=vn\,\mathbb E[g^{n-1}]$ of the Wick calculus (\texttt{Renormalization.integral\_mul\_pow\_gaussianReal}, which is now
a corollary).
\end{proof}

\begin{lemma}[Resampling one coordinate]\label{lem:sg:resample}
  \lean{MeasureTheory.measurePreserving_update_pi, MeasureTheory.integrable_update_pi,
        MeasureTheory.integral_pi_eq_integral_update}
  \leanok
Let $\pi=\bigotimes_{i}\mu_i$ be a finite product of probability measures. Replacing the $j$-th coordinate of
$x\sim\pi$ by an independent $t\sim\mu_j$ preserves $\pi$: the map $(x,t)\mapsto x[j:=t]$ pushes $\pi\otimes\mu_j$ to $\pi$.
Hence for integrable $f$, $\int f\,d\pi=\int\!\!\int f(x[j:=t])\,d\mu_j(t)\,d\pi(x)$.
\end{lemma}

\begin{proof}
  \leanok
Both measures agree on boxes $\prod_i s_i$: the preimage of the box is $\prod_i s_i'\times s_j$ with $s_i'=s_i$ for $i\ne j$
and $s_j'=\mathbb R$, of measure $\prod_{i\ne j}\mu_i(s_i)\,\mu_j(s_j)$. The integral formula is the change of variables for the
pushforward followed by Fubini.
\end{proof}

\begin{theorem}[Coordinatewise Stein identity]\label{thm:sg:steinpi}
  \lean{ProbabilityTheory.integral_eval_mul_pi_gaussianReal_eq}
  \leanok
  \uses{thm:sg:stein1d, lem:sg:resample}
Let $\xi$ have independent coordinates $\xi_i\sim\mathcal N(0,v_i)$ and let $G$ have partial derivatives $G_i'$ in the sense that
$t\mapsto G(x[i:=t])$ is differentiable with derivative $G_i'(x)$ at $t=x_i$. If $\xi_jG(\xi)$ and $G_j'(\xi)$ are integrable then
$\mathbb E[\xi_jG(\xi)]=v_j\,\mathbb E[G_j'(\xi)]$.
\end{theorem}

\begin{proof}
  \leanok
Resample the $j$-th coordinate (Lemma~\ref{lem:sg:resample}). By Fubini the slices $t\mapsto G(x[j:=t])$ and its derivative are
integrable against $\mathcal N(0,v_j)$ for almost every $x$, and Theorem~\ref{thm:sg:stein1d} applies to each slice.
\end{proof}

\begin{theorem}[Multivariate Gaussian integration by parts, Panchenko Eq.~(1.25)]\label{thm:sg:steinvec}
  \lean{ProbabilityTheory.sum_abs_mulVec_le, ProbabilityTheory.hasDerivAt_mulVec_update,
        ProbabilityTheory.integrable_exp_mul_abs_gaussianReal, ProbabilityTheory.integrable_exp_sum_mul_abs_pi,
        ProbabilityTheory.integrable_comp_mulVec_of_exp_growth,
        ProbabilityTheory.integrable_eval_mul_comp_mulVec_of_exp_growth,
        ProbabilityTheory.integral_mulVec_mul_comp_mulVec_eq_sum,
        ProbabilityTheory.posSemidef_covariance_matrix, ProbabilityTheory.exists_matrix_map_eq_pi_gaussianReal,
        ProbabilityTheory.IsGaussianProcess.hasGaussianLaw_pi, ProbabilityTheory.abs_comp_inr_le_of_exp_growth,
        ProbabilityTheory.integral_mul_comp_eq_sum_integral_mul,
        ProbabilityTheory.integral_mul_comp_eq_sum_integral_mul_of_sum,
        ProbabilityTheory.abs_prod_le_exp_sum, ProbabilityTheory.fderiv_prod_apply_pi_single,
        ProbabilityTheory.integral_mul_prod_eq_sum_integral_prod_erase}
  \leanok
  \uses{thm:sg:steinpi}
Let $X=(X_k)_{k\in\kappa}$, $\kappa$ finite, be a centered jointly Gaussian vector (Mathlib's \texttt{HasGaussianLaw}; a
\texttt{IsGaussianProcess} on a finite index set is one), and let $F:\mathbb R^\kappa\to\mathbb R$ be differentiable with
$|F(y)|\le C\,e^{c\|y\|_1}$ and $|\partial_lF(y)|\le D_l\,e^{d_l\|y\|_1}$. Then
\[\mathbb E[X_i\,F(X)]=\sum_{l\in\kappa}\mathbb E[X_iX_l]\;\mathbb E[\partial_lF(X)].\]
More generally the same holds with $X_i$ replaced by a variable $W_i$ from a centered Gaussian family $W$ that is jointly
Gaussian with $X$, with $\mathbb E[W_iX_l]$ as the covariance. For the monomial $F(y)=\prod_iy_i$ (whose value and
partial derivatives are bounded by $e^{\|y\|_1}$) this is the multilinear identity
$\mathbb E[W\prod_iY_i]=\sum_j\mathbb E[WY_j]\,\mathbb E[\prod_{i\ne j}Y_i]$, which replaces the earlier
Isserlis-based proof in the Renormalization chapter.
\end{theorem}

\begin{proof}
  \leanok
The covariance matrix $S_{ij}=\mathrm{Cov}(X_i,X_j)$ is positive semidefinite (it is the covariance form of the Gaussian
vector), so $X$ has the law of $A\xi$ with $\xi$ i.i.d.\ $\mathcal N(0,1)$ and $AA^\top=S$ ($A=\sqrt S$; this combines Mathlib's
\texttt{multivariateGaussian} with the fact that a Gaussian measure is determined by mean and covariance). For $G(\xi)=F(A\xi)$
the chain rule gives $\partial_jG(\xi)=\sum_lA_{lj}\partial_lF(A\xi)$, and Theorem~\ref{thm:sg:steinpi} yields
$\mathbb E[(A\xi)_iF(A\xi)]=\sum_jA_{ij}\sum_lA_{lj}\mathbb E[\partial_lF(A\xi)]=\sum_lS_{il}\mathbb E[\partial_lF(A\xi)]$.
The integrability hypotheses of Theorem~\ref{thm:sg:steinpi} follow from the growth bound:
$|\xi_jF(A\xi)|\le|\xi_j|\,Ce^{|c|\sum_ka_k|\xi_k|}$ with $a_k=\sum_l|A_{lk}|$, which is integrable as a product of the
one-dimensional integrable functions $e^{b|t|}$ and $|t|e^{b|t|}$ against $\mathcal N(0,1)$. For the extension to an auxiliary
Gaussian family apply the first part to the concatenated vector and the function $F\circ\mathrm{pr}_2$, whose partial derivatives in
the $W$-directions vanish.
\end{proof}

\section{Integration by parts for the Gibbs measure}\label{sec:sg:ibp}

Throughout, $\alpha$ is a finite configuration space, $(x,y)=(x(a),y(a))_{a\in\alpha}$ is a centered jointly Gaussian field with
$C(a,b)=\mathbb E[x(a)y(b)]$, $h:\alpha\to\mathbb R$ is a deterministic bias (for example $\log G$ for a reference weight, or an
external field), and $\langle\cdot\rangle$ is the Gibbs measure \texttt{Measure.count.tilted} of the energy $y+h$. For a finite index
set $R$ of replicas, replicas are the product $\bigotimes_{r\in R}\langle\cdot\rangle$, \texttt{Measure.pi}; Panchenko's $\sigma^1,\dots,\sigma^n$ is $R=\{1,\dots,n\}$,
and $\Phi$ is an arbitrary function of the $|R|$ replicas.

\begin{lemma}[Calculus of replica averages]\label{lem:sg:replicacalc}
  \lean{SpinGlasses.integral_pi_count_tilted, SpinGlasses.differentiable_gibbsWeight,
        SpinGlasses.differentiable_replicaAverage, SpinGlasses.hasDerivAt_prod_gibbsWeight,
        SpinGlasses.fderiv_integral_pi_count_tilted, SpinGlasses.sum_sum_mul_integral_eq,
        SpinGlasses.abs_integral_pi_count_tilted_le, SpinGlasses.abs_integral_pi_count_tilted_mul_eval_le,
        SpinGlasses.differentiable_integral_pi_count_tilted_add, SpinGlasses.fderiv_integral_pi_count_tilted_add}
  \leanok
For $f:\alpha\to\mathbb R$ write $w_f(a)=e^{f(a)}/\sum_be^{f(b)}$. (1) The $R$-replica average of $\Psi$ is the finite sum
$\sum_{\sigma\in\alpha^R}\prod_rw_f(\sigma_r)\Psi(\sigma)$, a smooth function of $f$. (2) Its derivative in the direction $e_\rho$ is
$\big\langle(\sum_r\mathbf 1[\sigma_r=\rho]-|R|\,w_f(\rho))\Psi\big\rangle_f$. (3) $|\langle\Psi\rangle|\le\sup|\Psi|$ and
$|\langle\Phi\,\chi(\sigma_{r_0})\rangle|\le\sup|\Phi|\,\langle|\chi|\rangle$.
\end{lemma}

\begin{proof}
  \leanok
(1) is the product formula for the product measure of singletons (\texttt{Measure.pi\_singleton}) together with the formula for the
mass of a singleton of the tilted counting measure. For (2) one differentiates along the line $f+te_\rho$: the product of $n$
weights is $\exp(\sum_r(f(\sigma_r)+t\delta_{\sigma_r\rho}))/Z(t)^n$ with $Z'(0)=e^{f(\rho)}$, and the quotient rule gives
$(\prod w)\,(\sum_r\delta_{\sigma_r\rho}-n\,w(\rho))$. (3) is the triangle inequality and the identity
$\int|\chi(\sigma_{r_0})|\,d\pi=\int|\chi|\,d\langle\cdot\rangle$.
\end{proof}

\begin{theorem}[Replica covariance formula, Panchenko Exercise~1.1 and Eq.~(1.29)]\label{thm:sg:replica}
  \lean{SpinGlasses.integral_gibbs_replica_average_gaussian, SpinGlasses.integral_pi_option_count_tilted,
        SpinGlasses.integral_gibbs_replica_average_gaussian_option}
  \leanok
  \uses{thm:sg:steinvec, lem:sg:replicacalc}
With the notation above, for every $r_0\in R$ and every function $\Phi$ of the $|R|$ replicas,
\[\mathbb E\big\langle\Phi\,x(\sigma^{r_0})\big\rangle
 =\mathbb E\Big\langle\Phi\Big(\sum_{r\in R}C(\sigma^{r_0},\sigma^r)-|R|\,C(\sigma^{r_0},\sigma^{n+1})\Big)\Big\rangle,\]
where $\sigma^{n+1}$ is an additional independent replica (written as an integral of $C(\sigma^{r_0},\cdot)$ against the Gibbs measure,
and, in the second form, as the replica indexed by \texttt{none} in the product over \texttt{Option R}).
\end{theorem}

\begin{proof}
  \leanok
Write $x(\sigma^{r_0})=\sum_ax(a)\mathbf 1[\sigma^{r_0}=a]$, so the left-hand side is $\sum_a\mathbb E[x(a)F_a(y)]$ with
$F_a(y)=\langle\mathbf 1[\sigma^{r_0}=a]\Phi\rangle_{y+h}$, a smooth bounded function with bounded partial derivatives
(Lemma~\ref{lem:sg:replicacalc}). By Theorem~\ref{thm:sg:steinvec}, $\mathbb E[x(a)F_a(y)]=\sum_bC(a,b)\,\mathbb E[\partial_bF_a(y)]$, and
$\partial_bF_a=\langle(\sum_r\mathbf 1[\sigma^r=b]-|R|\,w(b))\mathbf 1[\sigma^{r_0}=a]\Phi\rangle$. Summing over $a$ and $b$ gives
$\sum_r C(\sigma^{r_0},\sigma^r)-|R|\sum_bC(\sigma^{r_0},b)w(b)$ inside the Gibbs average, and $\sum_bC(\sigma^{r_0},b)w(b)$ is
the average of $C(\sigma^{r_0},\cdot)$ against the Gibbs measure.
\end{proof}

\begin{theorem}[Gaussian integration by parts for the Gibbs measure, Panchenko Lemma~1.1, Eq.~(1.28)]\label{thm:sg:lemma11}
  \lean{SpinGlasses.integral_gibbs_average_gaussian}
  \leanok
  \uses{thm:sg:replica}
$\mathbb E\langle x(\sigma)\rangle=\mathbb E\langle C(\sigma^1,\sigma^1)-C(\sigma^1,\sigma^2)\rangle$, where $\langle\cdot\rangle$ on the right
is the product of two replicas.
\end{theorem}

\begin{proof}
  \leanok
This is Theorem~\ref{thm:sg:replica} with one replica ($R=\mathrm{Unit}$) and $\Phi\equiv1$.
\end{proof}

\section{A countable configuration space}\label{sec:sg:countable}

\begin{lemma}[Analytic tools for series over replicas]\label{lem:sg:tannery}
  \lean{summable_prod_pi, tendsto_tsum_prod_pi, summable_prod_pi_mul_abs_eval,
        SpinGlasses.truncated_gibbs_weights, SpinGlasses.tendsto_truncated_replica_average,
        ProbabilityTheory.integrable_tsum_of_tsum_eLpNorm_ne_top}
  \leanok
Let $J$ be finite and $f_j:\alpha\to[0,\infty)$ summable. Then $\sigma\mapsto\prod_jf_j(\sigma_j)$ is summable on $J\to\alpha$. If
weights $w_m\to w$ pointwise with $0\le w_m\le B$, $B$ summable and $\sum_\sigma\prod_jB(\sigma_j)|\Psi(\sigma)|<\infty$, then
$\sum_\sigma\prod_jw_m(\sigma_j)\Psi(\sigma)\to\sum_\sigma\prod_jw(\sigma_j)\Psi(\sigma)$. In particular, for $g>0$ summable and finite
sets $A_m\uparrow\alpha$ containing $a_0$, the truncated weights $\mathbf 1_{A_m}g/\sum_{A_m}g$ are bounded by $g/g(a_0)$ and converge to $g/\sum g$. A series with
summable $L^p$ norms, $p\ge1$, is integrable (the case $p=2$ is used for the Gaussian series of Section~\ref{sec:sg:pspin}).
\end{lemma}

\begin{proof}
  \leanok
Partial sums of $\prod_jf_j(\sigma_j)$ over a finite set $T$ are bounded by $\prod_j\sum_{a\in T_j}f_j(a)\le\prod_j\sum_af_j(a)$, where $T_j$ is
the set of $j$-th coordinates of elements of $T$ (\texttt{Finset.prod\_univ\_sum}). The limit statement is Tannery's theorem with dominating function
$\prod_jB(\sigma_j)|\Psi(\sigma)|$. For the truncated weights, $\sum_{A_m}g\ge g(a_0)>0$ and $\sum_{A_m}g\to\sum g$ along the exhaustion.
\end{proof}

\begin{lemma}[Gaussian exponential moments]\label{lem:sg:gaussexp}
  \lean{ProbabilityTheory.hasGaussianLaw_hasLaw, ProbabilityTheory.integral_exp_mul_of_hasGaussianLaw,
        ProbabilityTheory.integrable_exp_mul_of_hasGaussianLaw, ProbabilityTheory.eLpNorm_one_mul_exp_le,
        ProbabilityTheory.ae_summable_and_integrable_tsum_mul_exp}
  \leanok
If $Z$ is Gaussian then $\mathbb E\,e^{tZ}=e^{t\mathbb EZ+t^2\mathrm{Var}Z/2}$. If $Z$ is centered Gaussian with $\mathbb EZ^2\le v$ and
$\mathbb EX^2\le a$ then $\mathbb E|X|e^{Z}\le\sqrt a\,e^v$. Consequently, for countably many such pairs $(X_b,Z_b)$ with uniform bounds and
$G_b\ge0$ summable, $\sum_bG_b|X_b|e^{Z_b}$ converges almost surely and is integrable.
\end{lemma}

\begin{proof}
  \leanok
  \uses{lem:sg:tannery}
The law of $Z$ is Gaussian, so the first claim is the Gaussian moment generating function. By Cauchy--Schwarz (Holder with exponents
$2,2,1$) $\|Xe^Z\|_1\le\|X\|_2\|e^Z\|_2$ and $\|e^Z\|_2^2=\mathbb Ee^{2Z}=e^{2\mathrm{Var}Z}\le e^{2v}$. The series has summable $L^1$ norms
$\le\sqrt a\,e^v\sum_bG_b$, hence converges absolutely a.s.\ and is integrable (Lemma~\ref{lem:sg:tannery}).
\end{proof}

\begin{theorem}[Finite truncations of the replica formula]\label{thm:sg:truncation}
  \lean{SpinGlasses.integral_pi_count_tilted_subtype, SpinGlasses.abs_tsum_truncated_mul_eval_le,
        SpinGlasses.abs_tsum_truncated_le, SpinGlasses.aemeasurable_tsum_prod_pi_of_forall_notMem,
        SpinGlasses.integral_truncated_gibbs_replica_average_gaussian}
  \leanok
  \uses{thm:sg:replica}
Let $\alpha$ be countable, $G>0$, $(x,y)$ a centered Gaussian process, and $A\subseteq\alpha$ a finite set. Then Theorem~\ref{thm:sg:replica}
holds for the truncated Gibbs weights $\mathbf 1_A(a)G(a)e^{y(a)}/\sum_{b\in A}G(b)e^{y(b)}$, with the replica averages written as series over
$R\to\alpha$ and $\mathrm{Option}\,R\to\alpha$. Moreover each truncated replica average is bounded by $M\sum_b(g_b/g_{a_0})|x_b|$, respectively
by the sup of the observable.
\end{theorem}

\begin{proof}
  \leanok
Apply Theorem~\ref{thm:sg:replica} on the finite type $A$ with the bias $h=\log G$. The replica average over $A^R$ of $\Psi\circ\mathrm{val}$ equals the
series over $R\to\alpha$ with the truncated weights (the weight vanishes off $A$, so the series is a finite sum over $\prod_{r}A$). The bounds are
Lemma~\ref{lem:sg:replicacalc}(3) applied on $A$.
\end{proof}

\begin{theorem}[Replica covariance formula for a countable configuration space, Panchenko Lemma~1.2]\label{thm:sg:lemma12}
  \lean{SpinGlasses.integral_gibbs_replica_average_gaussian_countable}
  \leanok
  \uses{thm:sg:truncation, lem:sg:gaussexp, lem:sg:tannery}
Let $\alpha$ be countable, $G:\alpha\to(0,\infty)$ summable, and $(x,y)$ a centered jointly Gaussian process (\texttt{IsGaussianProcess}) with
$\mathbb Ex(a)^2\le v$ and $\mathbb Ey(a)^2\le v$ for all $a$. Let $\Phi$ be a function of $|R|$ replicas with $|\Phi|\le M$, and
$w(a)=G(a)e^{y(a)}/\sum_bG(b)e^{y(b)}$ the Gibbs weights. Then
\[\mathbb E\sum_{\sigma\in\alpha^R}\prod_rw(\sigma_r)\,\Phi(\sigma)\,x(\sigma_{r_0})
 =\mathbb E\sum_{\tau\in\alpha^{R\sqcup\{*\}}}\prod_{o}w(\tau_o)\,\Phi(\tau|_R)\Big(\sum_{r}C(\tau_{r_0},\tau_r)-|R|\,C(\tau_{r_0},\tau_*)\Big).\]
\end{theorem}

\begin{proof}
  \leanok
Choose finite sets $A_m\uparrow\alpha$ containing a base point $a_1$ and apply Theorem~\ref{thm:sg:truncation} to each $A_m$; it remains to pass to the limit
$m\to\infty$ in both sides by dominated convergence. Put $g_b=G(b)e^{y(b)}$ and $Z_b=y(b)-y(a_1)$, so that $g_b/g_{a_1}=(G(b)/G(a_1))e^{Z_b}$. The $Z_b$ are centered Gaussians with
$\mathbb EZ_b^2\le4v$. By Lemma~\ref{lem:sg:gaussexp} (with $X_b=1$ and $X_b=x(b)$), almost surely $\sum_bg_b<\infty$ and $\sum_b(g_b/g_{a_1})|x(b)|<\infty$, and
$\sum_bG(b)|x(b)|e^{Z_b}$ is integrable. For such $\omega$, Lemma~\ref{lem:sg:tannery} gives the pointwise convergence of the truncated averages on
both sides (the summability hypothesis on $\Psi$ follows from the summability of $\prod B$ and of $B|x|$). The left-hand averages are dominated by
$M\sum_b(g_b/g_{a_1})|x(b)|$, which is integrable, and the right-hand averages by $M\cdot2|R|v$, because $|C(a,b)|\le(\mathbb Ex(a)^2+\mathbb Ey(b)^2)/2\le v$.
Dominated convergence shows that both sides of the truncated identities converge, hence the limits agree.
\end{proof}

\begin{remark}
Not formalized: the case of a general (non-atomic or sigma-finite) reference measure $G$ in Lemma~1.2, and the bound $\sum G\le1$ of the book
(only summability is used). Eq.~(1.29) is stated for an arbitrary $\Phi$ (the book assumes boundedness only for the countable case, as here).
\end{remark}

\section{Gaussian series and the Gaussian law of the mixed Hamiltonian}\label{sec:sg:serieslaw}

\begin{lemma}[The law of a Gaussian series]\label{lem:sg:serieslaw}
  \lean{ProbabilityTheory.isGaussian_pi_gaussianReal, ProbabilityTheory.hasGaussianLaw_coord,
        ProbabilityTheory.iIndepFun_coord, ProbabilityTheory.map_sum_range_dotProduct,
        ProbabilityTheory.summable_dotProduct_self, ProbabilityTheory.map_tsum_dotProduct_eq_gaussianReal,
        ProbabilityTheory.integral_tsum_dotProduct_eq_zero, ProbabilityTheory.sqrt_dotProduct_self_sum_le,
        ProbabilityTheory.hasGaussianLaw_tsum_dotProduct, ProbabilityTheory.ae_summable_of_tsum_eLpNorm_ne_top}
  \leanok
  \uses{lem:sg:gaussseries}
Let $g_p$, $p\in\mathbb N$, be independent vectors with i.i.d.\ $\mathcal N(0,1)$ coordinates and $a_p$ coefficient vectors with
$\sum_p\|a_p\|<\infty$. Then $H_a=\sum_pa_p\cdot g_p$ converges almost surely, is exactly Gaussian,
$H_a\sim\mathcal N(0,\sum_p\|a_p\|^2)$, and finitely many such series $(H_{a^s})_{s\in S}$ form a Gaussian vector (Mathlib's
\texttt{HasGaussianLaw}).
\end{lemma}

\begin{proof}
  \leanok
The summands $X_p=a_p\cdot g_p$ are independent centered Gaussians of variance $\|a_p\|^2$, so the partial sum $S_n=\sum_{p<n}X_p$ is Gaussian
with variance $v_n=\sum_{p<n}\|a_p\|^2$ (independent Gaussians are jointly Gaussian; the variance is the covariance formula of
Lemma~\ref{lem:sg:gaussseries} for the coefficients truncated at $n$). Since $\sum_p\|X_p\|_{L^1}\le\sum_p\|a_p\|<\infty$,
$\sum_p|X_p|<\infty$ almost surely, hence $S_n\to H_a$ almost surely. The characteristic function of $H_a$ is the limit of that of $S_n$ by
dominated convergence ($|e^{itS_n}|=1$), i.e.\ $\lim_ne^{-v_nt^2/2}=e^{-vt^2/2}$ with $v=\sum_p\|a_p\|^2$, and the characteristic function determines the law.
For the vector, a continuous linear functional $L(x)=\sum_sc_sx_s$ of $(H_{a^s})_s$ equals almost surely the single series with coefficients
$\sum_sc_sa^s_p$ (summable by Minkowski's inequality), so $L\circ(H_{a^s})_s$ is real Gaussian, which is Mathlib's criterion for a Gaussian measure.
\end{proof}

\begin{theorem}[The mixed $p$-spin Hamiltonian is a Gaussian process]\label{thm:sg:mixedgauss}
  \lean{SpinGlasses.mixedPSpin_hasGaussianLaw, SpinGlasses.integral_mixedPSpin_eq_zero,
        SpinGlasses.integral_mixedPSpin_hypercube_mul_mixedPSpin_hypercube, SpinGlasses.mixedPSpin_hypercube_hyps,

  \leanok

\end{theorem}

\begin{proof}
  \leanok
By definition $H_N(\sigma)=\sum_pa_p(\sigma)\cdot g_p$ with $a_p(\sigma)=\beta_pN^{-(p-1)/2}(\prod_k\sigma_{i_k})_{i}$ and
$\sum_p\|a_p(\sigma)\|\le\sqrt N\sum_p|\beta_p|$ (Theorem~\ref{thm:sg:mixedcov}). Lemma~\ref{lem:sg:serieslaw} gives the Gaussian law, the mean zero,

\end{proof}

\section{Gaussian comparison for the free energy}\label{sec:sg:comparison}

\begin{lemma}[Derivative of the expected log-partition function along a path]\label{lem:sg:pathderiv}
  \lean{SpinGlasses.hasDerivAt_log_sum_exp_path, SpinGlasses.hasDerivAt_integral_log_sum_exp_path}
  \leanok
  \uses{lem:sg:gibbsavg, thm:sg:deriv}
Let $f(s,\omega,a)$ be differentiable in $s$ on a neighbourhood of $t$ with derivative $f'(s,\omega,a)$ bounded by an integrable $B(\omega)$. Then
$\frac{d}{ds}\mathbb E\log\sum_ae^{f(s,\omega,a)}\big|_{s=t}=\mathbb E\langle f'(t)\rangle_{f(t)}$, where $\langle\cdot\rangle_{f(t)}$ is the
Gibbs average of the energy $f(t,\omega,\cdot)$. The temperature derivative of Theorem~\ref{thm:sg:deriv} is the case $f(s,\omega,a)=sH(\omega,a)$.
\end{lemma}

\begin{proof}
  \leanok
Pointwise, $\frac{d}{ds}\log\sum_ae^{f}=\sum_ae^{f}f'/\sum_ae^{f}=\langle f'\rangle$; differentiate under the integral by dominated convergence
(\texttt{hasDerivAt\_integral\_of\_dominated\_loc\_of\_deriv\_le}), the Gibbs average being bounded by $\sup_a|f'|\le B$.
\end{proof}

\begin{theorem}[Gaussian comparison, the smooth Slepian inequality]\label{thm:sg:comparison}
  \lean{SpinGlasses.hasGaussianLaw_sumElim_lincomb, SpinGlasses.integral_lincomb_mul_lincomb,
        SpinGlasses.hasDerivAt_integral_log_sum_exp_trig, SpinGlasses.integral_log_sum_exp_le_of_covariance_le}
  \leanok
  \uses{lem:sg:pathderiv, thm:sg:lemma11}
Let $(X,Y)$ be a centered jointly Gaussian pair of fields on a finite set $\alpha$ with $\mathbb EX(a)Y(b)=0$, equal variances
$\mathbb EX(a)^2=\mathbb EY(a)^2$, and $\mathbb EX(a)X(b)\le\mathbb EY(a)Y(b)$ for all $a,b$. Then for every bias $h:\alpha\to\mathbb R$,
\[\mathbb E\log\sum_ae^{Y(a)+h(a)}\le\mathbb E\log\sum_ae^{X(a)+h(a)}.\]
\end{theorem}

\begin{proof}
  \leanok
Put $u_s=\cos s\,X+\sin s\,Y+h$ and $\varphi(s)=\mathbb E\log\sum_ae^{u_s(a)}$. By Lemma~\ref{lem:sg:pathderiv}, $\varphi'(s)=\mathbb E\langle x_s\rangle$
with $x_s=-\sin s\,X+\cos s\,Y=u_s'$. The pair $(x_s,u_s)$ is jointly Gaussian and
$C(a,b)=\mathbb Ex_s(a)u_s(b)=\sin s\cos s\,(B-A)(a,b)$ with $A=\mathrm{Cov}X$, $B=\mathrm{Cov}Y$, because $\mathbb EXY=0$. Gaussian integration by
parts for the Gibbs measure (Lemma~1.1, Theorem~\ref{thm:sg:lemma11}) gives
$\varphi'(s)=\mathbb E\langle C(\sigma^1,\sigma^1)-C(\sigma^1,\sigma^2)\rangle=-\sin s\cos s\,\mathbb E\langle(B-A)(\sigma^1,\sigma^2)\rangle\le0$
on $[0,\pi/2]$, since $(B-A)(a,a)=0$ and $B\ge A$. Hence $\varphi(\pi/2)\le\varphi(0)$, which is the claim. (The trigonometric path is smooth on all of
$\mathbb R$, so no endpoint continuity or singular derivative at $t=0,1$ occurs, unlike $\sqrt tX+\sqrt{1-t}Y$.)
\end{proof}

\section{The Guerra--Toninelli theorem: superadditivity and the thermodynamic limit}\label{sec:sg:gt}

\begin{lemma}[Fekete's lemma for superadditive sequences, Panchenko Lemma~1.3]\label{lem:sg:fekete}
  \lean{tendsto_div_atTop_sSup_of_superadditive}
  \leanok
If $u_{m+n}\ge u_m+u_n$ for all $m,n$ and $u_n/n$ is bounded above, then $u_n/n\to\sup_{n\ge1}u_n/n$.
\end{lemma}

\begin{proof}
  \leanok
The sequence $-u$ is subadditive; apply Mathlib's Fekete lemma (\texttt{Subadditive.tendsto\_lim}) and negate the limit $\inf_n(-u_n)/n$.
No definition of superadditivity is introduced.
\end{proof}

\begin{theorem}[Superadditivity of the free energy, Guerra--Toninelli, Panchenko Eq.~(1.36)]\label{thm:sg:superadd}
  \lean{SpinGlasses.add_mul_apply_le_of_convexOn, SpinGlasses.dotProduct_cast_append,
        SpinGlasses.dotProduct_units_cast_self, SpinGlasses.integrable_log_sum_exp_of_hasGaussianLaw,
        SpinGlasses.integral_log_sum_exp_add_le, ProbabilityTheory.HasGaussianLaw.prodMk_prod}
  \leanok
  \uses{thm:sg:comparison}
Let $H_N$ be centered Gaussian Hamiltonians on $\{\pm1\}^N$ (disorders on arbitrary probability spaces $\Omega_N$) with
$\mathbb EH_N(\sigma)H_N(\tau)=N\xi(R(\sigma,\tau))$ for a function $\xi$ that is convex on $[-1,1]$. Then for $N,M>0$,
\[\mathbb E\log Z_N+\mathbb E\log Z_M\le\mathbb E\log Z_{N+M},\qquad\text{i.e.}\quad NF_N+MF_M\le(N+M)F_{N+M}.\]
\end{theorem}

\begin{proof}
  \leanok
On $\Omega_{N+M}\times(\Omega_N\times\Omega_M)$ put $X(\rho,\tau)=H_{N+M}(\rho\frown\tau)$ (concatenation of the configurations) and
$Y(\rho,\tau)=H_N(\rho)+H_M(\tau)$. The triple of Hamiltonians is jointly Gaussian (a product of Gaussian measures), and $X$ is independent of $Y$, so
$\mathbb EXY=0$. The overlap splits, $R(\rho\frown\tau,\rho'\frown\tau')=\frac{N}{N+M}R(\rho,\rho')+\frac{M}{N+M}R(\tau,\tau')$, hence
$\mathbb EX(\rho,\tau)X(\rho',\tau')=(N+M)\xi\big(\frac{NR+MR'}{N+M}\big)\le N\xi(R)+M\xi(R')=\mathbb EY(\rho,\tau)Y(\rho',\tau')$ by convexity of $\xi$,
and both variances equal $(N+M)\xi(1)$. Theorem~\ref{thm:sg:comparison} gives $\mathbb E\log\sum e^Y\le\mathbb E\log\sum e^X$; the right side is
$\mathbb E\log Z_{N+M}$ and the left side is $\mathbb E\log(Z_NZ_M)=\mathbb E\log Z_N+\mathbb E\log Z_M$.
\end{proof}

\begin{lemma}[Convexity of $\xi$]\label{lem:sg:convexxi}
  \lean{SpinGlasses.convexOn_sum_sq_mul_pow}
  \leanok
If $\sum_p|\beta_p|<\infty$ and $\beta_p=0$ for all odd $p\ge3$, then $\xi(x)=\sum_p\beta_p^2x^p$ is convex on $[-1,1]$.
\end{lemma}

\begin{proof}
  \leanok
Each term is convex: $x^p$ for even $p$, the linear function $x$ for $p=1$, and the term vanishes for odd $p\ge3$. The inequality
$\xi(ax+by)\le a\xi(x)+b\xi(y)$ passes to the sum of the convergent series.
\end{proof}

\begin{theorem}[Existence of the thermodynamic limit, Guerra--Toninelli, Panchenko Theorem~1.1]\label{thm:sg:thermolimit}
  \lean{SpinGlasses.integral_log_le_log_integral, SpinGlasses.integral_log_sum_exp_le,
        SpinGlasses.tendsto_free_energy, SpinGlasses.tendsto_free_energy_mul,
        SpinGlasses.sk_tendsto_free_energy, SpinGlasses.sk_tendsto_ground_state_energy,
        SpinGlasses.sk_monotoneOn_free_energy_limit, SpinGlasses.mixedPSpin_tendsto_free_energy,
        SpinGlasses.mixedPSpin_tendsto_ground_state_energy}
  \leanok
  \uses{thm:sg:superadd, lem:sg:fekete, lem:sg:convexxi, thm:sg:mixedgauss, thm:sg:zerotemp, thm:sg:skcov}
Under the hypotheses of Theorem~\ref{thm:sg:superadd}, for every inverse temperature $t$ the free energy
$F_N(t)=N^{-1}\mathbb E\log Z_N(t)$ converges to $\sup_{N\ge1}F_N(t)$. This applies to the SK model
(covariance function $\xi(x)=x^2$) and to the mixed $p$-spin models with $\beta_0=0$, $\sum_p|\beta_p|<\infty$ and $\beta_p=0$ for odd
$p\ge3$ ($\xi(x)=\sum_p\beta_p^2x^p$). Consequently the hypothesis \texttt{h\_lim} of Theorem~\ref{thm:sg:zerotemp} holds: for these models
$N^{-1}\mathbb E\max_\sigma H_N(\sigma)$ converges, and for SK the function $\beta\mapsto\beta^{-1}(F(\beta)-\log2)$ is non-decreasing.
\end{theorem}

\begin{proof}
  \leanok
The Hamiltonian $tH_N$ has covariance $N(t^2\xi)(R)$ and $t^2\xi$ is convex, so we may take $t=1$. Put $u_N=\mathbb E\log Z_N$ ($u_0=0$); it is
superadditive by Theorem~\ref{thm:sg:superadd}. It is bounded above: by Jensen's inequality for the logarithm, $\mathbb E\log Z_N\le\log\mathbb EZ_N
=\log\sum_\sigma e^{\mathrm{Var}H_N(\sigma)/2}=N(\log2+\xi(1)/2)$, since $\mathrm{Var}\,H_N(\sigma)=N\xi(1)$. Fekete's lemma
(Lemma~\ref{lem:sg:fekete}) shows $u_N/N\to\sup_{N\ge1}u_N/N$. For SK, $\xi(x)=x^2$ is convex and the covariance is Theorem~\ref{thm:sg:skcov}; the
Gaussian law is that of the linear image $G\mapsto(\sigma^\top G\sigma)_\sigma$ of the Gaussian matrix. For the mixed model use
Lemma~\ref{lem:sg:convexxi} and Theorem~\ref{thm:sg:mixedgauss}. The consequences follow from Theorem~\ref{thm:sg:zerotemp}.
\end{proof}

\begin{remark}
The identification of the SK Hamiltonian with the pure $p=2$ model (which would need the equivalence of the nested disorder
\texttt{Fin N -> Fin N -> R} with the flat disorder \texttt{(Fin 2 -> Fin N) -> R}) is not needed: the interpolation sees the Hamiltonian only through
its Gaussian law and covariance function $\xi$, and the SK covariance $N R^2$ is that of $\xi(x)=x^2$ directly. The hypothesis $\beta_p=0$ for odd $p\ge3$
is exactly the convexity of $\xi$ on $[-1,1]$; Theorem~\ref{thm:sg:superadd} holds for any convex $\xi$.
\end{remark}

\section{Concentration of the free energy}\label{sec:sg:fe-conc}

This section proves Panchenko's Theorem~1.2 (the Gaussian concentration of the free energy) and its consequence Eq.~(1.42)
for the mixed $p$-spin models. The argument is a variant of the book's proof that uses only two independent copies of the field and
the same Gaussian integration by parts as the comparison inequality. It gives the better variance bound $2v$ in place of $8v$. The deviations from the
book's argument, and the reasons for them, are collected in Remark~\ref{rem:sg:deviation} at the end of the section.

\begin{lemma}[Gaussian vectors: integrability, differentiation and exponentials]\label{lem:sg:expgrowth}
  \lean{ProbabilityTheory.integrable_comp_of_exp_growth, ProbabilityTheory.hasDerivAt_integral_comp_of_exp_growth,
        ProbabilityTheory.integrable_mul_exp_comp_of_sum, ProbabilityTheory.integrable_exp_comp_of_sum,
        ProbabilityTheory.integral_mul_exp_comp_eq_sum_integral_mul_of_sum, ProbabilityTheory.HasGaussianLaw.sumElim_prod}
  \leanok
  \uses{thm:sg:steinvec}
Let $X$ be a centered Gaussian vector indexed by a finite set. (i) If $F$ is measurable with $|F(y)|\le Ce^{c\|y\|_1}$ then $F(X)$ is
integrable. (ii) If $G(s,\cdot)$ is differentiable in $s$ with $|\partial_sG(s,y)|\le Ce^{c\|y\|_1}$ uniformly in $s$, then
$\frac{d}{ds}\mathbb EG(s,X)=\mathbb E\,\partial_sG(s,X)$. (iii) If $(W,X)$ is a centered Gaussian family and $A$ is differentiable with
$|A(y)|\le c_1+c_2\|y\|_1$ and $|\partial_lA|\le c_3$, then $W_ie^{A(X)}$ and $e^{A(X)}$ are integrable and
$\mathbb E[W_ie^{A(X)}]=\sum_l\mathbb E[W_iX_l]\,\mathbb E[e^{A(X)}\partial_lA(X)]$. (iv) If $g$ is a Gaussian vector under $P$, then
$z\mapsto(g(z_1),g(z_2))$ is a Gaussian vector under $P\otimes P$.
\end{lemma}

\begin{proof}
  \leanok
Write $X\overset d=A\xi$ with $\xi$ i.i.d.\ standard Gaussians (Lemma~\ref{thm:sg:steinvec}'s representation); $e^{c\|A\xi\|_1}$ is integrable, which
gives (i), and dominated differentiation under the integral gives (ii). Part (iii) is Stein's identity for $F=\exp\circ A$, whose value and partial
derivatives have exponential growth. Part (iv) is the fact that a product of Gaussian measures is Gaussian.
\end{proof}

\begin{theorem}[Exponential moments of the free-energy difference of two copies]\label{thm:sg:expmoment}
  \lean{SpinGlasses.abs_log_sum_exp_sub_le_of_forall, SpinGlasses.abs_log_sum_exp_sub_le,
        SpinGlasses.hasDerivAt_cos_mul_add_sin_mul, SpinGlasses.abs_neg_sin_mul_add_cos_mul_le,
        SpinGlasses.hasFDerivAt_log_sum_exp,
        SpinGlasses.hasDerivAt_exp_mul_log_sum_exp_trig, SpinGlasses.exp_mul_log_sum_exp_trig_le,
        SpinGlasses.abs_exp_mul_gibbs_average_le, SpinGlasses.hasDerivAt_integral_exp_mul_log_sum_exp_trig,
        SpinGlasses.gibbsWeight_mem_Icc, SpinGlasses.abs_log_sum_exp_le, SpinGlasses.hasFDerivAt_gibbs_exponent,
        SpinGlasses.differentiable_gibbs_exponent, SpinGlasses.fderiv_gibbs_exponent_apply_single,
        SpinGlasses.abs_gibbs_exponent_le, SpinGlasses.abs_fderiv_gibbs_exponent_le, SpinGlasses.exp_gibbs_exponent,
        SpinGlasses.integral_exp_mul_gibbs_average_trig, SpinGlasses.sum_gibbsWeight,
        SpinGlasses.sum_sum_gibbsWeight_mul_le, SpinGlasses.abs_sum_sum_gibbsWeight_mul_le,
        SpinGlasses.integral_exp_mul_log_sum_exp_sub_le}
  \leanok
  \uses{lem:sg:expgrowth, thm:sg:comparison, lem:sg:pathderiv}
Let $(g_1,g_2)$ be a centered jointly Gaussian pair of fields on a finite set $\alpha$ with $\mathbb Eg_1g_2=0$ and the same covariance
$C=\mathbb Eg_1g_1=\mathbb Eg_2g_2$, with $C(x,y)\le a$ for all $x,y$. Then for every bias $h$ and every $t\in\mathbb R$,
\[\mathbb E\exp\Big(t\Big(\log\sum_xe^{g_1(x)+h(x)}-\log\sum_xe^{g_2(x)+h(x)}\Big)\Big)\le e^{at^2}.\]
\end{theorem}

\begin{proof}
  \leanok
Put $x_s=\cos s\,g_1+\sin s\,g_2$, $x'_s=-\sin s\,g_1+\cos s\,g_2$, $\Phi_s=\exp\big(t(\log Z(x_s)-\log Z(g_2))\big)$ and $\varphi(s)=\mathbb E\Phi_s$,
so $\varphi(0)$ is the quantity to bound and $\varphi(\pi/2)=1$. Differentiating under the integral (Lemma~\ref{lem:sg:expgrowth}~(ii)),
$\varphi'(s)=\mathbb E\big[\Phi_s\,t\langle x'_s\rangle_s\big]=t\sum_\sigma\mathbb E\big[x'_s(\sigma)e^{A_\sigma}\big]$, where $e^{A_\sigma}=\Phi_sp_\sigma(x_s)$,
$p$ denoting the Gibbs weights. The field $x'_s$ is uncorrelated with $x_s$ and has covariance $\cos s\,C$ with $g_2$, so
Lemma~\ref{lem:sg:expgrowth}~(iii) only sees the dependence of $e^{A_\sigma}$ on $g_2$, where $\partial_{g_2(b)}A_\sigma=-t\,p_b(g_2)$:
\[\varphi'(s)=-t^2\cos s\;\mathbb E\Big[\Phi_s\sum_{x,y}p_x(x_s)p_y(g_2)C(x,y)\Big]\ge-at^2\cos s\,\varphi(s)\quad(0\le s\le\tfrac\pi2),\]
because $\sum p=1$ and $C\le a$. Hence $\varphi(s)e^{at^2\sin s}$ is non-decreasing on $[0,\pi/2]$ and $\varphi(0)\le\varphi(\pi/2)e^{at^2}=e^{at^2}$.
\end{proof}

\begin{lemma}[Symmetrization and the second moment of a sub-Gaussian variable]\label{lem:sg:symm}
  \lean{ProbabilityTheory.sq_le_exp_add_exp_neg_sub_two, ProbabilityTheory.HasSubgaussianMGF.integral_sq_le,
        ProbabilityTheory.hasSubgaussianMGF_sub_integral_of_integral_exp_sub_le}
  \leanok
(i) If $X$ is sub-Gaussian with parameter $c$ (Mathlib's \texttt{HasSubgaussianMGF}) then $\mathbb EX^2\le c$. (ii) Let $F$ have exponential moments of
all orders. If $\mathbb Ee^{t(F(z_1)-F(z_2))}\le e^{ct^2/2}$ for the pair $z\sim\mu\otimes\mu$ and all $t$, then $F-\mathbb EF$ is sub-Gaussian with parameter $c$.
\end{lemma}

\begin{proof}
  \leanok
(i) From $u^2\le e^u+e^{-u}-2$ we get $t^2\mathbb EX^2\le2(e^{ct^2/2}-1)\le ct^2e^{ct^2/2}$; let $t\to0$. (ii) By Jensen, $e^{-t\mathbb EF}\le\mathbb Ee^{-tF}$, so
$\mathbb Ee^{t(F-\mathbb EF)}\le\mathbb Ee^{tF}\,\mathbb Ee^{-tF}=\mathbb Ee^{t(F(z_1)-F(z_2))}$ by Fubini.
\end{proof}

\begin{theorem}[Gaussian concentration of the free energy, Panchenko Theorem~1.2, Eq.~(1.41)]\label{thm:sg:fecentration}
  \lean{SpinGlasses.hasSubgaussianMGF_log_sum_exp, ProbabilityTheory.integral_mul_le_half_add,
        ProbabilityTheory.HasSubgaussianMGF.measureReal_abs_ge_le,
        SpinGlasses.measureReal_abs_sub_integral_ge_le_of_log_sum_exp,
        SpinGlasses.integral_sq_sub_integral_le_of_log_sum_exp}
  \leanok
  \uses{thm:sg:expmoment, lem:sg:symm}
Let $g$ be a centered Gaussian field on a finite set $\alpha$ with $\mathbb Eg(a)^2\le v$ and let $F=\log\sum_ae^{g(a)+h(a)}$ for a bias $h$.
Then $F-\mathbb EF$ is sub-Gaussian with parameter $2v$, i.e.\ $\mathbb Ee^{t(F-\mathbb EF)}\le e^{vt^2}$; consequently for $x\ge0$
\[\mathbb P(|F-\mathbb EF|\ge x)\le2\exp\Big(-\frac{x^2}{4v}\Big),\qquad\mathbb E(F-\mathbb EF)^2\le2v.\]
\end{theorem}

\begin{proof}
  \leanok
Apply Lemma~\ref{lem:sg:symm}~(ii) on $\Omega\times\Omega$: the pair of independent copies $(g(z_1),g(z_2))$ is jointly Gaussian
(Lemma~\ref{lem:sg:expgrowth}~(iv)), centered, uncorrelated, with the common covariance $C(x,y)=\mathbb Eg(x)g(y)\le\frac12(\mathbb Eg(x)^2+\mathbb Eg(y)^2)\le v$,
so Theorem~\ref{thm:sg:expmoment} bounds the exponential moments by $e^{vt^2}$. The tail bound is Chernoff's inequality for the sub-Gaussian variable on both
sides (\texttt{HasSubgaussianMGF.measureReal\_abs\_ge\_le}, from \texttt{measure\_ge\_le} and \texttt{measure\_le\_le}) and the variance bound is Lemma~\ref{lem:sg:symm}~(i).
\end{proof}

\begin{theorem}[Concentration of the free energy of the mixed $p$-spin model, Panchenko Eq.~(1.42)]\label{thm:sg:mixedconc}
  \lean{SpinGlasses.integral_mixedPSpin_sq, SpinGlasses.mixedPSpin_free_energy_hasSubgaussianMGF,
        SpinGlasses.mixedPSpin_free_energy_concentration,
        SpinGlasses.mixedPSpin_free_energy_variance_le}
  \leanok
  \uses{thm:sg:fecentration, thm:sg:mixedgauss, thm:sg:mixedcov}
Let $\beta_0=0$, $\sum_p|\beta_p|<\infty$, and let $H_N$ be the mixed $p$-spin Hamiltonian on $\{\pm1\}^N$ with $\xi(1)=\sum_p\beta_p^2$. For
$X_N=N^{-1}\log\sum_\sigma e^{H_N(\sigma)}$ and $x\ge0$,
\[\mathbb P(|X_N-\mathbb EX_N|\ge x)\le2\exp\Big(-\frac{Nx^2}{4\xi(1)}\Big),\qquad\mathrm{Var}(X_N)\le\frac{2\xi(1)}N.\]
\end{theorem}

\begin{proof}
  \leanok
The Hamiltonian is a centered Gaussian process (Theorem~\ref{thm:sg:mixedgauss}) with $\mathbb EH_N(\sigma)^2=N\xi(1)$ since the self-overlap is $1$
(Theorem~\ref{thm:sg:mixedcov}). Apply Theorem~\ref{thm:sg:fecentration} with $v=N\xi(1)$ and rescale by $N^{-1}$ (a sub-Gaussian variable with parameter $c$ times $r$ is sub-Gaussian with parameter $r^2c$), so that $X_N-\mathbb EX_N$ is sub-Gaussian with parameter $2\xi(1)/N$; the tail and variance bounds are then its Chernoff and second-moment bounds.
\end{proof}

\begin{remark}[Deviations from Panchenko's proof of Theorem~1.2, and why]\label{rem:sg:deviation}
The statements of Theorem~\ref{thm:sg:fecentration} follow the book (Theorem~1.2, Eq.~(1.41)), except for the variance constant, but the proof
differs in three points. (The description of the book's argument below is the one recorded in our formalization plan; we did not re-derive it
from the text, so the comparison is only as precise as that description.)
\begin{enumerate}
\item \emph{Two copies instead of three fields.} The book's route interpolates between independent copies $g^1,g^2$ of the field through a common field $g$,
$g^j_t=\sqrt t\,g+\sqrt{1-t}\,g^j$ ($j=1,2$), and runs a Gr\"onwall argument for $\mathbb Ee^{\lambda(F_1-F_2)}$ along that two-sided coupling. We keep
\emph{one} of the two fields fixed. Along $x_s=\cos s\,g_1+\sin s\,g_2$ against the fixed $g_2$, the derivative $x_s'=-\sin s\,g_1+\cos s\,g_2$ is uncorrelated with
$x_s$, so in Gaussian integration by parts only the dependence on $g_2$ contributes, and a \emph{single} application of the Stein identity
(Lemma~\ref{lem:sg:expgrowth}~(iii)) gives $\varphi'(s)=-t^2\cos s\,\mathbb E[\Phi_s\sum p_x p_y C]$. The endpoints are the independent pair
($s=0$) and the diagonal ($s=\pi/2$, $\varphi=1$), which is all that the Gr\"onwall step needs. The saving is that only two fields are coupled and only one Stein identity is used; the price is that the bound is proved for the difference $F(g_1)-F(g_2)$ and has to be
transferred to $F-\mathbb EF$ by a separate symmetrization (Lemma~\ref{lem:sg:symm}~(ii), Jensen's inequality $e^{-t\mathbb EF}\le\mathbb Ee^{-tF}$).
\item \emph{Trigonometric path.} As in the comparison inequality (Theorem~\ref{thm:sg:comparison}) we use $\cos s,\sin s$ instead of $\sqrt t,\sqrt{1-t}$.
The path is smooth on all of $\mathbb R$, so the dominated differentiation under the integral
(Lemma~\ref{lem:sg:expgrowth}~(ii)) holds at every $s$ with a bound uniform in $s$, and no endpoint singularity of $\frac{d}{dt}\sqrt t$ at $t=0,1$ has to be
treated separately. The Gr\"onwall factor is $e^{at^2\sin s}$ and monotonicity is used on $[0,\pi/2]$, where $\cos s\ge0$.
\item \emph{Variance constant $2v$ instead of $8v$.} The book obtains the variance bound from the tail bound (integrating $\int_0^\infty 2x\,\mathbb P(|F-\mathbb EF|\ge x)dx$ with the
tail $2e^{-x^2/(4v)}$, which gives $8v$). In Lean the conclusion of the argument is the stronger statement that $F-\mathbb EF$ is sub-Gaussian
(Mathlib's \texttt{HasSubgaussianMGF}) with parameter $2v$, and we prove, once and for all, that a sub-Gaussian variable with parameter $c$ has second
moment at most $c$ (Lemma~\ref{lem:sg:symm}~(i), from $u^2\le e^u+e^{-u}-2$ and $t\to0$), so the variance is bounded by $c=2v$ directly. The tail bound
is the same as in the book ($2e^{-x^2/(4v)}$), so the improvement is only in the variance: a factor $4$. It is still not optimal, since the Gaussian
Poincar\'e inequality would give $\mathrm{Var}\,F\le v$; we have not formalized that route.
\end{enumerate}
A smaller difference in the set-up: the reference weights $G(\sigma)>0$ of the book are replaced by a deterministic bias $h=\log G$ (which also covers an external
field), exactly as in Theorem~\ref{thm:sg:lemma11}; and the Gaussian field is an arbitrary \texttt{HasGaussianLaw} family on a finite set, so the same
statement serves every model, with the mixed $p$-spin case Theorem~\ref{thm:sg:mixedconc} a two-line application.
\end{remark}

\section{The Aizenman--Sims--Starr scheme}\label{sec:sg:ass}

Let $Z_N=\sum_\sigma e^{\beta H_N(\sigma)}$ be the partition function of the SK model at inverse temperature $\beta>0$ and
$A_N=\mathbb E\log Z_{N+1}-\mathbb E\log Z_N$ (Panchenko, Eqs.~(1.48)--(1.58)). We write $\sigma'_i=\sigma_i\in\{\pm1\}\subset\mathbb R$,
$\gamma_K=\mathcal N(0,1)^{K}$ for a finite index set $K$, and $G'$ for the auxiliary Gibbs measure
$\mathrm{count.tilted}(\beta H')$ defined below.

\begin{lemma}[A centered Gaussian family is determined in law by its Gram matrix]\label{lem:sg:gram}
  \lean{ProbabilityTheory.map_toLp_eq_multivariateGaussian, ProbabilityTheory.map_eq_map_multivariateGaussian,
        ProbabilityTheory.map_eq_of_integral_mul_eq, ProbabilityTheory.hasGaussianLaw_dotProduct_family,
        ProbabilityTheory.map_eq_map_dotProduct_family, ProbabilityTheory.integral_comp_eq_integral_dotProduct,
        ProbabilityTheory.integrable_comp_iff_dotProduct, ProbabilityTheory.measurable_dotProduct_prod,
        ProbabilityTheory.map_eq_map_dotProduct_prod, ProbabilityTheory.integral_comp_eq_integral_dotProduct_prod,
        ProbabilityTheory.integrable_comp_iff_dotProduct_prod, ProbabilityTheory.hasGaussianLaw_bilinear_family}
  \leanok
Let $(X_r)_{r\in A}$ and $(Y_r)_{r\in A}$ be centered Gaussian vectors indexed by a finite set $A$, on possibly different probability spaces,
with $\mathbb EX_rX_s=\mathbb EY_rY_s$ for all $r,s$. Then they have the same law. In particular, if $a_r\in\mathbb R^{K}$ with
$\mathbb EX_rX_s=a_r\cdot a_s$, then $X$ has the law of $(a_r\cdot\xi)_r$, $\xi\sim\gamma_K$; and if
$\mathbb EX_rX_s=a^1_r\cdot a^1_s+a^2_r\cdot a^2_s$, then $X$ has the law of $(a^1_r\cdot u+a^2_r\cdot v)_r$ for independent blocks
$u\sim\gamma_{K_1}$, $v\sim\gamma_{K_2}$. Consequently $\mathbb Ef(X)=\mathbb Ef(a^1\!\cdot u+a^2\!\cdot v)$ for measurable $f$, and $f(X)$ is
integrable if and only if $f(a^1\!\cdot u+a^2\!\cdot v)$ is.
\end{lemma}

\begin{proof}
  \leanok
A centered Gaussian vector has the law $\mathcal N(0,S)$ with $S$ its covariance matrix (the proof of
\texttt{exists\_matrix\_map\_eq\_pi\_gaussianReal}, which depends on $X$ only through $S$). The linear forms $a_r\cdot\xi$ of i.i.d.\ standard
Gaussians form a Gaussian vector (the image of $\gamma_K$ under a linear map) with covariance $a_r\cdot a_s$. For two blocks, apply the single
block to $K_1\oplus K_2$ and transport $\gamma_{K_1\oplus K_2}$ to $\gamma_{K_1}\otimes\gamma_{K_2}$ by the measure preserving map
\texttt{sumPiEquivProdPi}. Bilinear forms $u_r\cdot(Gv_r)$ of a standard Gaussian matrix are finite combinations of its entries, hence jointly Gaussian.
\end{proof}

\begin{lemma}[Gibbs averages and the cavity spin]\label{lem:sg:cavity-algebra}
  \lean{SpinGlasses.log_integral_count_tilted, SpinGlasses.sum_exp_cons_split, SpinGlasses.gram_succ, SpinGlasses.gram_eq_add,
        SpinGlasses.mul_dotProduct_mulVec_eq_dotProduct}
  \leanok
(i) For a positive observable $F$ and any energy $f$ on a finite set, $\log\langle F\rangle=\log\sum_ae^{f(a)}F(a)-\log\sum_ae^{f(a)}$, where
$\langle\cdot\rangle$ is the average for $\mathrm{count.tilted}(f)$. (ii) For fields $h,z$ on $\{\pm1\}^N$ and $s\in\mathbb R$, writing $\rho=(\varepsilon,\sigma)$,
\[\sum_{\rho}e^{\beta(h(\sigma)+\varepsilon z(\sigma)+s)}=e^{\beta s}\sum_\sigma e^{\beta h(\sigma)}\,2\cosh(\beta z(\sigma)).\]
(iii) For $\rho,\tau\in\mathbb R^{N+1}$ and $c\in\mathbb R$,
\[\sum_{i,j\ge1}c^2\rho_i\rho_j\tau_i\tau_j+\sum_{j\ge0}c^2\rho_0\rho_j\tau_0\tau_j+\sum_{i\ge1}c^2\rho_i\rho_0\tau_i\tau_0=c^2(\rho\cdot\tau)^2,\]
and for $\sigma,\tau\in\mathbb R^N$, $\sum_{i,j}c_1^2\sigma_i\sigma_j\tau_i\tau_j+\sum_{i,j}c_2^2\sigma_i\sigma_j\tau_i\tau_j=(c_1^2+c_2^2)(\sigma\cdot\tau)^2$.
Finally $c\,\sigma\cdot(U\sigma)=(c\,\sigma_i\sigma_j)_{ij}\cdot U$ for a matrix $U$.
\end{lemma}

\begin{proof}
  \leanok
(i) $\langle F\rangle=\sum_ae^{f(a)}F(a)/\sum_ae^{f(a)}$. (ii) Split the sum over $\rho$ along $(\varepsilon,\sigma)$ and use
$e^{\beta\varepsilon z}+e^{-\beta\varepsilon z}=2\cosh(\beta z)$. (iii) Split $\sum_{i,j\le N}$ at the index $0$ (\texttt{Fin.sum\_univ\_succ}) into
the corner, the row, the column and the remaining block.
\end{proof}

\begin{theorem}[The cavity decomposition in law, Panchenko Eqs.~(1.50)--(1.54)]\label{thm:sg:cavity}
  \lean{SpinGlasses.hasGaussianLaw_skHamiltonian, SpinGlasses.integral_skHamiltonian_eq_zero,
        SpinGlasses.integral_skHamiltonian_mul_skHamiltonian_units, SpinGlasses.map_skHamiltonian_succ,
        SpinGlasses.map_skHamiltonian_eq_add}
  \leanok
  \uses{lem:sg:gram, lem:sg:cavity-algebra, thm:sg:skcov, lem:sg:bilinear}
Let $u\sim\gamma_{[N]\times[N]}$ and $\zeta\sim\gamma_{[N+1]\oplus[N]}$ be independent, and put
\[H'(\sigma)=\tfrac1{\sqrt{N+1}}\,\sigma\!\cdot\!(u\sigma),\quad z(\sigma)=\tfrac1{\sqrt{N+1}}\sum_{i}(\zeta_{\mathrm{inl}(i+1)}+\zeta_{\mathrm{inr}(i)})\sigma_i,\quad
y(\sigma)=\tfrac1{\sqrt{N(N+1)}}\,\sigma\!\cdot\!(v\sigma)\]
with $v\sim\gamma_{[N]\times[N]}$ independent of $u$. Then, as random vectors indexed by configurations, in law
\[H_{N+1}(\varepsilon,\sigma)=H'(\sigma)+\varepsilon z(\sigma)+\tfrac1{\sqrt{N+1}}\zeta_{\mathrm{inl}(0)},\qquad H_N(\sigma)=H'(\sigma)+y(\sigma)\quad(N\ge1).\]
\end{theorem}

\begin{proof}
  \leanok
$H_N$ is a finite family of bilinear forms of the standard Gaussian matrix, hence a centered Gaussian vector with
$\mathbb EH_N(\sigma)H_N(\tau)=N\,R(\sigma,\tau)^2=N^{-1}(\sigma\cdot\tau)^2$ (Theorem~\ref{thm:sg:skcov}). The right hand sides are linear forms of the blocks
with coefficient vectors whose Gram matrix is computed in Lemma~\ref{lem:sg:cavity-algebra}~(iii): $(N+1)^{-1}(\rho\cdot\tau)^2$ for the first (the
overlap of $(\varepsilon,\sigma)$ and $(\varepsilon',\tau)$ is $\varepsilon\varepsilon'+\sigma\cdot\tau$), and
$\big(\tfrac1{N+1}+\tfrac1{N(N+1)}\big)(\sigma\cdot\tau)^2=N^{-1}(\sigma\cdot\tau)^2$ for the second. Apply Lemma~\ref{lem:sg:gram}; the linear forms
equal the displayed expressions because $\varepsilon^2=1$.
\end{proof}

\begin{theorem}[$\mathbb E\log Z_{N+1}$ and $\mathbb E\log Z_N$ through the cavity, Panchenko Eqs.~(1.55)--(1.56)]\label{thm:sg:partition}
  \lean{SpinGlasses.integral_log_partition_succ_eq, SpinGlasses.integral_log_partition_eq_aux}
  \leanok
  \uses{thm:sg:cavity, lem:sg:cavity-algebra}
For $\beta>0$ the functions $R_1(u,\zeta)=\log\sum_\sigma e^{\beta H'(\sigma)}\,2\cosh(\beta z(\sigma))$ and
$R_2(u,v)=\log\sum_\sigma e^{\beta(H'(\sigma)+y(\sigma))}$ are integrable and
\[\mathbb E\log Z_{N+1}=\mathbb E\,R_1(u,\zeta),\qquad\mathbb E\log Z_N=\mathbb E\,R_2(u,v)\quad(N\ge1).\]
\end{theorem}

\begin{proof}
  \leanok
By Theorem~\ref{thm:sg:cavity} and Lemma~\ref{lem:sg:gram}, $\mathbb E\log Z_{N+1}=\mathbb E\log\sum_\rho e^{\beta(H'+\varepsilon z+s)}$ with
$s=(N+1)^{-1/2}\zeta_{\mathrm{inl}(0)}$, which by Lemma~\ref{lem:sg:cavity-algebra}~(ii) is $\beta s+R_1$. The variable $\beta s$ is a centered
Gaussian coordinate, and $\log Z_{N+1}$ is integrable (Lemma~\ref{lem:sg:integrable}: $|\log Z|\le\beta|\max H|+\log|\Sigma|$), hence so is $R_1$ and
$\mathbb E\,\beta s=0$. The second identity is the second law of Theorem~\ref{thm:sg:cavity}.
\end{proof}

\begin{theorem}[The Aizenman--Sims--Starr representation, Panchenko Eq.~(1.58)]\label{thm:sg:ass}
  \lean{SpinGlasses.aizenman_sims_starr_representation, SpinGlasses.integrable_and_eq_aizenman_sims_starr}
  \leanok
  \uses{thm:sg:partition, lem:sg:cavity-algebra, lem:sg:gibbsavg}
For $\beta>0$ and $N\ge1$, with $G'_u=\mathrm{count.tilted}(\beta H'_u)$ and $\langle\cdot\rangle'_u$ its average,
\[A_N=\mathbb E_u\Big[\mathbb E_\zeta\log\big\langle 2\cosh(\beta z(\sigma))\big\rangle'_u-\mathbb E_v\log\big\langle e^{\beta y(\sigma)}\big\rangle'_u\Big],\]
where $u,\zeta,v$ are independent and $z$, $y$ are independent of the auxiliary Gibbs measure $G'_u$.
\end{theorem}

\begin{proof}
  \leanok
By Theorem~\ref{thm:sg:partition} and Fubini, $A_N=\mathbb E_u[\mathbb E_\zeta R_1(u,\zeta)-\mathbb E_vR_2(u,v)]$. By Lemma~\ref{lem:sg:cavity-algebra}~(i),
$R_1(u,\zeta)=\log Z'(u)+\log\langle2\cosh\beta z\rangle'_u$ and $R_2(u,v)=\log Z'(u)+\log\langle e^{\beta y}\rangle'_u$ with the \emph{same}
$Z'(u)=\sum_\sigma e^{\beta H'_u(\sigma)}$, which does not depend on $\zeta,v$ and cancels. Integrability of $R_1(u,\cdot)$, $R_2(u,\cdot)$ for almost
every $u$ is part of Fubini, and subtracting the constant $\log Z'(u)$ preserves it.
\end{proof}

\begin{theorem}[Covariances of the cavity fields, Panchenko Eq.~(1.59)]\label{thm:sg:cavity-cov}
  \lean{SpinGlasses.integral_cavity_z_mul_z, SpinGlasses.integral_cavity_y_mul_y, SpinGlasses.cavity_z_eq_dotProduct,
        SpinGlasses.hasGaussianLaw_cavity_z, SpinGlasses.integral_cavity_z_eq_zero,
        SpinGlasses.hasGaussianLaw_cavity_y, SpinGlasses.integral_cavity_y_eq_zero}
  \leanok
  \uses{lem:sg:gram, lem:sg:cavity-algebra}
For $N\ge1$ and $\sigma,\tau\in\mathbb R^N$, with $a_N=N/(N+1)$ and $R=N^{-1}\sigma\cdot\tau$,
\[\mathbb E\,z(\sigma)z(\tau)=2a_NR,\qquad\mathbb E\,y(\sigma)y(\tau)=a_NR^2.\]
Both depend on $\sigma,\tau$ only through the overlap. Moreover, for any finite family of coefficient vectors $\sigma^s$, $(z(\sigma^s))_s$ and
$(y(\sigma^s))_s$ are centered jointly Gaussian families.
\end{theorem}

\begin{proof}
  \leanok
$z(\sigma)$ and $y(\sigma)$ are linear forms of standard Gaussians with coefficient vectors $\frac1{\sqrt{N+1}}\sigma$ (on both the $\mathrm{inl}$ and $\mathrm{inr}$
coordinates) and $\frac1{\sqrt{N(N+1)}}\sigma_i\sigma_j$, so the covariance is the dot product of coefficients:
$\frac2{N+1}\sigma\cdot\tau$ and $\frac1{N(N+1)}(\sigma\cdot\tau)^2$; being linear forms of standard Gaussians, the families are jointly
Gaussian and centered.
\end{proof}

\begin{lemma}[Cesaro averages and the liminf]\label{lem:sg:cesaro}
  \lean{Filter.Eventually.sub_le_inv_mul_sum_range, Filter.Eventually.inv_mul_sum_range_le_add,
        Filter.liminf_le_liminf_inv_mul_sum_range}
  \leanok
If $c\le u_n$ eventually, then for every $\varepsilon>0$ eventually $c-\varepsilon\le n^{-1}\sum_{j<n}u_j$ (and symmetrically for upper bounds). If $u$ is bounded,
$\liminf_nu_n\le\liminf_nn^{-1}\sum_{j<n}u_j$.
\end{lemma}

\begin{proof}
  \leanok
The negative part $\min(u_n-c,0)$ vanishes eventually, so its Cesaro averages tend to $0$ (Mathlib's \texttt{Filter.Tendsto.cesaro}) and bound
$n^{-1}\sum_{j<n}u_j-c$ from below. For the liminf, apply this with $c=\liminf u-\varepsilon/2$.
\end{proof}

\begin{theorem}[The free energy as a Cesaro average of the increments, Panchenko Eq.~(1.49)]\label{thm:sg:cesaro}
  \lean{SpinGlasses.integral_log_partition_zero, SpinGlasses.inv_mul_integral_log_partition_eq,
        SpinGlasses.liminf_increment_le_liminf_free_energy}
  \leanok
  \uses{lem:sg:cesaro}
For all $N$, $F_N=N^{-1}\mathbb E\log Z_N=N^{-1}\sum_{j<N}A_j$ (with $Z_0=1$). Consequently, if $(A_N)$ is bounded,
$\liminf_NA_N\le\liminf_NF_N$.
\end{theorem}

\begin{proof}
  \leanok
The empty system has the single configuration of energy $0$, so $\mathbb E\log Z_0=0$; the sum telescopes. The inequality is Lemma~\ref{lem:sg:cesaro}.
\end{proof}

\begin{remark}[Deviations from Panchenko's treatment of the cavity method, and why]\label{rem:sg:ass-deviation}
\begin{enumerate}
\item \emph{Equality in law instead of a pathwise decomposition.} The book writes $H_{N+1}(\varepsilon,\sigma)=H'_N(\sigma)+\varepsilon z_N(\sigma)+(N+1)^{-1/2}g_{N+1,N+1}$
\emph{pathwise}, in terms of the entries of the same disorder matrix, and reads the independence of $H'$, $z$ and $g_{N+1,N+1}$ off the independence of
the entries. We instead prove only an equality in law (Theorem~\ref{thm:sg:cavity}) by comparing covariances (Lemma~\ref{lem:sg:gram}). The disorder in
Lean is the \emph{nested} product $\mathrm{Fin}(N+1)\to\mathrm{Fin}(N+1)\to\mathbb R$, and splitting it into blocks would need a measure preserving
change of variables between iterated and flat product measures (Mathlib has none for the iterated product); the covariance check is a finite sum identity
and reuses the Gaussian-matrix lemmas already proved for the Guerra--Toninelli theorem. Since $\mathbb E\log Z_{N+1}$ is a functional of the law, nothing is
lost for the Aizenman--Sims--Starr formula.
\item \emph{The same trick for $H_N\overset d=H'+y$} (Eqs.~(1.53)--(1.54)): the book matches the covariance $N R^2=\frac{N^2}{N+1}R^2+\frac N{N+1}R^2$ and
invokes the uniqueness of Gaussian laws; we do the same, with the uniqueness provided by Mathlib's \texttt{IsGaussian.ext}.
\item \emph{The cavity site is $0$, not $N+1$.} We use \texttt{Fin.cons}/\texttt{Fin.succ} so that sums split by \texttt{Fin.sum\_univ\_succ}; by relabeling
invariance of the disorder this is the same statement.
\item \emph{The Cesaro statement carries boundedness hypotheses.} Mathlib's \texttt{liminf} on $\mathbb R$ is junk-valued for unbounded sequences, so the
inequality $\liminf A_N\le\liminf F_N$ of Theorem~\ref{thm:sg:cesaro} assumes $(A_N)$ is bounded. The book does not state this because the increments are bounded;
that bound is proved in Theorem~\ref{thm:sg:increment-bounds} below ($\log2-\beta^2/2\le A_N\le\log2+\beta^2$), which removes the hypothesis
(Corollary~\ref{cor:sg:cesaro-unconditional}). When $A_N$ converges, Mathlib's \texttt{Filter.Tendsto.cesaro} gives $F_N\to\lim A_N$ directly.
\end{enumerate}
\end{remark}

\section{Gram matrix approximation of Gibbs averages}\label{sec:sg:gram-approx}

This section proves Theorem~1.3 of the book and applies it to the Aizenman--Sims--Starr increments (Eq.~(1.73)). Throughout, $\iota$ is a finite set,
$w=(w_i)_{i\in\iota}$ a probability vector, and $\langle f\rangle=\sum_iw_if_i$. The statements do not mention a Hilbert space: the quantity
$\Phi(\Gamma)=\mathbb E\log\langle2\cosh z\rangle-\mathbb E\log\langle e^{y}\rangle$ of the book depends on the atomic measure
$\Gamma=\sum_iw_i\delta_{\rho^i}$ and on the processes $z,y$ only through $w$ and the Gram data $q_{ij}=\rho^i\cdot\rho^j$ (the covariances being
$\mathbb Ez_iz_j=C_z(q_{ij})$, $\mathbb Ey_iy_j=C_y(q_{ij})$). For $a\ge0$ let $\mathrm{cl}_a(x)=\max(-a,\min(x,a))$ and
$T_a(x)=\max(\min(x,e^a),e^{-a})$. A random variable $X$ is \emph{sub-Gaussian with parameter $c$} if $\mathbb Ee^{tX}\le e^{ct^2/2}$ for all $t\in\mathbb R$
(and $e^{tX}$ is integrable); a centered Gaussian with $\mathbb EX^2\le v$ is sub-Gaussian with parameter $v$, since $\mathbb Ee^{tX}=e^{t^2\mathbb EX^2/2}$.

\begin{lemma}[Exponential moments of $|X|$]\label{lem:sg:absmoment}
  \lean{ProbabilityTheory.HasGaussianLaw.hasSubgaussianMGF_of_integral_sq_le, ProbabilityTheory.HasSubgaussianMGF.integral_exp_abs_mul_le,
        ProbabilityTheory.HasSubgaussianMGF.integrable_of_measurable, ProbabilityTheory.HasSubgaussianMGF.integrable_two_cosh,
        ProbabilityTheory.HasSubgaussianMGF.integral_two_cosh_le, ProbabilityTheory.HasSubgaussianMGF.integral_exp_le}
  \leanok
Let $X$ be measurable and sub-Gaussian with parameter $c$. Then for every $t\in\mathbb R$, $e^{t|X|}$ is integrable and $\mathbb Ee^{t|X|}\le2e^{ct^2/2}$; $X$ and
$\cosh X$ are integrable, $\mathbb E\,2\cosh X\le2e^{c/2}$ and $\mathbb Ee^{X}\le e^{c/2}$.
\end{lemma}

\begin{proof}
  \leanok
Pointwise $e^{t|x|}\le e^{tx}+e^{-tx}$, so $\mathbb Ee^{t|X|}\le\mathbb Ee^{tX}+\mathbb Ee^{-tX}\le2e^{ct^2/2}$. Integrability of $X$ follows from $|x|\le e^{|x|}$,
and the remaining claims are the cases $t=\pm1$ of the definition.
\end{proof}

\begin{lemma}[Elementary inequalities for the truncation]\label{lem:sg:clamp}
  \lean{SpinGlasses.abs_log_le_add_inv, SpinGlasses.abs_log_sub_log_clamp_le, SpinGlasses.abs_clamp_sub_clamp_le,
        SpinGlasses.abs_log_sub_log_le, SpinGlasses.abs_log_clamp_sub_le, SpinGlasses.abs_log_clamp_le,
        SpinGlasses.abs_sub_comp_clamp_le, SpinGlasses.inv_sum_mul_le, SpinGlasses.sum_mul_pos}
  \leanok
Let $a\ge0$. (i) $|\log x|\le x+x^{-1}$ for $x>0$. (ii) $|\log x-\log T_a(x)|\le e^{-a}(x+x^{-1})$ for $x>0$. (iii) $T_a$ is $1$-Lipschitz, $\log T_a\in[-a,a]$,
and $|\log T_a(x)-\log T_a(x')|\le e^a|x-x'|$. (iv) If $\varphi>0$ and $\varphi(x)\le Ae^{|x|}$, then
$|\varphi(x)-\varphi(\mathrm{cl}_a(x))|\le2Ae^{-2a}e^{3|x|}$. (v) For a probability vector $w$ and $x_i>0$, $(\sum_iw_ix_i)^{-1}\le\sum_iw_ix_i^{-1}$.
\end{lemma}

\begin{proof}
  \leanok
(i) $\log x\le x-1$ applied to $x$ and $x^{-1}$. (ii) If $x\ge e^a$ then $T_a(x)=e^a$ and $0\le\log x-a\le e^{\log x-a}=xe^{-a}$ (using $u\le e^u$); if $x\le e^{-a}$
then $T_a(x)=e^{-a}$ and $0\le-a-\log x\le e^{-\log x-a}=x^{-1}e^{-a}$; otherwise the difference is $0$. (iii) $T_a$ is a composition of $\min$ and $\max$ with
constants. For $s,t\ge e^{-a}$ with $s\ge t$, $\log s-\log t=\log(s/t)\le s/t-1\le(s-t)e^a$. (iv) If $|x|\le a$ the difference vanishes. If $|x|>a$, then
$|\mathrm{cl}_a(x)|\le a<|x|$, so both $\varphi(x)$ and $\varphi(\mathrm{cl}_ax)$ lie in $(0,Ae^{|x|}]$ and $|\varphi(x)-\varphi(\mathrm{cl}_ax)|\le Ae^{|x|}\le Ae^{-2a}e^{3|x|}$
because $3|x|-2a\ge|x|$. (v) Jensen's inequality for the convex function $x\mapsto x^{-1}$ on $(0,\infty)$.
\end{proof}

\begin{theorem}[Truncation of the Gibbs free energy, Panchenko Eqs.~(1.65), (1.68)--(1.72)]\label{thm:sg:gibbs-truncation}
  \lean{SpinGlasses.integral_sum_mul_exp_abs_le, SpinGlasses.integrable_integral_le_of_le_sum_exp_abs,
        SpinGlasses.integrable_sum_mul_comp, SpinGlasses.integrable_log_sum_mul_comp,
        SpinGlasses.abs_integral_log_sub_integral_log_clamp_le}
  \leanok
  \uses{lem:sg:absmoment, lem:sg:clamp}
Let $\varphi:\mathbb R\to(0,\infty)$ be continuous with $\varphi(x)\le Ae^{|x|}$ and $\varphi(x)^{-1}\le Be^{|x|}$, let $g_i$ ($i\in\iota$) be measurable and
sub-Gaussian with parameter $v$ on a probability space, and put $X=\sum_iw_i\varphi(g_i)$, $X_a=\sum_iw_i\varphi(\mathrm{cl}_a(g_i))$. Then $\log X$ is integrable and for $a\ge0$
\[\Big|\mathbb E\log X-\mathbb E\log T_a(X_a)\Big|\le\Big((2A+2B)e^{v/2}+4Ae^{9v/2}\Big)e^{-a},\]
a bound independent of $\iota$, $w$ and of the joint law of the $g_i$.
\end{theorem}

\begin{proof}
  \leanok
Since $\varphi(g_i)\le Ae^{|g_i|}$ and, by Lemma~\ref{lem:sg:clamp}~(v), $X^{-1}\le\sum_iw_i\varphi(g_i)^{-1}\le B\sum_iw_ie^{|g_i|}$, Lemma~\ref{lem:sg:absmoment} gives
$\mathbb EX\le2Ae^{v/2}$ and $\mathbb EX^{-1}\le2Be^{v/2}$; with Lemma~\ref{lem:sg:clamp}~(i) this shows that $\log X$ is integrable (and $\log T_a(X_a)\in[-a,a]$ is bounded).
Pointwise, by Lemma~\ref{lem:sg:clamp}~(ii), (iii),
\[|\log X-\log T_a(X_a)|\le|\log X-\log T_a(X)|+|\log T_a(X)-\log T_a(X_a)|\le e^{-a}(X+X^{-1})+e^a|X-X_a|,\]
and by Lemma~\ref{lem:sg:clamp}~(iv), $|X-X_a|\le\sum_iw_i|\varphi(g_i)-\varphi(\mathrm{cl}_ag_i)|\le2Ae^{-2a}\sum_iw_ie^{3|g_i|}$. Taking expectations and using
Lemma~\ref{lem:sg:absmoment} ($\mathbb E\sum_iw_ie^{3|g_i|}\le2e^{9v/2}$) gives $e^{-a}(2A+2B)e^{v/2}+e^a\cdot2Ae^{-2a}\cdot2e^{9v/2}$.
\end{proof}

\begin{theorem}[Polynomial approximation, Panchenko Eq.~(1.66)]\label{thm:sg:poly}
  \lean{SpinGlasses.exists_polynomial_integral_log_approx}
  \leanok
  \uses{thm:sg:gibbs-truncation}
In the setting of Theorem~\ref{thm:sg:gibbs-truncation}, for every $\varepsilon>0$ there are $a\ge0$ and a polynomial $p$, depending only on $\varepsilon,v,\varphi,A,B$, such that
for all finite families $(g_i)$ and probability vectors $w$ as above, $|\mathbb E\log X-\mathbb E\,p(X_a)|\le\varepsilon$.
\end{theorem}

\begin{proof}
  \leanok
Choose $a\ge0$ with $Ce^{-a}\le\varepsilon/2$, $C$ the constant of Theorem~\ref{thm:sg:gibbs-truncation}. Since $\varphi(\mathrm{cl}_ag_i)\le Ae^a$, $X_a\in[0,Ae^a]$. The function $\log T_a$ is continuous,
so by the Weierstrass theorem (\texttt{exists\_polynomial\_near\_of\_continuousOn}) there is a polynomial $p$ with $|p-\log T_a|<\varepsilon/2$ on $[0,Ae^a]$; hence

\begin{theorem}[Replica expansion, Panchenko Eq.~(1.67)]\label{thm:sg:gibbs-replica}
  \lean{ProbabilityTheory.hasGaussianLaw_comp_index, ProbabilityTheory.integral_comp_comp_eq_integral_multivariateGaussian,
        ProbabilityTheory.map_toLp_dotProduct_pi_gaussianReal, SpinGlasses.integral_pow_sum_mul_comp_eq,
        SpinGlasses.integral_eval_sum_mul_comp_eq}
  \leanok
  \uses{lem:sg:gram}
Let $(g_i)_{i\in\iota}$ be a centered Gaussian family, $\psi$ bounded and measurable and $w$ a probability vector. Then for every $k$
\[\mathbb E\Big(\sum_iw_i\psi(g_i)\Big)^k=\sum_{j:[k]\to\iota}\Big(\prod_{l\le k}w_{j_l}\Big)\int\prod_{l\le k}\psi(x_l)\,d\mathcal N\big(0,(\mathbb Eg_{j_l}g_{j_{l'}})_{l,l'\le k}\big)(x),\]
and the same holds for $\mathbb Ep(\sum_iw_i\psi(g_i))$ with the coefficients of a polynomial $p$. In particular the $k$-th moment of a Gibbs average
is the Gibbs average over $k$ replicas of a function of the covariance matrix of the replicas only.
\end{theorem}

\begin{proof}
  \leanok
Expand $(\sum_iw_i\psi(g_i))^k=\sum_{j:[k]\to\iota}\prod_lw_{j_l}\psi(g_{j_l})$ (a finite sum, \texttt{Fintype.sum\_pow}); the terms are bounded, so the expectation passes inside.
For fixed $j$, $(g_{j_l})_{l\le k}$ is a centered Gaussian vector (a linear image of $(g_i)_i$) with covariance matrix $S_j=(\mathbb Eg_{j_l}g_{j_{l'}})$, hence has the law
$\mathcal N(0,S_j)$ (Lemma~\ref{lem:sg:gram}, the first part), and $\mathbb E\prod_l\psi(g_{j_l})$ is the displayed Gaussian integral.
\end{proof}

\begin{lemma}[Continuity of Gaussian integrals in the covariance matrix]\label{lem:sg:gausscont}
  \lean{ProbabilityTheory.continuousOn_integral_multivariateGaussian,
        ProbabilityTheory.continuousOn_integral_multivariateGaussian_of_bcf,
        SpinGlasses.continuousOn_integral_prod_multivariateGaussian,
        SpinGlasses.abs_integral_prod_multivariateGaussian_le}
  \leanok
Let $g$ be bounded and continuous on $\mathbb R^k$. Then $S\mapsto\int g\,d\mathcal N(0,S)$ is continuous on the cone of positive semidefinite matrices.
Consequently, for $\psi$ continuous with $|\psi|\le M$ and a continuous covariance function $C$, the replica functional
$F_k(Q)=\int\prod_{l\le k}\psi(x_l)\,d\mathcal N\big(0,(C(Q_{ll'}))_{l,l'}\big)$ is bounded by $M^k$ and continuous on the set of $Q\in\mathbb R^{k\times k}$ for which
$(C(Q_{ll'}))$ is positive semidefinite.
\end{lemma}

\begin{proof}
  \leanok
The space of matrices is first countable, so it suffices to treat sequences $S_n\to S$ of positive semidefinite matrices. The characteristic function of
$\mathcal N(0,S)$ is $t\mapsto e^{-t^\top St/2}$, which converges pointwise as $n\to\infty$; by L\'evy's continuity theorem (Mathlib's
\texttt{ProbabilityMeasure.tendsto\_of\_tendsto\_charFun}) $\mathcal N(0,S_n)\to\mathcal N(0,S)$ weakly, i.e.\ $\int g\,d\mathcal N(0,S_n)\to\int g\,d\mathcal N(0,S)$.
The bound is $|\prod_l\psi(x_l)|\le M^k$ against a probability measure; continuity of $F_k$ is the composition with the continuous map $Q\mapsto(C(Q_{ll'}))$.
\end{proof}

\begin{theorem}[Theorem~1.3: Gram matrix approximation of the Gibbs free energy, Panchenko Eq.~(1.64)]\label{thm:sg:gram13}
  \lean{SpinGlasses.exists_gram_approx, SpinGlasses.exists_gram_approx_free_energy, SpinGlasses.sum_prod_mul_comp_castLE,
        SpinGlasses.sum_mul_sum_comm, SpinGlasses.exists_continuousOn_gram_approx}
  \leanok
  \uses{thm:sg:poly, thm:sg:gibbs-replica, lem:sg:gausscont}
Let $C_z,C_y:\mathbb R\to\mathbb R$ be continuous, $v_z,v_y\ge0$, $\varepsilon>0$. There are $m\in\mathbb N$ and a bounded function $F$ on $m\times m$ matrices, continuous
on $\{Q:(C_z(Q_{ll'}))\text{ and }(C_y(Q_{ll'}))\text{ are positive semidefinite}\}$, such that for every finite set $\iota$, probability vector $w$ on $\iota$,
array $q\in\mathbb R^{\iota\times\iota}$, and all centered Gaussian families $z$, $y$ (on arbitrary probability spaces) with $\mathbb Ez_iz_j=C_z(q_{ij})$,
$\mathbb Ey_iy_j=C_y(q_{ij})$, $\mathbb Ez_i^2\le v_z$, $\mathbb Ey_i^2\le v_y$,
\[\Big|\Big(\mathbb E\log\langle2\cosh z\rangle-\mathbb E\log\langle e^{y}\rangle\Big)-\sum_{j:[m]\to\iota}\Big(\prod_{l\le m}w_{j_l}\Big)F\big((q_{j_lj_{l'}})_{l,l'\le m}\big)\Big|\le\varepsilon.\]
The right hand side is the $\Gamma$-average of the continuous bounded function $F$ of the Gram matrix of $m$ i.i.d.\ replicas, and $m,F$ do not depend on $\iota,w,q$.
\end{theorem}

\begin{proof}
  \leanok
Apply Theorem~\ref{thm:sg:poly} with accuracy $\varepsilon/2$ to $\varphi=2\cosh$ ($A=2$, $B=1$, since $2\cosh x\le2e^{|x|}$ and $(2\cosh x)^{-1}\le1\le e^{|x|}$) and to $\varphi=\exp$
($A=B=1$), obtaining levels $a_z,a_y$ and polynomials $p_z,p_y$; note $2\cosh(\mathrm{cl}_ax)=2\min(\cosh x,\cosh a)$, the book's truncation $\mathrm{ch}_a$. By
Theorem~\ref{thm:sg:gibbs-replica}, $\mathbb Ep_z(\langle 2\cosh\mathrm{cl}_{a_z}z\rangle)=\sum_k p_{z,k}\sum_{j:[k]\to\iota}\prod_lw_{j_l}F^z_k\big((C_z(q_{j_lj_{l'}}))\big)$
with $F^z_k$ the Gaussian functional of Lemma~\ref{lem:sg:gausscont}, and likewise for $y$. Let $m=\max(\deg p_z,\deg p_y)$ and let $F(Q)$ be the difference of the
two sums with $F^\cdot_k$ evaluated at the upper-left $k\times k$ block of $Q$. Since $\sum_iw_i=1$, summing out the last $m-k$ replicas does not change a function of the first $k$
(\texttt{sum\_prod\_mul\_comp\_castLE}), so the Gibbs average of $F$ over $m$ replicas equals the sum over $k$ of the $k$-replica averages. Continuity of $F$ on the stated set
uses that a principal submatrix of a positive semidefinite matrix is positive semidefinite; boundedness is $|F|\le\sum_k|p_{z,k}|(2e^{a_z})^k+\sum_k|p_{y,k}|e^{a_yk}$.
\end{proof}

\section{The Aizenman--Sims--Starr increments: bounds and the overlap approximation}\label{sec:sg:ass-approx}

We return to the notation of Section~\ref{sec:sg:ass}: $G'_u=\mathrm{count.tilted}(\beta H'_u)$ with Gibbs weights $\mathrm{wt}_u(\sigma)=e^{\beta H'_u(\sigma)}/\sum_\tau e^{\beta H'_u(\tau)}$,
and the cavity fields $z,y$ with covariances $2\beta^2a_NR$ and $\beta^2a_NR^2$ (Theorem~\ref{thm:sg:cavity-cov}; the factor $\beta$ is absorbed into $z,y$).

\begin{lemma}[Jensen bounds for Gibbs averages]\label{lem:sg:jensen}
  \lean{SpinGlasses.integral_log_le_log_integral, SpinGlasses.integral_sum_mul_le,
        SpinGlasses.log_two_le_integral_log_sum_mul_two_cosh, SpinGlasses.integral_log_sum_mul_two_cosh_le,
        SpinGlasses.zero_le_integral_log_sum_mul_exp, SpinGlasses.integral_log_sum_mul_exp_le,
        SpinGlasses.exp_le_exp_abs_mul, SpinGlasses.inv_exp_le_exp_abs_mul, SpinGlasses.two_cosh_le_exp_abs_mul,
        SpinGlasses.inv_two_cosh_le_exp_abs_mul}
  \leanok
  \uses{lem:sg:absmoment, thm:sg:gibbs-truncation}
Let $g_i$ be measurable and sub-Gaussian with parameter $v$ and $w$ a probability vector. Then
$\log2\le\mathbb E\log\langle2\cosh g\rangle\le\log2+v/2$, and if the $g_i$ are centered, $0\le\mathbb E\log\langle e^{g}\rangle\le v/2$.
\end{lemma}

\begin{proof}
  \leanok
For a positive integrable $X$ with integrable logarithm, $\log x\le x/c-1+\log c$ with $c=\mathbb EX$ gives $\mathbb E\log X\le\log\mathbb EX$. For $X=\langle2\cosh g\rangle$,
$\mathbb EX\le\max_i\mathbb E\,2\cosh g_i\le2e^{v/2}$ (Lemma~\ref{lem:sg:absmoment}), and $X\ge2$ because $\cosh\ge1$. For $X=\langle e^g\rangle$, $\mathbb EX\le e^{v/2}$, and by convexity
of $\exp$, $\log\langle e^g\rangle\ge\langle g\rangle$, whose expectation is $\sum_iw_i\mathbb Eg_i=0$. Integrability of the logarithms is Theorem~\ref{thm:sg:gibbs-truncation}.
\end{proof}

\begin{theorem}[A priori bounds for the increments]\label{thm:sg:increment-bounds}
  \lean{SpinGlasses.sk_gibbsWeight, SpinGlasses.sk_overlap_self, SpinGlasses.sk_cavity_z, SpinGlasses.sk_cavity_y,

\end{theorem}

\begin{proof}

By Theorem~\ref{thm:sg:ass}, $A_N=\mathbb E_uF(u)$ with $F(u)=\mathbb E_\zeta\log\langle2\cosh z\rangle'_u-\mathbb E_v\log\langle e^{y}\rangle'_u$ (here $z,y$ include the factor $\beta$),

\end{proof}

\begin{remark}[Deviations from Panchenko's treatment of Theorem~1.3, and why]\label{rem:sg:gram-deviation}
A longer, less terse explanation of every item below, including which items weaken the theorem, is in Section~\ref{sec:sg:gram-background}.
\begin{enumerate}
\item \emph{No Hilbert space, finitely supported measures.} The book states Theorem~1.3 for atomic probability measures $\Gamma$ on the unit sphere of a separable Hilbert space.
Since $\Phi(\Gamma)$ only involves $w$ and the Gram data, we state it for a finite family with arbitrary covariance functions; the cavity measures $G'_u$ of the SK model are finitely supported.
Extending to countably many atoms is a limiting argument that we have not formalized.
\item \emph{Truncation by moments rather than by the concentration (1.69).} The book controls $\log\langle\cosh z\rangle$ through its Gaussian concentration. Only the expectation of the truncation error is needed,
and it is bounded by $e^{-a}\,\mathbb E(X+X^{-1})$ plus an exponential moment of the field; both are uniform in the number of atoms, so the earlier concentration theorem is not used here.
\item \emph{Uniformity.} The truncation level and the polynomials depend only on $\varepsilon$ and on bounds $v_z,v_y$ for the variances, not on the covariance functions; the covariance functions enter only the
Gaussian integrals $F_k$, which are continuous in the Gram matrix on the cone where the covariance matrix is positive semidefinite (Lemma~\ref{lem:sg:gausscont}). In the SK application the covariances $2\beta^2a_NR$ and $\beta^2a_NR^2$
depend on $N$ through $a_N=N/(N+1)$, so the function $F$ of Theorem~\ref{thm:sg:overlap-approx} depends on $N$ through the Gaussian integrals only; the statement is nevertheless uniform in $N$. Passing to the limit $a_N\to1$ is not formalized; Section~\ref{sec:sg:aN-limit} explains where $a_N$ comes from, why the limit is needed, and how we expect it to be proved.
\item \emph{The truncation of the exponential term.} For $\varphi=2\cosh$, $2\cosh(\mathrm{cl}_ax)=2\min(\cosh x,\cosh a)$ is the book's $\mathrm{ch}_a$. For $\varphi=\exp$ we truncate the field in the same way.
\end{enumerate}
\end{remark}

\section{Background: why Theorem~1.3 matters, and exactly how our version differs from the book's}\label{sec:sg:gram-background}

This section is an explanation, not a collection of new statements. It is written for a reader who knows the statistical-physics side of the story only loosely, and it answers three questions:
what is Theorem~1.3 of Panchenko's book for, what precisely did we prove instead, and which of our choices make the Lean statement weaker, equal or stronger than the book's. Where we describe what the book
does in later sections we are reporting our reading of the book and of our own formalization plan; we have not re-derived those later arguments here, and we say so when a claim is only expected.

\subsection*{The problem Theorem~1.3 solves}

The free energy $F_N=N^{-1}\mathbb E\log Z_N$ of the SK model depends on the whole random Gibbs measure $G_N$ on $\{-1,+1\}^N$, a measure on a space whose dimension grows with $N$.
To compute $\lim F_N$ one wants to compare models of different sizes, and for that one needs a space in which all the $G_N$ live together. Physics suggests the right coordinates: the \emph{overlaps}
$R_{l,l'}=N^{-1}\sigma^l\cdot\sigma^{l'}$ of replicas $\sigma^1,\sigma^2,\dots$ drawn independently from $G_N$. The overlap array has a fixed shape ($\mathbb N\times\mathbb N$) whatever $N$ is, so
its law can converge as $N\to\infty$, and the limiting law is what the book calls the data of the \emph{asymptotic Gibbs measure}. (In Lean this is Section~\ref{sec:sg:asymptotic-gibbs}.)

The Aizenman--Sims--Starr scheme (Section~\ref{sec:sg:ass}) reduces $\lim F_N$ to the increments $A_N=\mathbb E\log Z_{N+1}-\mathbb E\log Z_N$, and writes $A_N$ as
$\Phi(G'_N)$, where $\Phi(\Gamma)=\mathbb E\log\langle2\cosh z\rangle_\Gamma-\mathbb E\log\langle e^{y}\rangle_\Gamma$ is a functional of the Gibbs measure $\Gamma=G'_N$ and of two Gaussian processes $z,y$ built from it.
The functional $\Phi$ is a complicated nonlinear object (an expectation of a logarithm of a Gibbs average). Convergence of the \emph{overlap law} does not obviously imply anything about $\Phi$.

\emph{Theorem~1.3 is the bridge.} It says that $\Phi(\Gamma)$ is, up to an error $\varepsilon$ that we may choose, equal to a plain Gibbs average of a nice function of the overlaps:
$\Phi(\Gamma)\approx\langle F(R)\rangle_\Gamma=\int F\big((\rho^l\cdot\rho^{l'})_{l,l'\le m}\big)\,d\Gamma^{\otimes m}$, with $F$ bounded and continuous, and with $m$ and $F$ not depending on $\Gamma$.
A Gibbs average of a bounded continuous function of the overlap matrix depends on $\Gamma$ only through the \emph{law of the overlap array}, and this is continuous for weak convergence of that law.
So convergence of overlap laws, which is available by compactness (Theorem~\ref{thm:sg:asymptotic-overlap}), gives convergence of $A_N$ up to $\varepsilon$, for every $\varepsilon$. This is the engine behind the lower bound
$\liminf A_N\ge\inf_G\Phi(G)$ as we understand the book's plan; it is not yet formalized, and the Lean content of this chapter stops short of it (Remark~\ref{rem:sg:asymptotic-deviation}).

\subsection*{The idea of the proof, in plain words}

The danger in $\Phi$ is the logarithm: $\log\langle e^{z}\rangle$ is not a nice function of the random field $z$, and $\langle\cdot\rangle$ may involve exponentially many atoms. The proof tames it in four steps.
\begin{enumerate}
\item \emph{Truncate.} Clamp the argument of the logarithm into $[e^{-a},e^{a}]$ and clamp the field at level $\pm a$. Because the Gibbs average of $e^{|z|}$-type quantities has bounded moments, the expected error is $O(e^{-a})$;
we show this using only first moments and Jensen's inequality (Theorem~\ref{thm:sg:gibbs-truncation}), and the bound does \emph{not} depend on how many atoms $\Gamma$ has.
\item \emph{Polynomial.} On the bounded interval $[0,Ae^{a}]$ the truncated logarithm is a continuous function, hence within $\varepsilon$ of a polynomial $p$ (Weierstrass). Now $\Phi$ is approximated by $\mathbb E\,p(\langle\psi(z)\rangle)$.
\item \emph{Expand the powers.} A power $\langle\psi(z)\rangle^k$ is the average over $k$ independent replicas of $\prod_l\psi(z_{j_l})$. Taking the Gaussian expectation inside, $\mathbb E\prod_l\psi(z_{j_l})$ is an integral
against a centered Gaussian whose covariance matrix is $(C_z(\rho^{j_l}\cdot\rho^{j_{l'}}))_{l,l'}$, and a centered Gaussian law is determined by its covariance matrix. So this expectation is a function $F_k$ of the
$k\times k$ Gram matrix of the replicas, nothing else about $\Gamma$ enters.
\item \emph{Continuity.} $S\mapsto\int g\,d\mathcal N(0,S)$ is continuous (L\'evy's theorem on characteristic functions), so $F_k$ is continuous, and bounded because $\psi$ is.
\end{enumerate}
The same four steps are what the Lean files do; the only structural difference is in step~1 (see decision~C below).

\subsection*{What the book states, and what we state}

\begin{itemize}
\item \textbf{Book (as we use it):} for atomic probability measures $\Gamma=\sum_{i}w_i\delta_{\rho^i}$ on the unit sphere of a separable Hilbert space $H$ (countably many atoms $\rho^i$), and Gaussian processes
$z,y$ indexed by $H$ with covariances $C_z(\rho^1\cdot\rho^2)$, $C_y(\rho^1\cdot\rho^2)$: for every $\varepsilon>0$ there are $m$ and a continuous bounded $F$ of $m\times m$ Gram matrices, such that
$|\Phi(\Gamma)-\langle F(R)\rangle_\Gamma|\le\varepsilon$ for \emph{all} such $\Gamma$.
\item \textbf{Lean (Theorem~\ref{thm:sg:gram13}):} the same conclusion for every \emph{finite} family $(w_i,q_{ij})_{i\in\iota}$ ($\iota$ any finite type), where $q_{ij}$ plays the role of $\rho^i\cdot\rho^j$ and
$z,y$ are any centered Gaussian families with $\mathbb Ez_iz_j=C_z(q_{ij})$, $\mathbb Ey_iy_j=C_y(q_{ij})$ and variances at most $v_z,v_y$. The number $m$ and the function $F$ depend only on $\varepsilon,C_z,C_y,v_z,v_y$,
not on $\iota$, $w$ or $q$.
\end{itemize}

\subsection*{Decision A: we do not mention a Hilbert space}

\emph{What we did.} The statement has no Hilbert space, no unit sphere and no points $\rho^i$; it has weights and a matrix of numbers $q_{ij}$.

\emph{Why this loses nothing.} Look at what $\Phi$ actually uses. The weights $w_i$ enter the Gibbs average. The points $\rho^i$ enter only through the covariances $\mathbb Ez_iz_j=C_z(\rho^i\cdot\rho^j)$, and a centered Gaussian family
is determined in law by its covariances (Lemma~\ref{lem:sg:gram}). So $\Phi(\Gamma)$ is a function of $(w,q)$ with $q_{ij}=\rho^i\cdot\rho^j$. An analogy: a statistic of a point cloud that is computed only from pairwise distances does not care
in which space the cloud sits. Stating the theorem for a general array $q$ is therefore \emph{more general}: the Hilbert-space statement (finitely many atoms) is the special case $q_{ij}=\rho^i\cdot\rho^j$. We did not write that special case
as a separate corollary, since nothing downstream needed it.

\emph{Why we did it.} Mathlib has no convenient ``separable Hilbert space with unit sphere and Gaussian process indexed by it'' package, and every such object would have been a new definition or a heavy typeclass setup
for no mathematical gain.

\subsection*{Decision B: finitely many atoms instead of countably many \emph{(this is a genuine weakening)}}

\emph{What we did.} The index type $\iota$ is a \texttt{Fintype}. The book allows countably many atoms.

\emph{This is weaker than the book's theorem.} Our result does not cover a measure $\Gamma$ with infinitely many atoms (let alone a non-atomic measure). We want to be direct about this: for the statement ``Theorem~1.3 as printed'', the Lean version is a
special case. It is not a different proof of the same statement.

\emph{What the weakening does not touch.} The estimates are \emph{uniform in the number of atoms}: $m$, $F$ and the truncation level do not depend on $|\iota|$, so the theorem holds for all finite Gibbs measures at once with the same constants.
Moreover, every Gibbs measure in the SK model on $N$ spins is a measure on the \emph{finite} set $\Sigma_N=\{-1,+1\}^N$. Hence Eq.~(1.73), the statement that $A_N$ is within $\varepsilon$ of an average of $F$ over the overlap array of
the Gibbs measure $G'_u$, is proved with no gap (Theorem~\ref{thm:sg:overlap-approx}); likewise the a priori bounds on $A_N$ and the Ces\`aro corollary do not use Theorem~1.3 at all.

\emph{Why we chose to stop at finite families.} With finitely many atoms every Gibbs average $\sum_iw_ie^{z_i}$ is a finite sum of integrable random variables, so measurability, integrability and Fubini are routine.
With infinitely many atoms one must additionally show that the series converges almost surely, that it is integrable together with its logarithm, and that the Jensen and truncation estimates survive passing to the infinite sum.
We did not know in advance how much extra work that would be, and the SK application did not need it, so we scoped it out.

\emph{What the missing step would probably be, and where it could matter.} The natural route is to approximate an infinite $\Gamma$ by its restriction to the first $n$ atoms (renormalized) and let $n\to\infty$. For the right-hand side $\langle F(R)\rangle_\Gamma$ this is easy,
because $F$ is bounded: removing a tail of small total weight changes the average by at most $2\|F\|_\infty$ times the tail mass. For $\Phi(\Gamma)$ itself one needs a continuity statement of the same kind for $\mathbb E\log\langle e^{z}\rangle$ under tail removal;
we have not worked this out and do not claim it is routine. It could matter later: the asymptotic Gibbs measure $G$ produced by the Dovbysh--Sudakov theorem is a random measure on a Hilbert ball and is not known to be finitely supported,
so identifying the limit of the right-hand sides $\mathbb E\langle F(R)\rangle$ with $\mathbb E\,\Phi(G)$ is where we expect the countable-atom version of Theorem~1.3 to be needed. That identification is not part of the formalization yet.

\subsection*{Decision C: truncation by moments, not by Gaussian concentration \emph{(a different method, not a weaker result)}}

\emph{What the book does.} To control the effect of clamping the logarithm, the book proves that $\log\langle\cosh z\rangle$ concentrates around its mean with sub-Gaussian tails (Eq.~(1.69)), and uses that to show the clamp is
rarely active.

\emph{What we do.} We only need the \emph{expected} truncation error, and that is bounded directly: $|\log x-\log T_ax|\le e^{-a}(x+x^{-1})$, and $\mathbb E\langle\cdot\rangle$ and $\mathbb E\langle\cdot\rangle^{-1}$ are bounded using
Jensen's inequality for $x\mapsto x^{-1}$ and the exponential moment $\mathbb Ee^{t|g|}\le2e^{vt^2/2}$ of a sub-Gaussian variable (Lemma~\ref{lem:sg:absmoment}). Concentration is a statement about the \emph{distribution} of a random variable and is
stronger than what is needed for a bound on its \emph{mean}. The conclusion (Theorem~\ref{thm:sg:gibbs-truncation}) has the same form as the book's and is uniform in the number of atoms, so this is a shortcut in the sense of a shorter route,
not a weaker theorem.

\subsection*{Decision D: $F$ is continuous on the cone where covariances are positive semidefinite}

In Mathlib, the Gaussian measure $\mathcal N(0,S)$ is defined for every matrix $S$ but equals the Dirac mass at $0$ when $S$ is not positive semidefinite, so $S\mapsto\int g\,d\mathcal N(0,S)$ cannot be continuous on all matrices.
We therefore state continuity on the set of $Q$ for which $(C_z(Q_{ll'}))$ and $(C_y(Q_{ll'}))$ are positive semidefinite. This costs nothing for applications, because every Gram matrix of replicas lies in that set
(for the SK covariances $2\beta^2a_NR$ and $\beta^2a_NR^2$ this is because overlap matrices and their entrywise squares are positive semidefinite). It is a formality of how Gaussians with degenerate covariance are encoded, not a mathematical restriction.

\subsection*{Decision E: general covariance functions and explicit variance bounds}

Our theorem is stated for arbitrary continuous covariance functions $C_z,C_y$ together with variance bounds $v_z,v_y$, and the polynomial degree and truncation level depend on $\varepsilon,v_z,v_y$ only. The book's covariances are the specific ones of the SK model.
This is a generalization: the SK case is recovered with $C_z(t)=2\beta^2a_Nt$, $C_y(t)=\beta^2a_Nt^2$, and the $N$-dependence enters only through the Gaussian integrals, so the same $m$ and polynomials work for every $N$ (Theorem~\ref{thm:sg:overlap-approx}).
What is \emph{not} done is the limit $a_N\to1$ (continuity of $F$ in the covariance function), which is needed to compare the finite-$N$ functionals with the limiting one; see Section~\ref{sec:sg:aN-limit}.

\section{Background: the factor $a_N=N/(N+1)$, uniformity in $N$, and why the limit $a_N\to1$ is not formalized}\label{sec:sg:aN-limit}

This section expands item~3 (``Uniformity'') of Remark~\ref{rem:sg:gram-deviation}. As in Section~\ref{sec:sg:gram-background} it is an explanation for a human reader. It separates three things that are easy to confuse:
what the Lean theorem \emph{does} say about the dependence on $N$, what an application to a limit \emph{would need}, and the (short, as we expect) argument that connects the two, which we have not written out.
Statements about the book's later arguments, and about how hard the missing step is, are our expectation and are marked as such.

\subsection*{Where $a_N$ comes from}

In the cavity decomposition (Section~\ref{sec:sg:ass}) the field that the new spin $\varepsilon$ feels from the old spins is $z_N(\sigma)=(N+1)^{-1/2}\sum_{i\le N}(\zeta_{\mathrm{inl}(i+1)}+\zeta_{\mathrm{inr}(i)})\sigma_i$.
There are $N$ spins $\sigma_i$ but the normalization is $(N+1)^{-1/2}$, because the system has $N+1$ spins in total. Computing the covariance of the field for two configurations gives
\[\mathbb E\,z_N(\sigma^1)z_N(\sigma^2)=\frac{2N}{N+1}\,R(\sigma^1,\sigma^2)=2a_NR(\sigma^1,\sigma^2),\qquad a_N=\frac N{N+1},\]
and in the same way $\mathbb E\,y_N(\sigma^1)y_N(\sigma^2)=a_NR(\sigma^1,\sigma^2)^2$ (Theorem~\ref{thm:sg:cavity-cov}; the factor $\beta$ is absorbed into $z,y$ and gives the $\beta^2$ below). Intuitively: the fields are built from $N$ terms but scaled for $N+1$, so they are slightly
\emph{too small} by the factor $a_N=1-\frac1{N+1}$, which tends to $1$. For $N=100$ the discrepancy is about one percent. It is a finite-size correction that disappears in the limit, but it is there at every finite $N$, and the Lean theorem has to deal with it honestly.

\subsection*{What the Lean theorem says about the dependence on $N$}

Theorem~\ref{thm:sg:overlap-approx} (our formalization of Eq.~(1.73)) fixes $\beta>0$ and $\varepsilon>0$ and produces, \emph{once and for all}, truncation levels $a_z,a_y\ge0$ and polynomials $p_z,p_y$. The only inputs of this choice are $\varepsilon$ and the variance bounds
$v_z=2\beta^2$, $v_y=\beta^2$; these bound the variances for \emph{every} $N$, because $a_N\le1$. Then for \emph{every} $N\ge1$,
\[\Big|A_N-\mathbb E_u\Big[\langle F^z_{N}(R)\rangle'_u-\langle F^y_{N}(R)\rangle'_u\Big]\Big|\le\varepsilon,\]
where $F^z_N,F^y_N$ are the explicit combinations of the Gaussian integrals
$F^z_k(S)=\int\prod_{l\le k}2\cosh(\mathrm{cl}_{a_z}x_l)\,d\mathcal N(0,S)$ and $F^y_k(S)=\int\prod_{l\le k}e^{\mathrm{cl}_{a_y}x_l}\,d\mathcal N(0,S)$, evaluated at the covariance matrices $S=2\beta^2a_N(R_{ll'})$ and $S=\beta^2a_N(R_{ll'}^2)$.
So the number of replicas, the cut-off levels and the polynomial coefficients are the same for all $N$; the sole place where $N$ enters the \emph{function} $F_N$ is the scalar $a_N$ inside the covariance matrix of the Gaussian integral. This is the uniformity that the Lean statement proves.

\subsection*{What the Lean theorem does \emph{not} say, and why that matters}

The function $F_N$ is a different function for each $N$ (it depends on $a_N$). Now recall how the approximation is meant to be used (this is our reading of the book's strategy, not something formalized here). We take a subsequence $N_k$ along which the law of the overlap array
under the Gibbs measure converges weakly, to a limit $\mu$ (Theorem~\ref{thm:sg:asymptotic-overlap}). Weak convergence means: for each \emph{fixed} bounded continuous test function $F$, $\mathbb E\langle F(R)\rangle_{N_k}\to\int F\,d\mu$.
Our approximation gives us $\mathbb E\langle F_{N_k}(R)\rangle_{N_k}$, in which the test function itself changes with $k$. Weak convergence says nothing about such a moving target: a sequence of laws can converge while the integrals of a drifting sequence of test functions do not.
A picture: weak convergence is a statement about a camera that is focused on one object; a function $F_{N}$ that moves with $N$ is an object that keeps leaving the frame.

To repair this we need the test functions to settle down: a single fixed continuous bounded $F_\infty$ (the one built from the limiting covariances $2\beta^2R$ and $\beta^2R^2$, i.e.\ with $a_N$ replaced by $1$) such that $F_N$ is uniformly close to $F_\infty$ on the set of arguments that can occur,
namely overlap matrices. Then
\[\big|\mathbb E\langle F_N(R)\rangle-\mathbb E\langle F_\infty(R)\rangle\big|\le\sup_Q|F_N(Q)-F_\infty(Q)|\xrightarrow[N\to\infty]{}0,\]
because a Gibbs average of a function is at most the supremum of that function, and $\mathbb E\langle F_\infty(R)\rangle_{N_k}\to\int F_\infty\,d\mu$ by weak convergence. That is the step ``passing to the limit $a_N\to1$''.

\subsection*{How we expect the missing step to go (not formalized)}

Two routes seem available. We have not carried either out, so the following is an expectation, not a result.

\begin{enumerate}
\item \emph{At the level of $F$ (compactness).} The relevant arguments $Q$ are $k\times k$ overlap matrices, so they lie in the compact set $\mathcal Q_k=\{Q\in[-1,1]^{k\times k}:Q\text{ positive semidefinite}\}$. For $a\in[0,1]$ the matrix $a\cdot2\beta^2Q$ is positive semidefinite for $Q\in\mathcal Q_k$, and
$(a,Q)\mapsto\int g\,d\mathcal N(0,2\beta^2aQ)$ is continuous on $[0,1]\times\mathcal Q_k$ by the weak continuity of Gaussians in the covariance (Lemma~\ref{lem:sg:gausscont}) composed with the continuous map $(a,Q)\mapsto 2\beta^2aQ$. A continuous function on a compact set is uniformly continuous, hence
$\sup_{Q\in\mathcal Q_k}|F^z_k(a\cdot)-F^z_k(1\cdot)|\to0$ as $a\to1$; the same applies to $y$, where one uses that the entrywise square $(Q_{ll'}^2)$ of a positive semidefinite matrix is positive semidefinite (Schur product theorem). Since only finitely many $k\le\deg p$ occur and the coefficients do not depend on $N$,
$\sup_Q|F_N-F_\infty|\to0$. All ingredients (weak continuity in the covariance, compactness of a cube intersected with a closed cone) are already in the repository or in Mathlib.
\item \emph{At the level of $\Phi$ (comparison).} Alternatively one could compare $\Phi$ for the true fields (covariance $a_N\cdot$) with $\Phi$ for fields of covariance exactly $2\beta^2R$ by Gaussian interpolation, in the style of Section~\ref{sec:sg:comparison}, and bound the difference by a constant times $1-a_N$.
This would let Theorem~\ref{thm:sg:gram13} be applied with the limiting covariance functions directly. It needs a derivative bound that we have not established, so we regard the first route as the more likely one.
\end{enumerate}

\subsection*{Why we did not do it in this milestone}

\begin{itemize}
\item \emph{It is not needed for any statement we proved.} Theorem~\ref{thm:sg:overlap-approx} is true and complete as stated: a bound for each $N$ with $N$-uniform constants. The limit $a_N\to1$ is only needed when one passes to a limit along a subsequence, which is a later step of the book's argument
(the use of the asymptotic Gibbs measure), after the Dovbysh--Sudakov theorem. That later step is not part of the formalization yet (Remark~\ref{rem:sg:asymptotic-deviation}), so the limit has nothing to attach to.
\item \emph{We chose to keep each milestone to a closed set of statements.} The first route above is a self-contained lemma about uniform convergence of Gaussian integrals; we judged it better to formalize it together with the argument that uses it, so that its exact form
(sup over a compact set of overlap matrices, which polynomials, which $k$) is dictated by the application instead of guessed in advance.
\item \emph{It is not an obstruction.} To avoid any misunderstanding: we do not know of a mathematical difficulty here, and we believe it is a routine compactness argument. The honest summary is that it was postponed, not that it was found hard. If the reasoning in route~1 turns out to hide a technicality
(for example in how continuity on the cone interacts with the cube of overlaps), that would only be discovered when the lemma is written.
\end{itemize}

\section{The overlap array and the asymptotic Gibbs measures}\label{sec:sg:asymptotic-gibbs}

Fix $N\ge1$ and the random Gibbs measure $G_N(\sigma)\propto e^{\beta H_N(\sigma)}$ on $\Sigma_N=\{-1,+1\}^N$ (it depends on the disorder $G$). Conditionally on $G$, let
$\sigma^1,\sigma^2,\dots$ be i.i.d.\ replicas with law $G_N$, and let $R=(R_{l,l'})_{l,l'\ge1}$, $R_{l,l'}=N^{-1}\sigma^l\cdot\sigma^{l'}$, be their overlap array. Its
\emph{annealed law} $\Lambda_N=\mathbb E\,\big(\text{law of }R\text{ given }G\big)$ is a probability measure on $\mathbb R^{\mathbb N\times\mathbb N}$ (in Lean:
\texttt{Measure.map R (Measure.bind P (fun G => Measure.infinitePi (fun \_ => Measure.count.tilted (beta * H G))))}, written out in each statement, with the
law $P$ of the disorder). Let $\mathcal K$ be the set of arrays $A\in\mathbb R^{\mathbb N\times\mathbb N}$ which are symmetric, have $A_{ll}=1$ and $|A_{ll'}|\le1$, and all of whose finite
principal submatrices $(A_{ll'})_{l,l'<k}$ are positive semidefinite (``Gram-type arrays''). This section shows that $\Lambda_N$ is concentrated on $\mathcal K$ and
invariant under permutations of the replicas (Eq.~(1.74)), that $\mathcal K$ is compact, and hence that a subsequence of $(\Lambda_N)$ converges weakly to an exchangeable
probability measure on $\mathcal K$. These are exactly the hypotheses of the Dovbysh--Sudakov theorem (Theorem~1.7 of the book), which then represents the limit as the law of the Gram
array of an i.i.d.\ sample from a random measure on the unit ball of a Hilbert space (the \emph{asymptotic Gibbs measure}). That representation theorem is not formalized.

\begin{lemma}[A mixture of i.i.d.\ sequences is exchangeable]\label{lem:sg:mixture}
  \lean{MeasureTheory.Measure.measurable_pi_const, MeasureTheory.Measure.measurable_infinitePi_const,
        MeasureTheory.Measure.infinitePi_map_comp_equiv, MeasureTheory.Measure.measurePreserving_restrict_infinitePi,
        MeasureTheory.Measure.map_bind_eq_self,
        MeasureTheory.Measure.bind_infinitePi_map_comp_equiv}
  \leanok
Let $(\Omega,P)$ be a measure space and $\omega\mapsto\nu_\omega$ a measurable family of probability measures on $X$ (measurable as a map into the space of measures). Then
$\omega\mapsto\nu_\omega^{\otimes I}$ is a measurable family of probability measures on $X^I$ ($I$ finite or infinite), each $\nu_\omega^{\otimes I}$ is invariant under
$\sigma\mapsto\sigma\circ e$ for every bijection $e:I\to I$, and so is the mixture $\int\nu_\omega^{\otimes I}\,P(d\omega)$.
Restricting $\nu^{\otimes\kappa}$ along an injection $f:I\to\kappa$ from a finite set gives $\nu^{\otimes I}$ (the finite-replica marginals).
\end{lemma}

\begin{proof}
  \leanok
For a box $\prod_it_i$ over a finite set of indices, $\nu_\omega^{\otimes I}(\prod_it_i)=\prod_i\nu_\omega(t_i)$ is measurable in $\omega$. Boxes form a $\pi$-system generating the product
$\sigma$-algebra (for infinite $I$: the measurable cylinders, using that the measure of a cylinder is the finite product measure of its base), and the sets $A$ for which $\omega\mapsto\nu_\omega^{\otimes I}(A)$ is measurable form a Dynkin system
(complements via total mass $1$, countable disjoint unions via sums), so by the $\pi$--$\lambda$ theorem the family is measurable. Invariance of $\nu^{\otimes I}$ under a permutation of the coordinates is Mathlib's
reindexing theorem for infinite product measures (both sides agree on cylinders). The mixture $\mu=\int\nu_\omega^{\otimes I}P(d\omega)$ satisfies $\mu(T^{-1}B)=\int\nu_\omega^{\otimes I}(T^{-1}B)P(d\omega)=\int\nu_\omega^{\otimes I}(B)P(d\omega)=\mu(B)$.
\end{proof}

\begin{lemma}[Dot-product Gram matrices and the positive semidefinite cone]\label{lem:sg:gram-psd}
  \lean{Matrix.posSemidef_of_const_mul_dotProduct, Matrix.isClosed_setOf_posSemidef,
        Matrix.PosSemidef.sum_mul_mul_nonneg}
  \leanok
For vectors $v_l\in\mathbb R^\kappa$ and $c\ge0$ the matrix $(c\,v_l\cdot v_{l'})_{l,l'}$ is positive semidefinite (it is $c\,VV^\top$ for the matrix $V$ with rows $v_l$); in particular the overlap matrix
$(N^{-1}\sigma^l\cdot\sigma^{l'})_{l,l'\le k}$ of any $k$ configurations is positive semidefinite. The set of real positive semidefinite matrices is closed in $\mathbb R^{n\times n}$.
\end{lemma}

\begin{proof}
  \leanok
$x^\top VV^\top x=\|V^\top x\|^2\ge0$ and $VV^\top$ is symmetric. A matrix is positive semidefinite iff it is symmetric and $x^\top Mx\ge0$ for all $x$; each condition is closed in $M$.
\end{proof}

\begin{lemma}[The set of Gram-type arrays is compact]\label{lem:sg:gram-compact}
  \lean{SpinGlasses.isCompact_setOf_gram_array}
  \leanok
  \uses{lem:sg:gram-psd}
$\mathcal K$ is a compact subset of $\mathbb R^{\mathbb N\times\mathbb N}$ with the product topology.
\end{lemma}

\begin{proof}
  \leanok
$\mathcal K$ is the intersection of the closed conditions ``$A_{ll'}=A_{l'l}$'', ``$A_{ll}=1$'', ``$|A_{ll'}|\le1$'' and ``the $k\times k$ principal submatrix lies in the closed cone of positive semidefinite matrices'' (the map $A\mapsto(A_{ll'})_{l,l'<k}$ is continuous), and it is contained in
the compact cube $[-1,1]^{\mathbb N\times\mathbb N}$ (Tychonoff).
\end{proof}

\begin{theorem}[The annealed overlap array is a symmetric Gram-type exchangeable array, Panchenko Eq.~(1.74)]\label{thm:sg:annealed-gram}
  \lean{SpinGlasses.annealed_gram_array, SpinGlasses.sk_annealed_gram_array, SpinGlasses.measurable_count_tilted}
  \leanok
  \uses{lem:sg:mixture, lem:sg:gram-psd, lem:sg:gram-compact}
Let $(\Omega,P)$ be a probability space and $\omega\mapsto\nu_\omega$ a measurable family of probability measures on a measurable space $\alpha$. Let $c:\alpha\times\alpha\to\mathbb R$ be measurable,
symmetric, with $c(a,a)=1$, $|c|\le1$ and such that $(c(a_l,a_{l'}))_{l,l'<k}$ is positive semidefinite for all $k$ and $a_1,\dots,a_k$. Let $\Lambda$ be the law of the array $(c(\sigma^l,\sigma^{l'}))_{l,l'}$ when $\omega\sim P$ and,
given $\omega$, $\sigma^1,\sigma^2,\dots$ are i.i.d.\ with law $\nu_\omega$. Then $\Lambda$ is a probability measure, $\Lambda(\mathcal K)=1$, and $\Lambda$ is invariant under $A\mapsto(A_{e(l)e(l')})_{l,l'}$ for
every bijection $e$ of $\mathbb N$ (in particular under finite permutations of the replicas).

In the SK model take $\alpha=\Sigma_N$ (with $N\ge1$), $c(\sigma,\tau)=N^{-1}\sigma\cdot\tau$ and $\nu_\omega=\mathrm{count.tilted}(f(\omega,\cdot))$ for energies $f(\omega,\sigma)$ measurable in $\omega$
(for instance $f=\beta H_N$ or the auxiliary $f=\beta H'_N$ of Section~\ref{sec:sg:ass}).
\end{theorem}

\begin{proof}
  \leanok
Measurability of $\omega\mapsto\nu_\omega^{\otimes\mathbb N}$ and invariance are Lemma~\ref{lem:sg:mixture}; the array is a measurable function of the replicas, so $\Lambda$ is the image of a probability measure, and
$(c(\sigma^{e(l)},\sigma^{e(l')}))$ is the array of the permuted replicas $\sigma\circ e$, whose law is unchanged by Lemma~\ref{lem:sg:mixture}. Every array $(c(\sigma^l,\sigma^{l'}))$ lies in $\mathcal K$ by the hypotheses on $c$, so $\Lambda(\mathcal K)=1$
($\mathcal K$ is closed, hence measurable, by Lemma~\ref{lem:sg:gram-compact}). For the SK overlap: $c$ is symmetric; $c(\sigma,\sigma)=N^{-1}\cdot N=1$ since $\sigma_i^2=1$; $|c|\le1$ since each $|\sigma_i\tau_i|\le1$; and positive semidefiniteness is
Lemma~\ref{lem:sg:gram-psd}. The tilted counting measure on a finite set depends measurably on the energies because its mass of a set $s$ is $\sum_{a\in s}e^{f(\omega,a)}/\sum_be^{f(\omega,b)}$.
\end{proof}

\begin{lemma}[Prokhorov, sequential form; symmetries pass to limits]\label{lem:sg:prokhorov-subseq}
  \lean{MeasureTheory.ProbabilityMeasure.exists_tendsto_subseq_of_measure_eq_one, MeasureTheory.ProbabilityMeasure.map_eq_of_tendsto}
  \leanok
Let $E$ be a separable metrizable space, $K\subseteq E$ compact, and $\mu_n$ probability measures on $E$ with $\mu_n(K)=1$. Then a subsequence converges weakly to a probability measure $\mu$, and $\mu(K)=1$.
If a continuous map $T:E\to E$ satisfies $T_*\mu_n=\mu_n$ for all $n$ then $T_*\mu=\mu$.
\end{lemma}

\begin{proof}
  \leanok
The family $\{\mu_n\}$ is tight (each $\mu_n$ gives mass $0$ to $K^c$), so its closure in the space of probability measures is compact by Prokhorov's theorem; that space is metrizable (L\'evy--Prokhorov), hence the sequence has a convergent subsequence.
The limit gives mass $\ge\limsup\mu_n(K)=1$ to the closed set $K$ (portmanteau). If $\mu_n\to\mu$ then $T_*\mu_n\to T_*\mu$ by continuity of $T$, while $T_*\mu_n=\mu_n\to\mu$; limits are unique.
\end{proof}

\begin{theorem}[Asymptotic overlap laws]\label{thm:sg:asymptotic-overlap}
  \lean{SpinGlasses.exists_tendsto_exchangeable_gram_arrays, SpinGlasses.exists_sk_asymptotic_overlap_law}
  \leanok
  \uses{thm:sg:annealed-gram, lem:sg:prokhorov-subseq, lem:sg:gram-compact}
Let $\Lambda_n$ be probability measures on $\mathbb R^{\mathbb N\times\mathbb N}$ concentrated on $\mathcal K$ and invariant under all permutations of the indices. Then a subsequence converges weakly to a probability measure
$\mu$ which is again concentrated on $\mathcal K$ and invariant under all permutations. In particular, for the SK model let $\Lambda_N$ be the annealed law of the overlap array of the Gibbs measure $G_N=\mathrm{count.tilted}(\beta H_N)$,
the disorder being a standard Gaussian matrix: along a subsequence $N_k$, $\Lambda_{N_k}$ converges weakly (against every bounded continuous function on arrays) to an exchangeable probability measure on $\mathcal K$.
\end{theorem}

\begin{proof}
  \leanok
Apply Lemma~\ref{lem:sg:prokhorov-subseq} with $E=\mathbb R^{\mathbb N\times\mathbb N}$ (separable and metrizable as a countable product), $K=\mathcal K$ (Lemma~\ref{lem:sg:gram-compact}) and, for each bijection $e$ of $\mathbb N$,
the continuous map $T_e(A)=(A_{e(l)e(l')})$. For the SK model the hypotheses hold by Theorem~\ref{thm:sg:annealed-gram} (the SK Hamiltonian is a continuous function of the disorder, so measurable).
\end{proof}

\begin{remark}[Deviations from Panchenko's treatment of the asymptotic Gibbs measures, and why]\label{rem:sg:asymptotic-deviation}
\begin{enumerate}
\item \emph{The Dovbysh--Sudakov theorem is not formalized.} The book obtains the asymptotic Gibbs measure $G$ from the limit array via Theorem~1.7 (proved in Sections~1.4--1.5 from the Aldous--Hoover representation).
We stop at its hypotheses (symmetric, positive semidefinite, exchangeable limit law), so the statement of existence of $G$ on a Hilbert space is not a theorem of this repository; no Lean declaration asserts it.
\item \emph{All permutations, any Gibbs measure.} We prove invariance under every bijection of $\mathbb N$ (not only finite permutations), and Theorem~\ref{thm:sg:annealed-gram} is stated for an arbitrary measurable family of Gibbs measures, so it applies both to $G_N$
and to the auxiliary measure $G'_N$ of the Aizenman--Sims--Starr scheme (Eq.~(1.73)). The book's step identifying the asymptotic overlap laws of $G_N$ and $G'_N$ (via an orthogonal transformation of $G'$) is not formalized.
\item \emph{The diagonal.} The array includes the diagonal $R_{ll}=1$, which holds for $\pm1$ spins; the book's representation concerns $(R_{ll'})_{l\ne l'}$.
\item \emph{Compactness by Prokhorov.} We obtain the convergent subsequence from Prokhorov's theorem in Mathlib and metrizability of the space of probability measures on $\mathbb R^{\mathbb N\times\mathbb N}$, rather than a diagonal argument on finite-dimensional marginals.
\end{enumerate}
\end{remark}

\section{Tools for the Aldous--Hoover representation}\label{sec:sg:aldous-hoover}

Section~1.4 of the book proves the de Finetti and Aldous--Hoover representations of exchangeable sequences and arrays. Three general tools carry all of its probabilistic content,
and they are the only parts of that section formalized so far: the $L^2$ projection equality behind Lemmas~1.5 and~1.6, the Coding Lemma 1.4, and the induction that turns sequential
conditioning into conditional independence (Eqs.~(1.88) and~(1.94)). None of them mentions exchangeability. The statements about exchangeable sequences and arrays themselves
(Eq.~(1.84), Lemmas~1.5 and~1.6, Theorems~1.4--1.6) are \emph{not} formalized in this repository. The Lean files are
\texttt{ForMathlib/MeasureTheory/ConditionalExpectation/L2Projection.lean}, \texttt{SequentialDecoupling.lean} and \texttt{ForMathlib/Probability/Kernel/CodingLemma.lean}
(deviations in Remark~\ref{rem:sg:aldous-hoover-deviation}).

\begin{lemma}[$L^2$ projection equality for nested conditional expectations]\label{lem:sg:l2proj}
  \lean{MeasureTheory.integral_mul_condExp_of_stronglyMeasurable,
        MeasureTheory.integral_condExp_mul_condExp_of_le,
        MeasureTheory.integral_sq_condExp_sub_condExp_of_le,
        MeasureTheory.integral_sq_condExp_le_of_le,
        MeasureTheory.condExp_ae_eq_of_integral_sq_condExp_eq}
  \leanok
Let $\mu$ be a finite measure on $(\Omega,\mathscr B)$, let $\mathscr M_1\subseteq\mathscr M_2\subseteq\mathscr B$ be $\sigma$-algebras and $f\in L^2(\mu)$. Write $f_i=\mathbb E[f\mid\mathscr M_i]$. Then
$\mathbb E[f_1f_2]=\mathbb E[f_1^2]$ and
\[\mathbb E\big[(f_2-f_1)^2\big]=\mathbb E[f_2^2]-\mathbb E[f_1^2].\]
In particular $\mathbb E[f_1^2]\le\mathbb E[f_2^2]$, and if $\mathbb E[f_1^2]=\mathbb E[f_2^2]$ then $f_1=f_2$ almost surely.
\end{lemma}

\begin{proof}
  \leanok
Since $\mathscr M_1\subseteq\mathscr M_2$, the tower property gives $\mathbb E[f_2\mid\mathscr M_1]=f_1$. As $f_1$ is $\mathscr M_1$-measurable and $f_1,f_2\in L^2$, pulling $f_1$ out and using
$\mathbb E\,\mathbb E[\,\cdot\mid\mathscr M_1]=\mathbb E$ yields $\mathbb E[f_1f_2]=\mathbb E[f_1\,\mathbb E[f_2\mid\mathscr M_1]]=\mathbb E[f_1^2]$. Expanding $(f_2-f_1)^2$ gives the displayed identity. If the norms are
equal, the non-negative function $(f_2-f_1)^2$ has integral zero, hence vanishes almost surely.
\end{proof}

\begin{lemma}[Coding Lemma, Lemma~1.4 of the book]\label{lem:sg:coding}
  \lean{ProbabilityTheory.Kernel.exists_measurable_map_compProd_eq,
        ProbabilityTheory.exists_coding_function,
        ProbabilityTheory.identDistrib_coding,
        ProbabilityTheory.exists_coding_identDistrib}
  \leanok
Let $(X,Y)$ be a random pair on a finite measure space $(\Omega,P)$ with $X$ taking values in a measurable space $\Omega_1$ and $Y$ in a standard Borel space $\Omega_2$ (both a.e.\ measurable). Then there is
a measurable $f:\Omega_1\times[0,1]\to\Omega_2$ such that the image of $\mathrm{law}(X)\otimes\mathrm{Leb}$ under $(x,u)\mapsto(x,f(x,u))$ is the law of $(X,Y)$. Consequently, on any probability space
carrying $X'$ with the law of $X$ and a uniform $U$ independent of $X'$ we have $(X,Y)\overset d=(X',f(X',U))$; for instance on $\Omega\times[0,1]$ with the product measure, $(X,Y)\overset d=(X,f(X,u))$.
\end{lemma}

\begin{proof}
  \leanok
Disintegrate the law of $(X,Y)$ as $\mathrm{law}(X)\otimes\kappa$ for a Markov kernel $\kappa$ from $\Omega_1$ to $\Omega_2$ (the regular conditional distribution of $Y$ given $X$, which exists because $\Omega_2$ is standard Borel). A
Markov kernel into a standard Borel space is the image of the uniform law by a jointly measurable map: there is a measurable $f$ with $\mathrm{Leb}\circ f(x,\cdot)^{-1}=\kappa(x,\cdot)$ for all $x$ (Kallenberg, Lemma~4.22, in Mathlib). Evaluating both
sides on a measurable set $S\subseteq\Omega_1\times\Omega_2$ by Fubini gives $\int\mathrm{Leb}\{u:(x,f(x,u))\in S\}\,\mathrm{law}(X)(dx)=\int\kappa(x,S_x)\,\mathrm{law}(X)(dx)$. The transport statement holds because the joint law of $(X',U)$ is the product
$\mathrm{law}(X)\otimes\mathrm{Leb}$ by independence.
\end{proof}

\begin{theorem}[Sequential conditioning gives a product formula, cf.\ Eqs.~(1.88) and~(1.94)]\label{thm:sg:seq-decoupling}
  \lean{MeasureTheory.condExp_mul_eq_mul_condExp_of_condExp_eq,
        MeasureTheory.condExp_prod_eq_prod_condExp,
        MeasureTheory.condExp_prod_comp_eq_prod_condExp}
  \leanok
Let $\mu$ be a finite measure on $(\Omega,\mathscr B)$, $\mathscr M\subseteq\mathscr B$ a $\sigma$-algebra, and $Y_1,\dots,Y_n$ measurable random variables. Let $f_1,\dots,f_n$ be bounded measurable real functions. Suppose that
for each $k$, writing $\mathscr G_k=\mathscr M\vee\sigma(Y_j:j<k)$,
\[\mathbb E[f_k(Y_k)\mid\mathscr G_k]=\mathbb E[f_k(Y_k)\mid\mathscr M]\quad\text{a.s.}\]
Then $\mathbb E\big[\prod_{k\le n}f_k(Y_k)\mid\mathscr M\big]=\prod_{k\le n}\mathbb E[f_k(Y_k)\mid\mathscr M]$ almost surely. More generally the same holds for bounded measurable $Z_k$ and any
$\sigma$-algebras $\mathscr M\subseteq\mathscr G_k\subseteq\mathscr B$ for which $Z_j$ is $\mathscr G_k$-measurable whenever $j<k$.
\end{theorem}

\begin{proof}
  \leanok
Induction on $n$. Put $A=\prod_{j<n}Z_j$, which is bounded and $\mathscr G_n$-measurable. By the tower property, the pull-out property, the hypothesis and the pull-out property again,
\[\mathbb E[AZ_n\mid\mathscr M]=\mathbb E\big[\mathbb E[AZ_n\mid\mathscr G_n]\mid\mathscr M\big]=\mathbb E\big[A\,\mathbb E[Z_n\mid\mathscr G_n]\mid\mathscr M\big]=\mathbb E\big[A\,\mathbb E[Z_n\mid\mathscr M]\mid\mathscr M\big]=\mathbb E[A\mid\mathscr M]\,\mathbb E[Z_n\mid\mathscr M],\]
and the induction hypothesis applied to $Z_1,\dots,Z_{n-1}$ identifies $\mathbb E[A\mid\mathscr M]$.
\end{proof}

\begin{remark}[Deviations from the plan and from Panchenko's text, and what is left]\label{rem:sg:aldous-hoover-deviation}
\begin{enumerate}
\item \emph{Scope.} Only the three general tools above are formalized. De Finetti's theorem, the Aldous--Hoover representations and the Dovbysh--Sudakov theorem are not, and no Lean declaration asserts them.
\item \emph{Norm equality as an integral.} The equality case of Lemma~\ref{lem:sg:l2proj} is stated with $\int(\mathbb E[f\mid\mathscr M_i])^2$ rather than with the $L^2$ norms in the book, because this is the form in which the exchangeability arguments produce it.
\item \emph{Hypothesis only for the functions used.} The book assumes the conditioning identity for all bounded continuous $f$; Theorem~\ref{thm:sg:seq-decoupling} needs it only for the functions that occur in the product, and the filtration $\mathscr G_k$ is arbitrary.
It is the forward half of Mathlib's characterization of conditional independence (\texttt{iCondIndepFun\_iff}), which requires a standard Borel ambient space that is not needed here.
\item \emph{The coding lemma is not proved from scratch.} The book builds the quantile function by hand; here the existence of the kernel and of its uniform representation are Mathlib's \texttt{condDistrib} and \texttt{Kernel.exists\_measurable\_map\_eq\_unitInterval}, and the new content is the assembly into an identity of laws.
\end{enumerate}
\end{remark}
