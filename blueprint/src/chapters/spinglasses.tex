\chapter{The Sherrington--Kirkpatrick model and mixed $p$-spin models}\label{chap:sg}

This chapter formalizes Section~1.1 of Panchenko, \emph{The Sherrington--Kirkpatrick Model} (2013): the random
Hamiltonians of the Sherrington--Kirkpatrick (SK) model and of the pure and mixed $p$-spin models, their
covariance, the free energy and its smooth approximation of the ground-state energy, the monotonicity of the
free energy in the temperature, the zero-temperature limit, the Gibbs measure and the temperature derivative of
the free energy. The existence of the thermodynamic limit of the free energy (the Guerra--Toninelli theorem) is
the hypothesis \texttt{h\_lim} of the zero-temperature results of Section~\ref{sec:sg:thermo}; it is proved, for the
SK model and for the mixed $p$-spin models with only even $p$ and $p=1$, in Section~\ref{sec:sg:gt}. The Lean files are in
\texttt{SpinGlasses/} (modules \texttt{Basic}, \texttt{SKModel}, \texttt{FreeEnergy}, \texttt{ThermodynamicLimit},
\texttt{GibbsMeasure}, \texttt{MixedPSpin}), together with the general tools
\texttt{ForMathlib/Probability/Moments/GaussianPiLinearForm.lean}, \texttt{GaussianSeries.lean},
\texttt{ForMathlib/Probability/Independence/IntegralTsumMul.lean},
\texttt{ForMathlib/Topology/UniformSpace/LimitInterchange.lean} and
\texttt{ForMathlib/MeasureTheory/MeasurableSpace/Units.lean}.

The second part of the chapter (Sections~\ref{sec:sg:stein}--\ref{sec:sg:countable}) formalizes the Gaussian tools of
Section~1.2: Stein's lemma and Gaussian integration by parts for jointly Gaussian vectors, the replica covariance
formula for the Gibbs measure (Panchenko's Lemma~1.1 and Exercise~1.1) and its extension to a countable
configuration space (Lemma~1.2). The Lean files are \texttt{ForMathlib/MeasureTheory/Measure/PiUpdate.lean},
\texttt{ForMathlib/Probability/Moments/GaussianStein.lean}, \texttt{GaussianVector.lean}, \texttt{GaussianExp.lean},
\texttt{ForMathlib/Topology/Algebra/InfiniteSum/PiProd.lean}, \texttt{SpinGlasses/GaussianIBP.lean} and
\texttt{SpinGlasses/GaussianIBPCountable.lean}. The Guerra--Toninelli interpolation (Theorem~1.1) and the Gaussian
concentration inequality (Theorem~1.2) are not part of this chapter yet; the Guerra--Toninelli interpolation
(Theorem~1.1) is Sections~\ref{sec:sg:serieslaw}--\ref{sec:sg:gt}, with the Lean files
\texttt{ForMathlib/Analysis/Subadditive.lean}, \texttt{ForMathlib/Probability/Moments/GaussianSeriesLaw.lean},
\texttt{SpinGlasses/GaussianComparison.lean} and \texttt{SpinGlasses/GuerraToninelli.lean}.

\medskip\noindent\textbf{Notation.}
$\iota$ is a finite index set and $N=|\iota|$ (the number of spins; $\iota=\mathrm{Fin}\,N$ in the SK model, but all
statements hold for any finite type). A \emph{configuration} is $\sigma\in\Sigma_\iota=\{-1,+1\}^\iota$, a function
$\sigma:\iota\to\mathbb Z^\times$ in Lean (here $\mathbb Z^\times=\{\pm1\}$), read as the real vector
$(\sigma_i)_{i\in\iota}$ when sums are formed. The \emph{overlap} of two configurations is
$R(\sigma^1,\sigma^2)=N^{-1}\sum_i\sigma^1_i\sigma^2_i=N^{-1}\,\sigma^1\!\cdot\!\sigma^2$. The disorder
$G=(g_{ij})$ has i.i.d.\ $\mathcal N(0,1)$ entries, the law
\texttt{Measure.pi fun \_ => Measure.pi fun \_ => gaussianReal 0 1}, and the SK Hamiltonian is
$H_N(\sigma)=N^{-1/2}\sum_{i,j}g_{ij}\sigma_i\sigma_j=N^{-1/2}\,\sigma^\top G\sigma$. For an energy
$H:\Sigma\to\mathbb R$ and $\beta>0$ the partition function is $Z(\beta)=\sum_\sigma e^{\beta H(\sigma)}$ and the free
energy is $F_N(\beta)=N^{-1}\mathbb E\log Z_N(\beta)$.

\medskip\noindent\textbf{Which definitions are not introduced.}
Following the repository's convention this chapter introduces \emph{no} new Lean definition. The overlap, the
Hamiltonians, the partition function, the free energy, the covariance function $\xi$, the Gibbs weights and the
Gibbs measure are all written out explicitly in the statements: the overlap is
$N^{-1}\,\sigma\cdot\tau$ (an expression in \texttt{dotProduct}), the partition function is the sum
$\sum_\sigma e^{\beta H(\sigma)}$, the maximum is the repository's \texttt{Function.max}, and the Gibbs measure is
Mathlib's exponentially tilted counting measure \texttt{Measure.count.tilted} with tilt $\beta H$. Hamiltonians appear as an arbitrary function $H$ together with its
defining equation as a hypothesis (written \texttt{hH}).

\section{Configurations and overlap}\label{sec:sg:basic}

\begin{lemma}[Overlap]\label{lem:sg:overlap}
  \lean{SpinGlasses.card_inv_mul_dotProduct_self_eq_one, SpinGlasses.abs_card_inv_mul_dotProduct_le_one,
        SpinGlasses.card_spin_configurations, Matrix.mulVec_dotProduct_mulVec_of_transpose_mul_self}
  \leanok
Let $\iota$ be a nonempty finite type. (1) If $\sigma_i^2=1$ for all $i$ then $R(\sigma,\sigma)=1$.
(2) If $|\sigma_i|\le1$ and $|\tau_i|\le1$ for all $i$ then $|R(\sigma,\tau)|\le1$.
(3) $|\Sigma_\iota|=2^N$. (4) If $A^\top A=1$ for a matrix $A$ then $(Ax)\cdot(Ay)=x\cdot y$; in particular
$R(O\sigma,O\tau)=R(\sigma,\tau)$ for every orthogonal $O$.
\end{lemma}

\begin{proof}
  \leanok
(1) $\sum_i\sigma_i^2=N$. (2) $|\sum_i\sigma_i\tau_i|\le\sum_i|\sigma_i||\tau_i|\le N$ termwise, so no Cauchy--Schwarz
is needed. (3) $|\{\pm1\}^\iota|=2^{|\iota|}$ (\texttt{Fintype.card\_fun}). (4) $(Ax)\cdot(Ay)=x\cdot(A^\top A)y=x\cdot y$.
\end{proof}

\section{The SK Hamiltonian and its covariance}\label{sec:sg:sk}

\begin{lemma}[Covariance of linear forms of a Gaussian matrix]\label{lem:sg:bilinear}
  \lean{ProbabilityTheory.dotProduct_mulVec_of_eq_sum, ProbabilityTheory.memLp_linearForm_gaussianInit,
        ProbabilityTheory.integral_linearForm_gaussianInit, ProbabilityTheory.integral_gaussianMatrix_bilinear_mul_bilinear}
  \leanok
Let $G\in\mathbb R^{\iota\times\kappa}$ have i.i.d.\ $\mathcal N(0,1)$ entries and let $u,x\in\mathbb R^\iota$,
$v,y\in\mathbb R^\kappa$ be deterministic. Then $u^\top Gv=\sum_{i,j}u_iv_jg_{ij}$ is square-integrable and centered, and
\[\mathbb E\big[(u^\top Gv)(x^\top Gy)\big]=(u\cdot x)(v\cdot y).\]
\end{lemma}

\begin{proof}
  \leanok
Write $G$ as the random vector $q\mapsto g_q$ indexed by $q=(i,j)\in\iota\times\kappa$. Its coordinates are
independent standard Gaussians, so the second moment matrix is the identity
(\texttt{integral\_gaussianInit\_entry\_mul\_entry}), and for any random vector $X$ with square-integrable
coordinates $\mathbb E[(X\cdot a)(X\cdot b)]=a^\top\Gamma b$ with $\Gamma=\mathbb E[XX^\top]$
(\texttt{ProbabilityTheory.integral\_dotProduct\_mul\_dotProduct}). With $a_{(i,j)}=u_iv_j$ and
$b_{(i,j)}=x_iy_j$ we get $\sum_{i,j}u_iv_jx_iy_j=(u\cdot x)(v\cdot y)$.
\end{proof}

\begin{theorem}[Covariance of the SK Hamiltonian, Panchenko Eq.~(1.3)]\label{thm:sg:skcov}
  \lean{SpinGlasses.integral_skHamiltonian_mul_skHamiltonian}
  \leanok
For all real vectors $\sigma,\tau\in\mathbb R^\iota$,
$\mathbb E[H_N(\sigma)H_N(\tau)]=N\,R(\sigma,\tau)^2$.
\end{theorem}

\begin{proof}
  \leanok
  \uses{lem:sg:bilinear}
Apply Lemma~\ref{lem:sg:bilinear} with $u=v=\sigma$ and $x=y=\tau$: the covariance of the unnormalized forms is
$(\sigma\cdot\tau)^2$. The normalization $N^{-1/2}$ appears twice, giving
$N^{-1}(\sigma\cdot\tau)^2=N\,(N^{-1}\sigma\cdot\tau)^2$.
\end{proof}

\begin{lemma}[Dean's comfort identity]\label{lem:sg:comfort}
  \lean{SpinGlasses.sum_lt_mul_units_eq_sub}
  \leanok
For a totally ordered index set and $\sigma\in\{\pm1\}^\iota$, with the nested sums over $i<j$,
\[\sum_{i<j}g_{ij}\sigma_i\sigma_j=\sum_{i<j,\ \sigma_i=\sigma_j}g_{ij}-\sum_{i<j,\ \sigma_i\ne\sigma_j}g_{ij}.\]
\end{lemma}

\begin{proof}
  \leanok
For $\sigma_i,\sigma_j\in\{\pm1\}$ the product $\sigma_i\sigma_j$ equals $+1$ if $\sigma_i=\sigma_j$ and $-1$
otherwise; split each term accordingly.
\end{proof}

\section{Free energy and the smooth approximation of the ground-state energy}\label{sec:sg:free}

The estimate of this section is deterministic; it is proved first for a finite set, then integrated, then specialized
to the hypercube, so that it applies to any finite Gibbs system and not only to SK.

\begin{lemma}[Log-sum-exp sandwich]\label{lem:sg:lse}
  \lean{SpinGlasses.max_le_inv_mul_log_sum_exp, SpinGlasses.inv_mul_log_sum_exp_le,
        SpinGlasses.abs_inv_mul_log_sum_exp_sub_max_le}
  \leanok
For a finite nonempty set $\alpha$, $f:\alpha\to\mathbb R$ and $\beta>0$,
\[\max_a f(a)\le\frac1\beta\log\sum_a e^{\beta f(a)}\le\frac{\log|\alpha|}{\beta}+\max_a f(a).\]
\end{lemma}

\begin{proof}
  \leanok
The sum contains the term $e^{\beta\max f}$, and it is at most $|\alpha|e^{\beta\max f}$. Take logarithms and divide
by $\beta>0$.
\end{proof}

\begin{lemma}[Integrability of the random landscape]\label{lem:sg:integrable}
  \lean{SpinGlasses.integrable_max, SpinGlasses.measurable_log_sum_exp, SpinGlasses.integrable_log_sum_exp}
  \leanok
Let $H:\Omega\to\mathbb R^\alpha$ have measurable, integrable coordinates $\omega\mapsto H(\omega)_a$ on a finite
measure space, $\alpha$ finite nonempty and $\beta>0$. Then $\omega\mapsto\max_aH(\omega)_a$ and
$\omega\mapsto\log\sum_ae^{\beta H(\omega)_a}$ are integrable.
\end{lemma}

\begin{proof}
  \leanok
  \uses{lem:sg:lse}
$|\max_aH_a|\le\sum_a|H_a|$, which is integrable. By Lemma~\ref{lem:sg:lse},
$|\log Z|\le\beta|\max H|+\log|\alpha|$, again integrable on a finite measure space; measurability is that of a
finite composition.
\end{proof}

\begin{theorem}[Smooth approximation, Panchenko Eq.~(1.7)]\label{thm:sg:smooth}
  \lean{SpinGlasses.integral_max_le_inv_mul_integral_log_sum_exp,
        SpinGlasses.inv_mul_integral_log_sum_exp_le,
        SpinGlasses.abs_inv_mul_integral_log_sum_exp_sub_integral_max_le,
        SpinGlasses.log_card_spin_configurations, SpinGlasses.smooth_approximation_sandwich,
        SpinGlasses.smooth_approximation_error_bound}
  \leanok
Let $H$ be a random energy on $\Sigma_\iota$ with measurable integrable coordinates under a probability measure, and
$\beta>0$. Then
\[\frac1N\mathbb E\max_\sigma H(\sigma)\le\frac{1}{N\beta}\mathbb E\log Z(\beta)\le
\frac{\log2}{\beta}+\frac1N\mathbb E\max_\sigma H(\sigma),\]
hence $\big|\frac1{N\beta}\mathbb E\log Z(\beta)-\frac1N\mathbb E\max_\sigma H(\sigma)\big|\le\frac{\log2}\beta$.
(For $N=0$ both sides vanish by the convention $0^{-1}=0$.)
\end{theorem}

\begin{proof}
  \leanok
  \uses{lem:sg:lse, lem:sg:integrable}
Integrate Lemma~\ref{lem:sg:lse} (Bochner integrals are monotone, and the integrands are integrable by
Lemma~\ref{lem:sg:integrable}), then use $\log|\Sigma_\iota|=N\log2$ (Lemma~\ref{lem:sg:overlap}(3)) and divide
by $N$.
\end{proof}

\begin{theorem}[Smooth approximation for the SK model]\label{thm:sg:sksmooth}
  \lean{SpinGlasses.sk_measurable_integrable, SpinGlasses.sk_smooth_approximation_sandwich,
        SpinGlasses.sk_smooth_approximation_error_bound}
  \leanok
For the SK Hamiltonian under i.i.d.\ Gaussian disorder, the hypotheses of Theorem~\ref{thm:sg:smooth} hold, so its
conclusion holds with $H=H_N$.
\end{theorem}

\begin{proof}
  \leanok
  \uses{thm:sg:smooth, lem:sg:bilinear}
For fixed $\sigma$, $G\mapsto H_N(\sigma)$ is a finite linear combination of the coordinates $g_{ij}$, hence
measurable, and it is integrable because it is square-integrable (Lemma~\ref{lem:sg:bilinear}).
\end{proof}

\section{Monotonicity in the temperature and the zero-temperature limit}\label{sec:sg:thermo}

\begin{lemma}[Lyapunov monotonicity of the log-partition function]\label{lem:sg:lyapunov}
  \lean{monotoneOn_inv_mul_log_sum_mul_exp_sub_log_sum,
        SpinGlasses.monotoneOn_inv_mul_log_sum_exp_sub_log_card,
        SpinGlasses.monotoneOn_inv_mul_integral_log_sum_exp_sub_log_card,
        SpinGlasses.monotoneOn_inv_mul_free_energy_sub_log_two}
  \leanok
For a finite nonempty set $\alpha$ and $f:\alpha\to\mathbb R$, the function
$\beta\mapsto\beta^{-1}\big(\log\sum_ae^{\beta f(a)}-\log|\alpha|\big)$ is non-decreasing on $(0,\infty)$. The same
holds for $\beta\mapsto\beta^{-1}\big(\mathbb E\log\sum_ae^{\beta H_a}-\log|\alpha|\big)$ for a random energy as in
Lemma~\ref{lem:sg:integrable}, and for a random energy on the hypercube the function
$\beta\mapsto\beta^{-1}\big(F_N(\beta)-\log2\big)$ is non-decreasing on $(0,\infty)$. More generally, for nonnegative weights
$w_a$ with $\sum_aw_a>0$ the function $\beta\mapsto\beta^{-1}\big(\log\sum_aw_ae^{\beta f(a)}-\log\sum_aw_a\big)$ is
non-decreasing on $(0,\infty)$; the uniform statement is the case $w\equiv1$.
\end{lemma}

\begin{proof}
  \leanok
  \uses{lem:sg:integrable}
Let $\mu_0$ be the uniform probability measure on $\alpha$. Then $\beta^{-1}(\log\sum_ae^{\beta f}-\log|\alpha|)=
\log\|e^f\|_{L^\beta(\mu_0)}$, and the claim is that the $L^p$ norm increases with $p$. Directly: for
$0<\beta_1\le\beta_2$ set $p=\beta_2/\beta_1\ge1$ and apply the power-mean inequality
$\big(\sum_a|\alpha|^{-1}z_a\big)^p\le\sum_a|\alpha|^{-1}z_a^p$ to $z_a=e^{\beta_1f(a)}$. Taking logarithms gives
$\frac{\beta_2}{\beta_1}\log\mathrm{mean}(e^{\beta_1f})\le\log\mathrm{mean}(e^{\beta_2f})$; divide by $\beta_2$. The
random version integrates the pointwise inequality, and the hypercube version divides by $N$ and uses
$\log|\Sigma_\iota|=N\log2$.
\end{proof}

\begin{theorem}[Thermodynamic limit and zero-temperature limit, Panchenko Eq.~(1.10)]\label{thm:sg:zerotemp}
  \lean{exists_tendsto_of_dist_le_of_tendsto, SpinGlasses.monotoneOn_inv_mul_limit_sub_log_two,
        SpinGlasses.exists_tendsto_ground_state_energy, SpinGlasses.sk_monotoneOn_inv_mul_limit_sub_log_two,
        SpinGlasses.sk_exists_tendsto_ground_state_energy}
  \leanok
Consider a family of hypercube models indexed by $k\to\infty$ (for SK, $k=N$ and $\iota=\mathrm{Fin}\,N$), with
measurable, integrable coordinates, and suppose that for every $\beta>0$ the free energy converges,
$F_k(\beta)\to F(\beta)$. Then:
(1) $\beta\mapsto\beta^{-1}(F(\beta)-\log2)$ is non-decreasing on $(0,\infty)$;
(2) the normalized ground-state energies $E_k=|\iota_k|^{-1}\mathbb E\max_\sigma H_k(\sigma)$ converge to some $E_0$, and
$\beta^{-1}F(\beta)\to E_0$ as $\beta\to\infty$.
\end{theorem}

\begin{proof}
  \leanok
  \uses{lem:sg:lyapunov, thm:sg:smooth}
(1) A pointwise limit of monotone functions is monotone (Mathlib's
\texttt{monotoneOn\_of\_frequently\_monotoneOn\_of\_tendsto}) applied to Lemma~\ref{lem:sg:lyapunov}.
(2) By Theorem~\ref{thm:sg:smooth}, $|\beta^{-1}F_k(\beta)-E_k|\le\varepsilon(\beta):=\log2/\beta$ uniformly in $k$. The
limit $g(\beta)=\lim_kF_k(\beta)/\beta$ then satisfies $|g(\beta)-g(\beta')|\le\varepsilon(\beta)+\varepsilon(\beta')$ (let
$k\to\infty$ in the triangle inequality), so $g$ is Cauchy as $\beta\to\infty$ and has a limit $E_0$; the uniform
approximation then makes $E_k$ converge to the same $E_0$ (interchange of limits, in the abstract form of
\texttt{exists\_tendsto\_of\_dist\_le\_of\_tendsto}). In particular, unlike the statement in Panchenko, the
convergence of the ground-state energy is a conclusion and not an assumption.
\end{proof}

\section{The Gibbs measure}\label{sec:sg:gibbs}

For $\beta\in\mathbb R$ and $H:\alpha\to\mathbb R$ on a finite set, the \emph{Gibbs measure}
$G_\beta(a)=e^{\beta H(a)}/Z(\beta)$ is the exponential tilt of the counting measure by $\beta H$; in Lean it is
\texttt{Measure.count.tilted} with tilt $\beta H$, and it is a probability measure whenever $\alpha$ is nonempty. We write
$\langle g\rangle_\beta=\int g\,dG_\beta$.

\begin{lemma}[Gibbs averages]\label{lem:sg:gibbsavg}
  \lean{SpinGlasses.integral_count_eq_sum, SpinGlasses.mgf_count, SpinGlasses.integral_count_tilted,
        SpinGlasses.real_count_tilted_singleton, SpinGlasses.norm_integral_count_tilted_le}
  \leanok
(1) $Z(\beta)=\sum_ae^{\beta H(a)}$ is the moment-generating function of $H$ for the counting measure.
(2) $\langle g\rangle_\beta=\sum_a\frac{e^{\beta H(a)}}{Z(\beta)}\,g(a)$ and $G_\beta(\{a\})=e^{\beta H(a)}/Z(\beta)$.
(3) $|\langle g\rangle_\beta|\le\sum_a|g(a)|$.
\end{lemma}

\begin{proof}
  \leanok
(1) The integral against counting measure on a finite set is the sum. (2) Mathlib's
\texttt{integral\_tilted\_mul\_eq\_mgf}, and the indicator of $\{a\}$ for the mass of the singleton.
(3) $G_\beta$ is a probability measure and $|g|\le\sum_a|g(a)|$ pointwise.
\end{proof}

\begin{theorem}[Temperature derivative of the free energy]\label{thm:sg:deriv}
  \lean{SpinGlasses.hasDerivAt_log_sum_exp, SpinGlasses.hasDerivAt_integral_log_sum_exp,
        SpinGlasses.sk_hasDerivAt_free_energy}
  \leanok
(1) For every $H:\alpha\to\mathbb R$, $\frac{d}{d\beta}\log Z(\beta)=\langle H\rangle_\beta$.
(2) For a random energy with measurable, integrable coordinates and $\beta>0$,
$\frac{d}{d\beta}\mathbb E\log Z(\beta)=\mathbb E\langle H\rangle_\beta$.
(3) For the SK model, $\frac d{d\beta}F_N(\beta)=N^{-1}\mathbb E\langle H_N\rangle_\beta$ for $\beta>0$.
\end{theorem}

\begin{proof}
  \leanok
  \uses{lem:sg:gibbsavg, lem:sg:integrable}
(1) $\log Z=\mathrm{cgf}_H$, the cumulant-generating function of $H$ for the counting measure; every exponential moment
is finite on a finite set, so Mathlib's \texttt{integral\_tilted\_mul\_self} gives
$\mathrm{cgf}_H'(\beta)=\int H\,d(\mu.\mathrm{tilted}(\beta H))$. (2) Differentiate under the integral sign
(dominated differentiation): the pointwise derivative is $\langle H(\omega)\rangle_\beta$ by (1), and
$|\langle H(\omega)\rangle_\beta|\le\sum_a|H(\omega)_a|$ uniformly in $\beta$ by Lemma~\ref{lem:sg:gibbsavg}(3), which
is integrable by assumption. (3) is (2) with the SK landscape (Theorem~\ref{thm:sg:sksmooth}), multiplied by $N^{-1}$.
\end{proof}

\begin{theorem}[Zero-temperature concentration]\label{thm:sg:concentration}
  \lean{SpinGlasses.tendsto_real_count_tilted_singleton_of_lt, SpinGlasses.tendsto_integral_count_tilted_max}
  \leanok
Let $\alpha$ be finite nonempty and $H:\alpha\to\mathbb R$. (1) If $H(a)<\max H$ then $G_\beta(\{a\})\to0$ as
$\beta\to\infty$; indeed $G_\beta(\{a\})\le e^{-\beta(\max H-H(a))}$. (2) $\langle H\rangle_\beta\to\max H$.
\end{theorem}

\begin{proof}
  \leanok
  \uses{lem:sg:gibbsavg}
(1) $Z(\beta)\ge e^{\beta\max H}$ since it contains the term of a maximizer, hence
$e^{\beta H(a)}/Z(\beta)\le e^{-\beta(\max H-H(a))}\to0$. (2) $\max H-\langle H\rangle_\beta=\sum_aG_\beta(\{a\})(\max H-H(a))$
because the weights sum to one; every term with $H(a)<\max H$ tends to $0$ by (1) and the others vanish identically.
\end{proof}

\begin{remark}
The convergence of the Gibbs weights of the ground states to $1/|\mathrm{argmax}\,H|$ is not formalized, since it
is not used anywhere in this chapter.
\end{remark}

\section{Pure and mixed $p$-spin models}\label{sec:sg:pspin}

For $p\ge1$ the pure $p$-spin Hamiltonian is
\[H_{N,p}(\sigma)=N^{-(p-1)/2}\sum_{i_1,\dots,i_p\in\iota}g_{i_1\dots i_p}\sigma_{i_1}\cdots\sigma_{i_p},\]
with i.i.d.\ standard Gaussian disorder $g:(\mathrm{Fin}\,p\to\iota)\to\mathbb R$ (the case $p=2$ is the SK
Hamiltonian). The mixed $p$-spin Hamiltonian is $H_N(\sigma)=\sum_{p\ge1}\beta_pH_{N,p}(\sigma)$ with independent
disorders $g_p$, i.e.\ $g\sim\bigotimes_p\mathcal N(0,1)^{\iota^p}$ (\texttt{Measure.infinitePi}), and the covariance
function is $\xi(x)=\sum_{p\ge1}\beta_p^2x^p$. In Lean the condition $\beta_0=0$ replaces ``the sum starts at $p=1$''.

\begin{lemma}[Gaussian linear forms and series]\label{lem:sg:gaussseries}
  \lean{ProbabilityTheory.integral_dotProduct_mul_dotProduct_pi_gaussianReal,
        ProbabilityTheory.integral_dotProduct_pi_gaussianReal,
        ProbabilityTheory.memLp_dotProduct_pi_gaussianReal,
        ProbabilityTheory.integral_tsum_mul_tsum_of_indepFun,
        ProbabilityTheory.integrable_tsum_of_tsum_eLpNorm_ne_top,
        ProbabilityTheory.eLpNorm_two_le_ofReal_of_integral_sq_le,
        ProbabilityTheory.integral_tsum_dotProduct_mul_tsum_dotProduct,
        ProbabilityTheory.measurable_tsum_dotProduct, ProbabilityTheory.integrable_tsum_dotProduct}
  \leanok
(1) For $g$ with i.i.d.\ $\mathcal N(0,1)$ coordinates indexed by a finite set $\kappa$ and $a,b\in\mathbb R^\kappa$, the
linear forms $a\cdot g$ are square-integrable, centered, and $\mathbb E[(a\cdot g)(b\cdot g)]=a\cdot b$.
(2) Let $X_p,Y_p$ ($p$ in a countable set) be square-integrable with $\sum_p\|X_p\|_{L^2}<\infty$,
$\sum_p\|Y_p\|_{L^2}<\infty$, each $X_p$ centered and independent of $Y_q$ for $p\ne q$. Then
$\sum_pX_p$ is integrable and $\mathbb E[(\sum_pX_p)(\sum_qY_q)]=\sum_p\mathbb E[X_pY_p]$.
(3) Let $g_p$ ($p$ countable) be independent with i.i.d.\ $\mathcal N(0,1)$ coordinates and let $a_p,b_p$ be coefficient
vectors with $\sum_p\|a_p\|_2<\infty$ and $\sum_p\|b_p\|_2<\infty$. Then $\sum_pa_p\cdot g_p$ is measurable and integrable and
$\mathbb E[(\sum_pa_p\cdot g_p)(\sum_pb_p\cdot g_p)]=\sum_pa_p\cdot b_p$.
\end{lemma}

\begin{proof}
  \leanok
(1) The law of $g$ is the standard Gaussian of $\mathbb R^\kappa$, whose covariance bilinear form is the inner product,
and $\mathbb E g=0$. (2) The series converge absolutely almost surely, since $\sum_p\|X_p\|_{L^1}\le\sum_p\|X_p\|_{L^2}$
(Mathlib's \texttt{summable\_norm\_of\_tsum\_eLpNorm\_ne\_top}), so $(\sum_pX_p)(\sum_qY_q)=\sum_{(p,q)}X_pY_q$ almost
everywhere. By H\"older, $\mathbb E|X_pY_q|\le\|X_p\|_{L^2}\|Y_q\|_{L^2}$, and $\sum_{p,q}\|X_p\|_{L^2}\|Y_q\|_{L^2}<\infty$, so the
double series may be integrated termwise (\texttt{integral\_tsum}). For $p\ne q$, independence and centering give
$\mathbb E[X_pY_q]=\mathbb E[X_p]\mathbb E[Y_q]=0$, which leaves the diagonal. (3) Apply (2) to $X_p=a_p\cdot g_p$ and
$Y_p=b_p\cdot g_p$: they are independent for $p\ne q$ because they are functions of different coordinates of the
infinite product, $\|X_p\|_{L^2}=\|a_p\|_2$ by (1), and $\mathbb E[X_pY_p]=a_p\cdot b_p$ by (1).
\end{proof}

\begin{theorem}[Covariance of the pure $p$-spin Hamiltonian, Panchenko Eq.~(1.13)]\label{thm:sg:purecov}
  \lean{SpinGlasses.sum_prod_mul_prod_eq_pow, SpinGlasses.pureCoeff_dotProduct_pureCoeff,
        SpinGlasses.integral_purePSpin_mul_purePSpin}
  \leanok
For $p\ge1$, a nonempty finite $\iota$ and any $\sigma,\tau\in\mathbb R^\iota$,
$\mathbb E[H_{N,p}(\sigma)H_{N,p}(\tau)]=N\,R(\sigma,\tau)^p$.
\end{theorem}

\begin{proof}
  \leanok
  \uses{lem:sg:gaussseries}
$H_{N,p}(\sigma)=u(\sigma)\cdot g$ with $u(\sigma)_{i_1\dots i_p}=N^{-(p-1)/2}\prod_k\sigma_{i_k}$. By
Lemma~\ref{lem:sg:gaussseries}(1) the covariance is $u(\sigma)\cdot u(\tau)=N^{-(p-1)}\sum_{i_1\dots i_p}\prod_k\sigma_{i_k}\tau_{i_k}
=N^{-(p-1)}(\sigma\cdot\tau)^p$, since the sum over multi-indices of a product factorizes
(\texttt{Fintype.sum\_pow}). This equals $N(N^{-1}\sigma\cdot\tau)^p$ for $p\ge1$.
\end{proof}

\begin{lemma}[Summability]\label{lem:sg:summable}
  \lean{SpinGlasses.summable_abs_of_summable_two_pow_mul_sq, SpinGlasses.summable_sq_of_summable_abs,
        SpinGlasses.summable_sq_mul_pow_of_summable_abs}
  \leanok
(1) If $\sum_p2^p\beta_p^2<\infty$ (Panchenko's condition) then $\sum_p|\beta_p|<\infty$.
(2) If $\sum_p|\beta_p|<\infty$ then $\sum_p\beta_p^2<\infty$ and $\xi(x)=\sum_p\beta_p^2x^p$ converges absolutely for
$|x|\le1$.
\end{lemma}

\begin{proof}
  \leanok
(1) $2|\beta_p|\le2^p\beta_p^2+2^{-p}$, since $(2^p|\beta_p|-1)^2\ge0$, and $\sum_p2^{-p}<\infty$. (2) $\beta_p^2\le
(\sum_q|\beta_q|)\,|\beta_p|$ and $|\beta_p^2x^p|\le\beta_p^2$.
\end{proof}

\begin{theorem}[Covariance of the mixed $p$-spin Hamiltonian, Panchenko Eq.~(1.14)]\label{thm:sg:mixedcov}
  \lean{SpinGlasses.mixedPSpin_measurable, SpinGlasses.mixedPSpin_integrable,
        SpinGlasses.integral_mixedPSpin_mul_mixedPSpin}
  \leanok
Let $\iota$ be nonempty and finite, $\beta_0=0$, $\sum_p|\beta_p|<\infty$ and $\sigma,\tau\in[-1,1]^\iota$. Then the
series $H_N(\sigma)=\sum_p\beta_pH_{N,p}(\sigma)$ is measurable and integrable in the disorder, and
\[\mathbb E[H_N(\sigma)H_N(\tau)]=N\,\xi\big(R(\sigma,\tau)\big).\]
\end{theorem}

\begin{proof}
  \leanok
  \uses{lem:sg:gaussseries, thm:sg:purecov, lem:sg:summable}
$H_N(\sigma)=\sum_pa_p(\sigma)\cdot g_p$ with $a_p(\sigma)=\beta_pu_p(\sigma)$. By Theorem~\ref{thm:sg:purecov},
$\|a_p(\sigma)\|_2^2=\beta_p^2NR(\sigma,\sigma)^p\le\beta_p^2N$ because $0\le R(\sigma,\sigma)\le1$ for
$\sigma\in[-1,1]^\iota$; so $\|a_p(\sigma)\|_2\le|\beta_p|\sqrt N$ is summable in $p$. Lemma~\ref{lem:sg:gaussseries}(3) then gives
measurability, integrability and $\mathbb E[H_N(\sigma)H_N(\tau)]=\sum_pa_p(\sigma)\cdot a_p(\tau)
=\sum_p\beta_p^2NR(\sigma,\tau)^p$ (for $p=0$ both sides vanish because $\beta_0=0$).
\end{proof}

\begin{theorem}[Free energy of the mixed $p$-spin model]\label{thm:sg:mixedfree}
  \lean{SpinGlasses.mixedPSpin_smooth_approximation_error_bound,

\begin{proof}

\end{proof}

\section{Stein's lemma and Gaussian integration by parts}\label{sec:sg:stein}

\begin{theorem}[Stein's lemma, Panchenko Eq.~(1.19)]\label{thm:sg:stein1d}
  \lean{ProbabilityTheory.integrable_of_integrable_id_mul, ProbabilityTheory.hasDerivAt_gaussianPDFReal_zero,
        ProbabilityTheory.integrable_gaussianReal_iff_smul, ProbabilityTheory.integrable_pow_gaussianReal,
        ProbabilityTheory.integral_id_mul_gaussianReal_eq}
  \leanok
Let $v\ge0$, let $g\sim\mathcal N(0,v)$, and let $F:\mathbb R\to\mathbb R$ be differentiable with derivative $F'$ such
that $gF(g)$ and $F'(g)$ are integrable. Then $\mathbb E[g\,F(g)]=v\,\mathbb E[F'(g)]$.
\end{theorem}

\begin{proof}
  \leanok
For $v=0$ both sides vanish ($g=0$). Let $v>0$ and let $p$ be the density of $g$, so $p'(x)=-(x/v)p(x)$. Since $F$ is
continuous and $|F(x)|\le|xF(x)|$ for $|x|\ge1$, integrability of $xF(x)$ against $p$ gives integrability of $F$ against
$p$. Mathlib's integration by parts on $\mathbb R$ (all three products $Fp'$, $F'p$, $Fp$ integrable) gives
$\int Fp'=-\int F'p$, that is $-v^{-1}\int xF(x)p(x)\,dx=-\int F'p$. As a check, the case $F(x)=x^n$ recovers the moment recursion
$\mathbb E[g^{n+1}]=vn\,\mathbb E[g^{n-1}]$ of the Wick calculus (\texttt{Renormalization.integral\_mul\_pow\_gaussianReal}, which is now
a corollary).
\end{proof}

\begin{lemma}[Resampling one coordinate]\label{lem:sg:resample}
  \lean{MeasureTheory.measurePreserving_update_pi, MeasureTheory.integrable_update_pi,
        MeasureTheory.integral_pi_eq_integral_update}
  \leanok
Let $\pi=\bigotimes_{i}\mu_i$ be a finite product of probability measures. Replacing the $j$-th coordinate of
$x\sim\pi$ by an independent $t\sim\mu_j$ preserves $\pi$: the map $(x,t)\mapsto x[j:=t]$ pushes $\pi\otimes\mu_j$ to $\pi$.
Hence for integrable $f$, $\int f\,d\pi=\int\!\!\int f(x[j:=t])\,d\mu_j(t)\,d\pi(x)$.
\end{lemma}

\begin{proof}
  \leanok
Both measures agree on boxes $\prod_i s_i$: the preimage of the box is $\prod_i s_i'\times s_j$ with $s_i'=s_i$ for $i\ne j$
and $s_j'=\mathbb R$, of measure $\prod_{i\ne j}\mu_i(s_i)\,\mu_j(s_j)$. The integral formula is the change of variables for the
pushforward followed by Fubini.
\end{proof}

\begin{theorem}[Coordinatewise Stein identity]\label{thm:sg:steinpi}
  \lean{ProbabilityTheory.integral_eval_mul_pi_gaussianReal_eq}
  \leanok
  \uses{thm:sg:stein1d, lem:sg:resample}
Let $\xi$ have independent coordinates $\xi_i\sim\mathcal N(0,v_i)$ and let $G$ have partial derivatives $G_i'$ in the sense that
$t\mapsto G(x[i:=t])$ is differentiable with derivative $G_i'(x)$ at $t=x_i$. If $\xi_jG(\xi)$ and $G_j'(\xi)$ are integrable then
$\mathbb E[\xi_jG(\xi)]=v_j\,\mathbb E[G_j'(\xi)]$.
\end{theorem}

\begin{proof}
  \leanok
Resample the $j$-th coordinate (Lemma~\ref{lem:sg:resample}). By Fubini the slices $t\mapsto G(x[j:=t])$ and its derivative are
integrable against $\mathcal N(0,v_j)$ for almost every $x$, and Theorem~\ref{thm:sg:stein1d} applies to each slice.
\end{proof}

\begin{theorem}[Multivariate Gaussian integration by parts, Panchenko Eq.~(1.25)]\label{thm:sg:steinvec}
  \lean{ProbabilityTheory.sum_abs_mulVec_le, ProbabilityTheory.hasDerivAt_mulVec_update,
        ProbabilityTheory.integrable_exp_mul_abs_gaussianReal, ProbabilityTheory.integrable_exp_sum_mul_abs_pi,
        ProbabilityTheory.integrable_comp_mulVec_of_exp_growth,
        ProbabilityTheory.integrable_eval_mul_comp_mulVec_of_exp_growth,
        ProbabilityTheory.integral_mulVec_mul_comp_mulVec_eq_sum,
        ProbabilityTheory.posSemidef_covariance_matrix, ProbabilityTheory.exists_matrix_map_eq_pi_gaussianReal,
        ProbabilityTheory.IsGaussianProcess.hasGaussianLaw_pi, ProbabilityTheory.abs_comp_inr_le_of_exp_growth,
        ProbabilityTheory.integral_mul_comp_eq_sum_integral_mul,
        ProbabilityTheory.integral_mul_comp_eq_sum_integral_mul_of_sum,
        ProbabilityTheory.abs_prod_le_exp_sum, ProbabilityTheory.fderiv_prod_apply_pi_single,
        ProbabilityTheory.integral_mul_prod_eq_sum_integral_prod_erase}
  \leanok
  \uses{thm:sg:steinpi}
Let $X=(X_k)_{k\in\kappa}$, $\kappa$ finite, be a centered jointly Gaussian vector (Mathlib's \texttt{HasGaussianLaw}; a
\texttt{IsGaussianProcess} on a finite index set is one), and let $F:\mathbb R^\kappa\to\mathbb R$ be differentiable with
$|F(y)|\le C\,e^{c\|y\|_1}$ and $|\partial_lF(y)|\le D_l\,e^{d_l\|y\|_1}$. Then
\[\mathbb E[X_i\,F(X)]=\sum_{l\in\kappa}\mathbb E[X_iX_l]\;\mathbb E[\partial_lF(X)].\]
More generally the same holds with $X_i$ replaced by a variable $W_i$ from a centered Gaussian family $W$ that is jointly
Gaussian with $X$, with $\mathbb E[W_iX_l]$ as the covariance. For the monomial $F(y)=\prod_iy_i$ (whose value and
partial derivatives are bounded by $e^{\|y\|_1}$) this is the multilinear identity
$\mathbb E[W\prod_iY_i]=\sum_j\mathbb E[WY_j]\,\mathbb E[\prod_{i\ne j}Y_i]$, which replaces the earlier
Isserlis-based proof in the Renormalization chapter.
\end{theorem}

\begin{proof}
  \leanok
The covariance matrix $S_{ij}=\mathrm{Cov}(X_i,X_j)$ is positive semidefinite (it is the covariance form of the Gaussian
vector), so $X$ has the law of $A\xi$ with $\xi$ i.i.d.\ $\mathcal N(0,1)$ and $AA^\top=S$ ($A=\sqrt S$; this combines Mathlib's
\texttt{multivariateGaussian} with the fact that a Gaussian measure is determined by mean and covariance). For $G(\xi)=F(A\xi)$
the chain rule gives $\partial_jG(\xi)=\sum_lA_{lj}\partial_lF(A\xi)$, and Theorem~\ref{thm:sg:steinpi} yields
$\mathbb E[(A\xi)_iF(A\xi)]=\sum_jA_{ij}\sum_lA_{lj}\mathbb E[\partial_lF(A\xi)]=\sum_lS_{il}\mathbb E[\partial_lF(A\xi)]$.
The integrability hypotheses of Theorem~\ref{thm:sg:steinpi} follow from the growth bound:
$|\xi_jF(A\xi)|\le|\xi_j|\,Ce^{|c|\sum_ka_k|\xi_k|}$ with $a_k=\sum_l|A_{lk}|$, which is integrable as a product of the
one-dimensional integrable functions $e^{b|t|}$ and $|t|e^{b|t|}$ against $\mathcal N(0,1)$. For the extension to an auxiliary
Gaussian family apply the first part to the concatenated vector and the function $F\circ\mathrm{pr}_2$, whose partial derivatives in
the $W$-directions vanish.
\end{proof}

\section{Integration by parts for the Gibbs measure}\label{sec:sg:ibp}

Throughout, $\alpha$ is a finite configuration space, $(x,y)=(x(a),y(a))_{a\in\alpha}$ is a centered jointly Gaussian field with
$C(a,b)=\mathbb E[x(a)y(b)]$, $h:\alpha\to\mathbb R$ is a deterministic bias (for example $\log G$ for a reference weight, or an
external field), and $\langle\cdot\rangle$ is the Gibbs measure \texttt{Measure.count.tilted} of the energy $y+h$. For a finite index
set $R$ of replicas, replicas are the product $\bigotimes_{r\in R}\langle\cdot\rangle$, \texttt{Measure.pi}; Panchenko's $\sigma^1,\dots,\sigma^n$ is $R=\{1,\dots,n\}$,
and $\Phi$ is an arbitrary function of the $|R|$ replicas.

\begin{lemma}[Calculus of replica averages]\label{lem:sg:replicacalc}
  \lean{SpinGlasses.integral_pi_count_tilted, SpinGlasses.differentiable_gibbsWeight,
        SpinGlasses.differentiable_replicaAverage, SpinGlasses.hasDerivAt_prod_gibbsWeight,
        SpinGlasses.fderiv_integral_pi_count_tilted, SpinGlasses.sum_sum_mul_integral_eq,
        SpinGlasses.abs_integral_pi_count_tilted_le, SpinGlasses.abs_integral_pi_count_tilted_mul_eval_le,
        SpinGlasses.differentiable_integral_pi_count_tilted_add, SpinGlasses.fderiv_integral_pi_count_tilted_add}
  \leanok
For $f:\alpha\to\mathbb R$ write $w_f(a)=e^{f(a)}/\sum_be^{f(b)}$. (1) The $R$-replica average of $\Psi$ is the finite sum
$\sum_{\sigma\in\alpha^R}\prod_rw_f(\sigma_r)\Psi(\sigma)$, a smooth function of $f$. (2) Its derivative in the direction $e_\rho$ is
$\big\langle(\sum_r\mathbf 1[\sigma_r=\rho]-|R|\,w_f(\rho))\Psi\big\rangle_f$. (3) $|\langle\Psi\rangle|\le\sup|\Psi|$ and
$|\langle\Phi\,\chi(\sigma_{r_0})\rangle|\le\sup|\Phi|\,\langle|\chi|\rangle$.
\end{lemma}

\begin{proof}
  \leanok
(1) is the product formula for the product measure of singletons (\texttt{Measure.pi\_singleton}) together with the formula for the
mass of a singleton of the tilted counting measure. For (2) one differentiates along the line $f+te_\rho$: the product of $n$
weights is $\exp(\sum_r(f(\sigma_r)+t\delta_{\sigma_r\rho}))/Z(t)^n$ with $Z'(0)=e^{f(\rho)}$, and the quotient rule gives
$(\prod w)\,(\sum_r\delta_{\sigma_r\rho}-n\,w(\rho))$. (3) is the triangle inequality and the identity
$\int|\chi(\sigma_{r_0})|\,d\pi=\int|\chi|\,d\langle\cdot\rangle$.
\end{proof}

\begin{theorem}[Replica covariance formula, Panchenko Exercise~1.1 and Eq.~(1.29)]\label{thm:sg:replica}
  \lean{SpinGlasses.integral_gibbs_replica_average_gaussian, SpinGlasses.integral_pi_option_count_tilted,
        SpinGlasses.integral_gibbs_replica_average_gaussian_option}
  \leanok
  \uses{thm:sg:steinvec, lem:sg:replicacalc}
With the notation above, for every $r_0\in R$ and every function $\Phi$ of the $|R|$ replicas,
\[\mathbb E\big\langle\Phi\,x(\sigma^{r_0})\big\rangle
 =\mathbb E\Big\langle\Phi\Big(\sum_{r\in R}C(\sigma^{r_0},\sigma^r)-|R|\,C(\sigma^{r_0},\sigma^{n+1})\Big)\Big\rangle,\]
where $\sigma^{n+1}$ is an additional independent replica (written as an integral of $C(\sigma^{r_0},\cdot)$ against the Gibbs measure,
and, in the second form, as the replica indexed by \texttt{none} in the product over \texttt{Option R}).
\end{theorem}

\begin{proof}
  \leanok
Write $x(\sigma^{r_0})=\sum_ax(a)\mathbf 1[\sigma^{r_0}=a]$, so the left-hand side is $\sum_a\mathbb E[x(a)F_a(y)]$ with
$F_a(y)=\langle\mathbf 1[\sigma^{r_0}=a]\Phi\rangle_{y+h}$, a smooth bounded function with bounded partial derivatives
(Lemma~\ref{lem:sg:replicacalc}). By Theorem~\ref{thm:sg:steinvec}, $\mathbb E[x(a)F_a(y)]=\sum_bC(a,b)\,\mathbb E[\partial_bF_a(y)]$, and
$\partial_bF_a=\langle(\sum_r\mathbf 1[\sigma^r=b]-|R|\,w(b))\mathbf 1[\sigma^{r_0}=a]\Phi\rangle$. Summing over $a$ and $b$ gives
$\sum_r C(\sigma^{r_0},\sigma^r)-|R|\sum_bC(\sigma^{r_0},b)w(b)$ inside the Gibbs average, and $\sum_bC(\sigma^{r_0},b)w(b)$ is
the average of $C(\sigma^{r_0},\cdot)$ against the Gibbs measure.
\end{proof}

\begin{theorem}[Gaussian integration by parts for the Gibbs measure, Panchenko Lemma~1.1, Eq.~(1.28)]\label{thm:sg:lemma11}
  \lean{SpinGlasses.integral_gibbs_average_gaussian}
  \leanok
  \uses{thm:sg:replica}
$\mathbb E\langle x(\sigma)\rangle=\mathbb E\langle C(\sigma^1,\sigma^1)-C(\sigma^1,\sigma^2)\rangle$, where $\langle\cdot\rangle$ on the right
is the product of two replicas.
\end{theorem}

\begin{proof}
  \leanok
This is Theorem~\ref{thm:sg:replica} with one replica ($R=\mathrm{Unit}$) and $\Phi\equiv1$.
\end{proof}

\section{A countable configuration space}\label{sec:sg:countable}

\begin{lemma}[Analytic tools for series over replicas]\label{lem:sg:tannery}
  \lean{summable_prod_pi, tendsto_tsum_prod_pi, summable_prod_pi_mul_abs_eval,
        SpinGlasses.truncated_gibbs_weights, SpinGlasses.tendsto_truncated_replica_average,
        ProbabilityTheory.integrable_tsum_of_tsum_eLpNorm_ne_top}
  \leanok
Let $J$ be finite and $f_j:\alpha\to[0,\infty)$ summable. Then $\sigma\mapsto\prod_jf_j(\sigma_j)$ is summable on $J\to\alpha$. If
weights $w_m\to w$ pointwise with $0\le w_m\le B$, $B$ summable and $\sum_\sigma\prod_jB(\sigma_j)|\Psi(\sigma)|<\infty$, then
$\sum_\sigma\prod_jw_m(\sigma_j)\Psi(\sigma)\to\sum_\sigma\prod_jw(\sigma_j)\Psi(\sigma)$. In particular, for $g>0$ summable and finite
sets $A_m\uparrow\alpha$ containing $a_0$, the truncated weights $\mathbf 1_{A_m}g/\sum_{A_m}g$ are bounded by $g/g(a_0)$ and converge to $g/\sum g$. A series with
summable $L^p$ norms, $p\ge1$, is integrable (the case $p=2$ is used for the Gaussian series of Section~\ref{sec:sg:pspin}).
\end{lemma}

\begin{proof}
  \leanok
Partial sums of $\prod_jf_j(\sigma_j)$ over a finite set $T$ are bounded by $\prod_j\sum_{a\in T_j}f_j(a)\le\prod_j\sum_af_j(a)$, where $T_j$ is
the set of $j$-th coordinates of elements of $T$ (\texttt{Finset.prod\_univ\_sum}). The limit statement is Tannery's theorem with dominating function
$\prod_jB(\sigma_j)|\Psi(\sigma)|$. For the truncated weights, $\sum_{A_m}g\ge g(a_0)>0$ and $\sum_{A_m}g\to\sum g$ along the exhaustion.
\end{proof}

\begin{lemma}[Gaussian exponential moments]\label{lem:sg:gaussexp}
  \lean{ProbabilityTheory.hasGaussianLaw_hasLaw, ProbabilityTheory.integral_exp_mul_of_hasGaussianLaw,
        ProbabilityTheory.integrable_exp_mul_of_hasGaussianLaw, ProbabilityTheory.eLpNorm_one_mul_exp_le,
        ProbabilityTheory.ae_summable_and_integrable_tsum_mul_exp}
  \leanok
If $Z$ is Gaussian then $\mathbb E\,e^{tZ}=e^{t\mathbb EZ+t^2\mathrm{Var}Z/2}$. If $Z$ is centered Gaussian with $\mathbb EZ^2\le v$ and
$\mathbb EX^2\le a$ then $\mathbb E|X|e^{Z}\le\sqrt a\,e^v$. Consequently, for countably many such pairs $(X_b,Z_b)$ with uniform bounds and
$G_b\ge0$ summable, $\sum_bG_b|X_b|e^{Z_b}$ converges almost surely and is integrable.
\end{lemma}

\begin{proof}
  \leanok
  \uses{lem:sg:tannery}
The law of $Z$ is Gaussian, so the first claim is the Gaussian moment generating function. By Cauchy--Schwarz (Holder with exponents
$2,2,1$) $\|Xe^Z\|_1\le\|X\|_2\|e^Z\|_2$ and $\|e^Z\|_2^2=\mathbb Ee^{2Z}=e^{2\mathrm{Var}Z}\le e^{2v}$. The series has summable $L^1$ norms
$\le\sqrt a\,e^v\sum_bG_b$, hence converges absolutely a.s.\ and is integrable (Lemma~\ref{lem:sg:tannery}).
\end{proof}

\begin{theorem}[Finite truncations of the replica formula]\label{thm:sg:truncation}
  \lean{SpinGlasses.integral_pi_count_tilted_subtype, SpinGlasses.abs_tsum_truncated_mul_eval_le,
        SpinGlasses.abs_tsum_truncated_le, SpinGlasses.aemeasurable_tsum_prod_pi_of_forall_notMem,
        SpinGlasses.integral_truncated_gibbs_replica_average_gaussian}
  \leanok
  \uses{thm:sg:replica}
Let $\alpha$ be countable, $G>0$, $(x,y)$ a centered Gaussian process, and $A\subseteq\alpha$ a finite set. Then Theorem~\ref{thm:sg:replica}
holds for the truncated Gibbs weights $\mathbf 1_A(a)G(a)e^{y(a)}/\sum_{b\in A}G(b)e^{y(b)}$, with the replica averages written as series over
$R\to\alpha$ and $\mathrm{Option}\,R\to\alpha$. Moreover each truncated replica average is bounded by $M\sum_b(g_b/g_{a_0})|x_b|$, respectively
by the sup of the observable.
\end{theorem}

\begin{proof}
  \leanok
Apply Theorem~\ref{thm:sg:replica} on the finite type $A$ with the bias $h=\log G$. The replica average over $A^R$ of $\Psi\circ\mathrm{val}$ equals the
series over $R\to\alpha$ with the truncated weights (the weight vanishes off $A$, so the series is a finite sum over $\prod_{r}A$). The bounds are
Lemma~\ref{lem:sg:replicacalc}(3) applied on $A$.
\end{proof}

\begin{theorem}[Replica covariance formula for a countable configuration space, Panchenko Lemma~1.2]\label{thm:sg:lemma12}
  \lean{SpinGlasses.integral_gibbs_replica_average_gaussian_countable}
  \leanok
  \uses{thm:sg:truncation, lem:sg:gaussexp, lem:sg:tannery}
Let $\alpha$ be countable, $G:\alpha\to(0,\infty)$ summable, and $(x,y)$ a centered jointly Gaussian process (\texttt{IsGaussianProcess}) with
$\mathbb Ex(a)^2\le v$ and $\mathbb Ey(a)^2\le v$ for all $a$. Let $\Phi$ be a function of $|R|$ replicas with $|\Phi|\le M$, and
$w(a)=G(a)e^{y(a)}/\sum_bG(b)e^{y(b)}$ the Gibbs weights. Then
\[\mathbb E\sum_{\sigma\in\alpha^R}\prod_rw(\sigma_r)\,\Phi(\sigma)\,x(\sigma_{r_0})
 =\mathbb E\sum_{\tau\in\alpha^{R\sqcup\{*\}}}\prod_{o}w(\tau_o)\,\Phi(\tau|_R)\Big(\sum_{r}C(\tau_{r_0},\tau_r)-|R|\,C(\tau_{r_0},\tau_*)\Big).\]
\end{theorem}

\begin{proof}
  \leanok
Choose finite sets $A_m\uparrow\alpha$ containing a base point $a_1$ and apply Theorem~\ref{thm:sg:truncation} to each $A_m$; it remains to pass to the limit
$m\to\infty$ in both sides by dominated convergence. Put $g_b=G(b)e^{y(b)}$ and $Z_b=y(b)-y(a_1)$, so that $g_b/g_{a_1}=(G(b)/G(a_1))e^{Z_b}$. The $Z_b$ are centered Gaussians with
$\mathbb EZ_b^2\le4v$. By Lemma~\ref{lem:sg:gaussexp} (with $X_b=1$ and $X_b=x(b)$), almost surely $\sum_bg_b<\infty$ and $\sum_b(g_b/g_{a_1})|x(b)|<\infty$, and
$\sum_bG(b)|x(b)|e^{Z_b}$ is integrable. For such $\omega$, Lemma~\ref{lem:sg:tannery} gives the pointwise convergence of the truncated averages on
both sides (the summability hypothesis on $\Psi$ follows from the summability of $\prod B$ and of $B|x|$). The left-hand averages are dominated by
$M\sum_b(g_b/g_{a_1})|x(b)|$, which is integrable, and the right-hand averages by $M\cdot2|R|v$, because $|C(a,b)|\le(\mathbb Ex(a)^2+\mathbb Ey(b)^2)/2\le v$.
Dominated convergence shows that both sides of the truncated identities converge, hence the limits agree.
\end{proof}

Not formalized: the case of a general (non-atomic or sigma-finite) reference measure $G$ in Lemma~1.2, and the bound $\sum G\le1$ of the book
(only summability is used). Eq.~(1.29) is stated for an arbitrary $\Phi$ (the book assumes boundedness only for the countable case, as here).
\end{remark}

\section{Gaussian series and the Gaussian law of the mixed Hamiltonian}\label{sec:sg:serieslaw}

\begin{lemma}[The law of a Gaussian series]\label{lem:sg:serieslaw}
  \lean{ProbabilityTheory.isGaussian_pi_gaussianReal, ProbabilityTheory.hasGaussianLaw_coord,
        ProbabilityTheory.iIndepFun_coord, ProbabilityTheory.map_sum_range_dotProduct,
        ProbabilityTheory.summable_dotProduct_self, ProbabilityTheory.map_tsum_dotProduct_eq_gaussianReal,
        ProbabilityTheory.integral_tsum_dotProduct_eq_zero, ProbabilityTheory.sqrt_dotProduct_self_sum_le,
        ProbabilityTheory.hasGaussianLaw_tsum_dotProduct, ProbabilityTheory.ae_summable_of_tsum_eLpNorm_ne_top}
  \leanok
  \uses{lem:sg:gaussseries}
Let $g_p$, $p\in\mathbb N$, be independent vectors with i.i.d.\ $\mathcal N(0,1)$ coordinates and $a_p$ coefficient vectors with
$\sum_p\|a_p\|<\infty$. Then $H_a=\sum_pa_p\cdot g_p$ converges almost surely, is exactly Gaussian,
$H_a\sim\mathcal N(0,\sum_p\|a_p\|^2)$, and finitely many such series $(H_{a^s})_{s\in S}$ form a Gaussian vector (Mathlib's
\texttt{HasGaussianLaw}).
\end{lemma}

\begin{proof}
  \leanok
The summands $X_p=a_p\cdot g_p$ are independent centered Gaussians of variance $\|a_p\|^2$, so the partial sum $S_n=\sum_{p<n}X_p$ is Gaussian
with variance $v_n=\sum_{p<n}\|a_p\|^2$ (independent Gaussians are jointly Gaussian; the variance is the covariance formula of
Lemma~\ref{lem:sg:gaussseries} for the coefficients truncated at $n$). Since $\sum_p\|X_p\|_{L^1}\le\sum_p\|a_p\|<\infty$,
$\sum_p|X_p|<\infty$ almost surely, hence $S_n\to H_a$ almost surely. The characteristic function of $H_a$ is the limit of that of $S_n$ by
dominated convergence ($|e^{itS_n}|=1$), i.e.\ $\lim_ne^{-v_nt^2/2}=e^{-vt^2/2}$ with $v=\sum_p\|a_p\|^2$, and the characteristic function determines the law.
For the vector, a continuous linear functional $L(x)=\sum_sc_sx_s$ of $(H_{a^s})_s$ equals almost surely the single series with coefficients
$\sum_sc_sa^s_p$ (summable by Minkowski's inequality), so $L\circ(H_{a^s})_s$ is real Gaussian, which is Mathlib's criterion for a Gaussian measure.
\end{proof}

\begin{theorem}[The mixed $p$-spin Hamiltonian is a Gaussian process]\label{thm:sg:mixedgauss}
  \lean{SpinGlasses.mixedPSpin_hasGaussianLaw, SpinGlasses.integral_mixedPSpin_eq_zero,
        SpinGlasses.integral_mixedPSpin_hypercube_mul_mixedPSpin_hypercube, SpinGlasses.mixedPSpin_hypercube_hyps,
        SpinGlasses.abs_units_cast_le_one, SpinGlasses.units_cast_sq}
  \leanok
  \uses{lem:sg:serieslaw, thm:sg:mixedcov}
Let $\beta_0=0$ and $\sum_p|\beta_p|<\infty$. On the hypercube, the vector $(H_N(\sigma))_{\sigma\in\{\pm1\}^N}$ of the mixed $p$-spin
Hamiltonian is a centered Gaussian vector with $\mathbb EH_N(\sigma)H_N(\tau)=N\xi(R(\sigma,\tau))$, $\xi(x)=\sum_p\beta_p^2x^p$; each
$H_N(\sigma)$ is measurable and integrable in the disorder.
\end{theorem}

\begin{proof}
  \leanok
By definition $H_N(\sigma)=\sum_pa_p(\sigma)\cdot g_p$ with $a_p(\sigma)=\beta_pN^{-(p-1)/2}(\prod_k\sigma_{i_k})_{i}$ and
$\sum_p\|a_p(\sigma)\|\le\sqrt N\sum_p|\beta_p|$ (Theorem~\ref{thm:sg:mixedcov}). Lemma~\ref{lem:sg:serieslaw} gives the Gaussian law, the mean zero,
and Theorem~\ref{thm:sg:mixedcov} (with $|\sigma_i|\le1$) the covariance.
\end{proof}

\section{Gaussian comparison for the free energy}\label{sec:sg:comparison}

\begin{lemma}[Derivative of the expected log-partition function along a path]\label{lem:sg:pathderiv}
  \lean{SpinGlasses.hasDerivAt_log_sum_exp_path, SpinGlasses.hasDerivAt_integral_log_sum_exp_path}
  \leanok
  \uses{lem:sg:gibbsavg, thm:sg:deriv}
Let $f(s,\omega,a)$ be differentiable in $s$ on a neighbourhood of $t$ with derivative $f'(s,\omega,a)$ bounded by an integrable $B(\omega)$. Then
$\frac{d}{ds}\mathbb E\log\sum_ae^{f(s,\omega,a)}\big|_{s=t}=\mathbb E\langle f'(t)\rangle_{f(t)}$, where $\langle\cdot\rangle_{f(t)}$ is the
Gibbs average of the energy $f(t,\omega,\cdot)$. The temperature derivative of Theorem~\ref{thm:sg:deriv} is the case $f(s,\omega,a)=sH(\omega,a)$.
\end{lemma}

\begin{proof}
  \leanok
Pointwise, $\frac{d}{ds}\log\sum_ae^{f}=\sum_ae^{f}f'/\sum_ae^{f}=\langle f'\rangle$; differentiate under the integral by dominated convergence
(\texttt{hasDerivAt\_integral\_of\_dominated\_loc\_of\_deriv\_le}), the Gibbs average being bounded by $\sup_a|f'|\le B$.
\end{proof}

\begin{theorem}[Gaussian comparison, the smooth Slepian inequality]\label{thm:sg:comparison}
  \lean{SpinGlasses.hasGaussianLaw_sumElim_lincomb, SpinGlasses.integral_lincomb_mul_lincomb,
        SpinGlasses.hasDerivAt_integral_log_sum_exp_trig, SpinGlasses.integral_log_sum_exp_le_of_covariance_le}
  \leanok
  \uses{lem:sg:pathderiv, thm:sg:lemma11}
Let $(X,Y)$ be a centered jointly Gaussian pair of fields on a finite set $\alpha$ with $\mathbb EX(a)Y(b)=0$, equal variances
$\mathbb EX(a)^2=\mathbb EY(a)^2$, and $\mathbb EX(a)X(b)\le\mathbb EY(a)Y(b)$ for all $a,b$. Then for every bias $h:\alpha\to\mathbb R$,
\[\mathbb E\log\sum_ae^{Y(a)+h(a)}\le\mathbb E\log\sum_ae^{X(a)+h(a)}.\]
\end{theorem}

\begin{proof}
  \leanok
Put $u_s=\cos s\,X+\sin s\,Y+h$ and $\varphi(s)=\mathbb E\log\sum_ae^{u_s(a)}$. By Lemma~\ref{lem:sg:pathderiv}, $\varphi'(s)=\mathbb E\langle x_s\rangle$
with $x_s=-\sin s\,X+\cos s\,Y=u_s'$. The pair $(x_s,u_s)$ is jointly Gaussian and
$C(a,b)=\mathbb Ex_s(a)u_s(b)=\sin s\cos s\,(B-A)(a,b)$ with $A=\mathrm{Cov}X$, $B=\mathrm{Cov}Y$, because $\mathbb EXY=0$. Gaussian integration by
parts for the Gibbs measure (Lemma~1.1, Theorem~\ref{thm:sg:lemma11}) gives
$\varphi'(s)=\mathbb E\langle C(\sigma^1,\sigma^1)-C(\sigma^1,\sigma^2)\rangle=-\sin s\cos s\,\mathbb E\langle(B-A)(\sigma^1,\sigma^2)\rangle\le0$
on $[0,\pi/2]$, since $(B-A)(a,a)=0$ and $B\ge A$. Hence $\varphi(\pi/2)\le\varphi(0)$, which is the claim. (The trigonometric path is smooth on all of
$\mathbb R$, so no endpoint continuity or singular derivative at $t=0,1$ occurs, unlike $\sqrt tX+\sqrt{1-t}Y$.)
\end{proof}

\section{The Guerra--Toninelli theorem: superadditivity and the thermodynamic limit}\label{sec:sg:gt}

\begin{lemma}[Fekete's lemma for superadditive sequences, Panchenko Lemma~1.3]\label{lem:sg:fekete}
  \lean{tendsto_div_atTop_sSup_of_superadditive}
  \leanok
If $u_{m+n}\ge u_m+u_n$ for all $m,n$ and $u_n/n$ is bounded above, then $u_n/n\to\sup_{n\ge1}u_n/n$.
\end{lemma}

\begin{proof}
  \leanok
The sequence $-u$ is subadditive; apply Mathlib's Fekete lemma (\texttt{Subadditive.tendsto\_lim}) and negate the limit $\inf_n(-u_n)/n$.
No definition of superadditivity is introduced.
\end{proof}

\begin{theorem}[Superadditivity of the free energy, Guerra--Toninelli, Panchenko Eq.~(1.36)]\label{thm:sg:superadd}
  \lean{SpinGlasses.add_mul_apply_le_of_convexOn, SpinGlasses.dotProduct_cast_append,
        SpinGlasses.dotProduct_units_cast_self, SpinGlasses.integrable_log_sum_exp_of_hasGaussianLaw,
        SpinGlasses.integral_log_sum_exp_add_le, ProbabilityTheory.HasGaussianLaw.prodMk_prod}
  \leanok
  \uses{thm:sg:comparison}
Let $H_N$ be centered Gaussian Hamiltonians on $\{\pm1\}^N$ (disorders on arbitrary probability spaces $\Omega_N$) with
$\mathbb EH_N(\sigma)H_N(\tau)=N\xi(R(\sigma,\tau))$ for a function $\xi$ that is convex on $[-1,1]$. Then for $N,M>0$,
\[\mathbb E\log Z_N+\mathbb E\log Z_M\le\mathbb E\log Z_{N+M},\qquad\text{i.e.}\quad NF_N+MF_M\le(N+M)F_{N+M}.\]
\end{theorem}

\begin{proof}
  \leanok
On $\Omega_{N+M}\times(\Omega_N\times\Omega_M)$ put $X(\rho,\tau)=H_{N+M}(\rho\frown\tau)$ (concatenation of the configurations) and
$Y(\rho,\tau)=H_N(\rho)+H_M(\tau)$. The triple of Hamiltonians is jointly Gaussian (a product of Gaussian measures), and $X$ is independent of $Y$, so
$\mathbb EXY=0$. The overlap splits, $R(\rho\frown\tau,\rho'\frown\tau')=\frac{N}{N+M}R(\rho,\rho')+\frac{M}{N+M}R(\tau,\tau')$, hence
$\mathbb EX(\rho,\tau)X(\rho',\tau')=(N+M)\xi\big(\frac{NR+MR'}{N+M}\big)\le N\xi(R)+M\xi(R')=\mathbb EY(\rho,\tau)Y(\rho',\tau')$ by convexity of $\xi$,
and both variances equal $(N+M)\xi(1)$. Theorem~\ref{thm:sg:comparison} gives $\mathbb E\log\sum e^Y\le\mathbb E\log\sum e^X$; the right side is
$\mathbb E\log Z_{N+M}$ and the left side is $\mathbb E\log(Z_NZ_M)=\mathbb E\log Z_N+\mathbb E\log Z_M$.
\end{proof}

\begin{lemma}[Convexity of $\xi$]\label{lem:sg:convexxi}
  \lean{SpinGlasses.convexOn_sum_sq_mul_pow}
  \leanok
If $\sum_p|\beta_p|<\infty$ and $\beta_p=0$ for all odd $p\ge3$, then $\xi(x)=\sum_p\beta_p^2x^p$ is convex on $[-1,1]$.
\end{lemma}

\begin{proof}
  \leanok
Each term is convex: $x^p$ for even $p$, the linear function $x$ for $p=1$, and the term vanishes for odd $p\ge3$. The inequality
$\xi(ax+by)\le a\xi(x)+b\xi(y)$ passes to the sum of the convergent series.
\end{proof}

\begin{theorem}[Existence of the thermodynamic limit, Guerra--Toninelli, Panchenko Theorem~1.1]\label{thm:sg:thermolimit}
  \lean{SpinGlasses.integral_log_le_log_integral, SpinGlasses.integral_log_sum_exp_le,
        SpinGlasses.tendsto_free_energy, SpinGlasses.tendsto_free_energy_mul,
        SpinGlasses.sk_tendsto_free_energy, SpinGlasses.sk_tendsto_ground_state_energy,
        SpinGlasses.sk_monotoneOn_free_energy_limit, SpinGlasses.mixedPSpin_tendsto_free_energy,
        SpinGlasses.mixedPSpin_tendsto_ground_state_energy}
  \leanok
  \uses{thm:sg:superadd, lem:sg:fekete, lem:sg:convexxi, thm:sg:mixedgauss, thm:sg:zerotemp, thm:sg:skcov}
Under the hypotheses of Theorem~\ref{thm:sg:superadd}, for every inverse temperature $t$ the free energy
$F_N(t)=N^{-1}\mathbb E\log Z_N(t)$ converges to $\sup_{N\ge1}F_N(t)$. This applies to the SK model
(covariance function $\xi(x)=x^2$) and to the mixed $p$-spin models with $\beta_0=0$, $\sum_p|\beta_p|<\infty$ and $\beta_p=0$ for odd
$p\ge3$ ($\xi(x)=\sum_p\beta_p^2x^p$). Consequently the hypothesis \texttt{h\_lim} of Theorem~\ref{thm:sg:zerotemp} holds: for these models
$N^{-1}\mathbb E\max_\sigma H_N(\sigma)$ converges, and for SK the function $\beta\mapsto\beta^{-1}(F(\beta)-\log2)$ is non-decreasing.
\end{theorem}

\begin{proof}
  \leanok
The Hamiltonian $tH_N$ has covariance $N(t^2\xi)(R)$ and $t^2\xi$ is convex, so we may take $t=1$. Put $u_N=\mathbb E\log Z_N$ ($u_0=0$); it is
superadditive by Theorem~\ref{thm:sg:superadd}. It is bounded above: by Jensen's inequality for the logarithm, $\mathbb E\log Z_N\le\log\mathbb EZ_N
=\log\sum_\sigma e^{\mathrm{Var}H_N(\sigma)/2}=N(\log2+\xi(1)/2)$, since $\mathrm{Var}\,H_N(\sigma)=N\xi(1)$. Fekete's lemma
(Lemma~\ref{lem:sg:fekete}) shows $u_N/N\to\sup_{N\ge1}u_N/N$. For SK, $\xi(x)=x^2$ is convex and the covariance is Theorem~\ref{thm:sg:skcov}; the
Gaussian law is that of the linear image $G\mapsto(\sigma^\top G\sigma)_\sigma$ of the Gaussian matrix. For the mixed model use
Lemma~\ref{lem:sg:convexxi} and Theorem~\ref{thm:sg:mixedgauss}. The consequences follow from Theorem~\ref{thm:sg:zerotemp}.
\end{proof}

\begin{remark}
The identification of the SK Hamiltonian with the pure $p=2$ model (which would need the equivalence of the nested disorder
\texttt{Fin N -> Fin N -> R} with the flat disorder \texttt{(Fin 2 -> Fin N) -> R}) is not needed: the interpolation sees the Hamiltonian only through
its Gaussian law and covariance function $\xi$, and the SK covariance $N R^2$ is that of $\xi(x)=x^2$ directly. The hypothesis $\beta_p=0$ for odd $p\ge3$
is exactly the convexity of $\xi$ on $[-1,1]$; Theorem~\ref{thm:sg:superadd} holds for any convex $\xi$.
\end{remark}
